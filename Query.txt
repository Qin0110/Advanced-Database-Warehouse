-- Function
CREATE OR REPLACE FUNCTION fn_reorder_quantity (
    p_branch_id IN Branch.branch_id%TYPE,
    p_item_id   IN Item.item_id%TYPE
) RETURN NUMBER
IS
    v_on_hand   NUMBER;
    v_daily     NUMBER;
    v_lead      NUMBER;
    v_cover_days NUMBER;
    v_needed    NUMBER;

BEGIN
    SELECT bs.quantity_on_hand,
           bs.units_sold_6m / 182,
           bs.lead_time_days,
           bs.days_of_cover
    INTO   v_on_hand,
           v_daily,
           v_lead,
           v_cover_days
    FROM   v_branch_stock_status bs
    WHERE  bs.branch_id = p_branch_id
    AND    bs.item_id   = p_item_id;

    v_needed :=
        (v_daily * (v_cover_days + v_lead))
        - v_on_hand;

    IF v_needed <= 0 THEN
        RETURN 0;
    END IF;

    RETURN CEIL(v_needed / 12) * 12;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RETURN 0;

END fn_reorder_quantity;

-- View Table 1
CREATE OR REPLACE VIEW v_delivery_performance AS
	SELECT dc.delivery_company_id, dc.company_name,  dc.base_delivery_fee,  dc.status AS status,
		EXTRACT(YEAR FROM o.order_date_time) AS year,
		COUNT(DISTINCT c.courier_id) AS no_riders,
		COUNT(dr.delivery_request_id) AS no_jobs_taken,
		SUM(CASE WHEN dr.status = 'Delivered' THEN 1 ELSE 0 END) AS delivery_success,
		SUM(CASE WHEN dr.status = 'Failed' THEN 1 ELSE 0 END) AS delivery_failed,
		ROUND(SUM(CASE WHEN dr.status = 'Delivered' THEN 1 ELSE 0 END) * 100 / NULLIF(SUM(CASE WHEN dr.status IN ('Delivered', 'Failed') THEN 1 ELSE 0 END), 0), 1) AS success_rate, 	
		ROUND(AVG(c.average_rating), 2) AS company_avg_rating,
		ROUND(AVG((CAST(dr.delivered_date_time AS DATE) - CAST(o.order_date_time AS DATE)) * 1440), 0) AS avg_minutes,
		SUM(dr.delivery_fee) AS fee_paid,
		SUM(o.total_amount) AS total_order_value
	FROM Delivery_Company dc
		JOIN Courier c ON c.delivery_company_id = dc.delivery_company_id
		JOIN Delivery_Request dr ON dr.courier_id = c.courier_id
		JOIN Orders o ON o.order_no = dr.order_no 
	GROUP BY dc.delivery_company_id, dc.company_name, dc.status, dc.base_delivery_fee, EXTRACT(YEAR FROM o.order_date_time);


-- View Table 2
CREATE OR REPLACE VIEW v_branch_stock_status AS 
WITH Past_6_Months_Sales AS (
    SELECT o.branch_id, 
           od.item_id, 
           SUM(od.quantity) AS units_6m
    FROM   Order_Detail od
           JOIN Orders o ON o.order_no = od.order_no
    WHERE  o.order_status = 'Completed'
    AND    o.order_date_time >= ADD_MONTHS((SELECT MAX(order_date_time) FROM Orders), -6)
    GROUP  BY o.branch_id, od.item_id
)
SELECT inv.branch_id, 
b.name AS branch_name, 
i.item_id,
       	i.name AS item_name, 
       	i.unit_price, 
	i.category,
       	i.status AS item_status, 
       	inv.quantity AS quantity_on_hand, 
       	inv.last_restock_date, 
       	inv.quantity * i.unit_price AS stock_value, 
       	s.supplier_id, 
	s.company_name AS supplier_name,
	s.contact_no AS supplier_phone,
       	s.lead_time_days, 
       	s.status AS supplier_status, 
       	CASE WHEN inv.quantity = 0 THEN 'Out of Stock'
            	WHEN inv.quantity < 20 THEN 'Critical'
            	WHEN inv.quantity < 100 THEN 'Low'
            	ELSE 'Good' 
       	END AS stock_level,
       	NVL(sold.units_6m, 0) AS units_sold_6m, 
       	CASE WHEN NVL(sold.units_6m, 0) = 0 THEN NULL 
            	ELSE ROUND(inv.quantity / (sold.units_6m / 182), 0) 
       	END AS days_of_cover,
       	ROUND(NVL(sold.units_6m, 0) / 26, 1) AS avg_weekly_demand
FROM   Inventory inv 
JOIN Item i ON i.item_id = inv.item_id
       	JOIN Branch b ON b.branch_id = inv.branch_id
       	JOIN Supplier s ON s.supplier_id = i.supplier_id 
       	LEFT JOIN Past_6_Months_Sales sold ON sold.branch_id = inv.branch_id AND sold.item_id = inv.item_id;

CREATE OR REPLACE PROCEDURE prc_restock_item (
    p_branch_id IN Branch.branch_id%TYPE,
    p_item_id   IN Item.item_id%TYPE,
    p_quantity  IN NUMBER
) AS
    e_invalid_quantity EXCEPTION;
    e_branch_closed    EXCEPTION;

    v_branch_name   Branch.name%TYPE;
    v_branch_status Branch.status%TYPE;
    v_item_name     Item.name%TYPE;
    v_supplier      Supplier.company_name%TYPE;
    v_lead          Supplier.lead_time_days%TYPE;
    v_before        Inventory.quantity%TYPE;
    v_after         Inventory.quantity%TYPE;

BEGIN
    SELECT b.name, b.status
    INTO   v_branch_name, v_branch_status
    FROM   Branch b
    WHERE  b.branch_id = p_branch_id;

    SELECT i.name, s.company_name, s.lead_time_days
    INTO   v_item_name, v_supplier, v_lead
    FROM   Item i
           JOIN Supplier s
             ON s.supplier_id = i.supplier_id
    WHERE  i.item_id = p_item_id;

    IF p_quantity IS NULL
       OR p_quantity <= 0
       OR p_quantity > 1000000 THEN
        RAISE e_invalid_quantity;
    END IF;

    IF v_branch_status <> 'Active' THEN
        RAISE e_branch_closed;
    END IF;

    UPDATE Inventory
    SET    quantity = quantity + p_quantity,
           last_restock_date = SYSDATE
    WHERE  branch_id = p_branch_id
    AND    item_id = p_item_id;

    IF SQL%ROWCOUNT = 0 THEN

        v_before := 0;

        INSERT INTO Inventory (
            branch_id,
            item_id,
            quantity,
            last_restock_date
        )
        VALUES (
            p_branch_id,
            p_item_id,
            p_quantity,
            SYSDATE
        );

        DBMS_OUTPUT.PUT_LINE(
            'New product line opened at ' || v_branch_name || '.'
        );

    ELSE
        SELECT quantity
        INTO   v_after
        FROM   Inventory
        WHERE  branch_id = p_branch_id
        AND    item_id = p_item_id;

        v_before := v_after - p_quantity;
    END IF;

    COMMIT;

    DBMS_OUTPUT.PUT_LINE(
        '---------------------------------------------------------------'
    );
    DBMS_OUTPUT.PUT_LINE('RESTOCK SUMMARY');

    DBMS_OUTPUT.PUT_LINE(
        '  Branch   : ' || p_branch_id || ' - ' || v_branch_name
    );

    DBMS_OUTPUT.PUT_LINE(
        '  Product  : ' || p_item_id || ' - ' || v_item_name
    );

    DBMS_OUTPUT.PUT_LINE(
        '  Supplier : ' || v_supplier ||
        ' (lead time ' || v_lead || ' day(s))'
    );

    DBMS_OUTPUT.PUT_LINE(
        '  Received : ' || p_quantity || ' unit(s)'
    );

    DBMS_OUTPUT.PUT_LINE(
        '  On hand  : ' || v_before ||
        ' -> ' || (v_before + p_quantity)
    );

    DBMS_OUTPUT.PUT_LINE(
        '  Next suggested order : ' ||
        fn_reorder_quantity(p_branch_id, p_item_id) ||
        ' unit(s)'
    );

    DBMS_OUTPUT.PUT_LINE(
        '---------------------------------------------------------------'
    );

EXCEPTION
    WHEN e_invalid_quantity THEN
        RAISE_APPLICATION_ERROR(
            -20000,
            'The quantity received must be a positive number no greater than ' ||
            '1,000,000 units. ' ||
            NVL(TO_CHAR(p_quantity), 'NULL') ||
            ' was given.'
        );

    WHEN e_branch_closed THEN
        RAISE_APPLICATION_ERROR(
            -20001,
            v_branch_name || ' is ' ||
            v_branch_status ||
            ' and cannot receive stock.'
        );

    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(
            -20003,
            'Branch ' || p_branch_id ||
            ' or item ' || p_item_id ||
            ' does not exist.'
        );

END prc_restock_item;

-- Procedure 2
CREATE OR REPLACE PROCEDURE prc_dispatch_delivery (
	p_order_no IN Orders.order_no%TYPE,
    	p_company_id IN Delivery_Company.delivery_company_id%TYPE,
    	p_recipient_name  IN Delivery_Request.recipient_name%TYPE,
    	p_recipient_phone IN Delivery_Request.recipient_phone_no%TYPE,
    	p_address IN Delivery_Request.delivery_address%TYPE
) AS
   	
 
    	CURSOR c_best_rider IS
        	SELECT c.courier_id, c.name, c.average_rating
       		FROM Courier c
        	WHERE c.delivery_company_id = p_company_id
			AND NOT EXISTS ( SELECT 1 FROM Delivery_Request dr 
			WHERE dr.courier_id = c.courier_id AND dr.status = 'Processing'
        	ORDER BY c.average_rating DESC NULLS LAST, c.courier_id;

 
    	v_order_type Orders.order_type%TYPE;
    	v_service Orders.service_type%TYPE;
    	v_status Orders.order_status%TYPE;
    	v_partner Delivery_Company.company_name%TYPE;
    	v_partner_stat Delivery_Company.status%TYPE;
    	v_base_fee Delivery_Company.base_delivery_fee%TYPE;
    	v_courier Courier.courier_id%TYPE;
    	v_rider_name Courier.name%TYPE;
    	v_rating Courier.average_rating%TYPE;
    	v_ref Delivery_Request.delivery_request_id%TYPE;
    	v_fee Delivery_Request.delivery_fee%TYPE;
BEGIN
    	SELECT o.order_type, o.service_type, o.order_status
    	INTO v_order_type, v_service, v_status
    	FROM Orders o WHERE o.order_no = p_order_no;

    	IF v_status <> 'Processing' 
		THEN RAISE_APPLICATION_ERROR(-20004, 'Order ' || p_order_no || ' is ' || v_status || '. Only an order still being processed can be dispatched.'); 
	END IF;
 
    	SELECT dc.company_name, dc.status, dc.base_delivery_fee
    	INTO   v_partner, v_partner_stat, v_base_fee
    	FROM   Delivery_Company dc WHERE dc.delivery_company_id = p_company_id;
 
    	IF v_partner_stat <> 'Active' 
		THEN RAISE_APPLICATION_ERROR(-20005, v_partner || ' is no longer under contract. Choose another partner.');
    	END IF;
 
    	OPEN c_best_rider;
    		FETCH c_best_rider INTO v_courier, v_rider_name, v_rating;
    		IF c_best_rider%NOTFOUND 
			THEN CLOSE c_best_rider;
        		RAISE_APPLICATION_ERROR(-20324, v_partner || ' has no rider on file to take this job.');
    		END IF;
    	CLOSE c_best_rider;
 
   	v_ref := 'DR' || TO_CHAR(delivery_request_seq.NEXTVAL, 'FM000000');
 
    	INSERT INTO Delivery_Request (delivery_request_id, recipient_name, recipient_phone_no, delivery_address, delivery_fee, delivered_date_time, status, rating, order_no, courier_id)
    	VALUES (v_ref, p_recipient_name, p_recipient_phone, p_address, NULL, NULL, 'Processing', NULL, p_order_no, v_courier);
 
    	SELECT delivery_fee INTO v_fee
    	FROM   Delivery_Request WHERE delivery_request_id = v_ref;
 
    	COMMIT;
 
    	DBMS_OUTPUT.PUT_LINE('-----------------------------------------------------------------------');
    	DBMS_OUTPUT.PUT_LINE('DISPATCH NOTE ' || v_ref);
    	DBMS_OUTPUT.PUT_LINE('Order     : ' || p_order_no || '  (' || v_order_type || ')');
    	DBMS_OUTPUT.PUT_LINE('Partner   : ' || v_partner);
    	DBMS_OUTPUT.PUT_LINE('Rider     : ' || v_courier || '  ' || v_rider_name || '(rating: ' || NVL(TO_CHAR(v_rating, '0.00'), ' n/a') || ')' );
    	DBMS_OUTPUT.PUT_LINE('Deliver to: ' || p_recipient_name || '  ' || p_recipient_phone);
    	DBMS_OUTPUT.PUT_LINE('Address   : ' || p_address);
    	DBMS_OUTPUT.PUT_LINE('Fee       : RM ' || TO_CHAR(v_fee, '990.00') || ' (base RM ' || TO_CHAR(v_base_fee, '990.00') || ' plus the weight of the basket)');
    	DBMS_OUTPUT.PUT_LINE('Promise   : delivered within 2 hours, by ' || TO_CHAR(SYSDATE + 2/24, 'HH24:MI'));
    	DBMS_OUTPUT.PUT_LINE('-----------------------------------------------------------------------');
	EXCEPTION
    		WHEN NO_DATA_FOUND 
		THEN RAISE_APPLICATION_ERROR(-20325, 'Order ' || p_order_no || ' or delivery company ' || p_company_id ||' does not exist.');
    	WHEN DUP_VAL_ON_INDEX 
		THEN RAISE_APPLICATION_ERROR(-20326, 'Order ' || p_order_no || ' has already been dispatched.');
    	

END prc_dispatch_delivery;
/

CREATE OR REPLACE PROCEDURE rpt_delivery_fulfilment (
    	p_year IN NUMBER DEFAULT NULL
) AS
    	CURSOR c_year (cp_year NUMBER) IS
        	SELECT DISTINCT year
        	FROM v_delivery_performance
        	WHERE (cp_year IS NULL OR year = cp_year)
        	ORDER BY year DESC;

    	CURSOR c_partner (cp_year NUMBER) IS
        	SELECT delivery_company_id, company_name, no_jobs_taken, success_rate, fee_paid
        	FROM v_delivery_performance
        	WHERE year = cp_year
        	ORDER BY no_jobs_taken DESC;

    	CURSOR c_rider (cp_company Delivery_Company.delivery_company_id%TYPE, cp_year NUMBER) IS
        	SELECT c.name,
               		NVL(c.average_rating, 0) AS rating,
               		COUNT(dr.delivery_request_id) AS jobs,
               		SUM(CASE WHEN dr.status = 'Delivered' THEN 1 ELSE 0 END) AS delivered,
               		SUM(CASE WHEN dr.status = 'Failed' THEN 1 ELSE 0 END) AS failed
        	FROM Courier c
               		JOIN Delivery_Request dr ON dr.courier_id = c.courier_id
               		JOIN Orders o ON o.order_no = dr.order_no
        	WHERE c.delivery_company_id = cp_company
        		AND EXTRACT(YEAR FROM o.order_date_time) = cp_year
        		GROUP BY c.name, c.average_rating
        		ORDER BY jobs DESC;

    	v_total_years   NUMBER := 0;
    	v_year_partners NUMBER;
	BEGIN
    		IF p_year IS NOT NULL AND p_year NOT BETWEEN 2020 AND 2026 THEN
        		RAISE_APPLICATION_ERROR(-20341, 'The trading history runs from 2020 to 2026. Year ' || p_year || ' is outside it.');
    		END IF;

    		DBMS_OUTPUT.PUT_LINE(RPAD('=', 85, '='));
    		IF p_year IS NULL THEN
        		DBMS_OUTPUT.PUT_LINE('88 SPEEDMART   MULTI-YEAR DELIVERY FULFILMENT AND RIDER REPORT (ALL YEARS)');
    		ELSE
        		DBMS_OUTPUT.PUT_LINE('88 SPEEDMART   DELIVERY FULFILMENT AND RIDER REPORT FOR YEAR ' || p_year);
    		END IF;
    			DBMS_OUTPUT.PUT_LINE('Generated on ' || TO_CHAR(SYSDATE, 'DD-MON-YYYY HH24:MI'));
    			DBMS_OUTPUT.PUT_LINE(RPAD('=', 85, '='));

    		FOR y IN c_year(p_year) LOOP
        		v_total_years := v_total_years + 1;
        		v_year_partners := 0;

        		DBMS_OUTPUT.PUT_LINE(CHR(10) || '---------------------------------------------------------------------------------');
        		DBMS_OUTPUT.PUT_LINE('>> PERFORMANCE REPORT FOR YEAR: ' || y.year);
        		DBMS_OUTPUT.PUT_LINE('---------------------------------------------------------------------------------');

        		FOR p IN c_partner(y.year) LOOP
            			v_year_partners := v_year_partners + 1;

            			DBMS_OUTPUT.PUT_LINE(CHR(10) || '  Partner: ' || p.company_name ||
                                 ' | Jobs: ' || p.no_jobs_taken ||
                                 ' | Success: ' || p.success_rate || '%' ||
                                 ' | Fees: RM ' || p.fee_paid);
            			DBMS_OUTPUT.PUT_LINE('  ' || RPAD('-', 75, '-'));
            			DBMS_OUTPUT.PUT_LINE('  ' || RPAD('Rider Name', 28) || LPAD('Rating', 8) || 
                                 LPAD('Jobs', 8) || LPAD('Delivered', 12) || LPAD('Failed', 8));
            			DBMS_OUTPUT.PUT_LINE('  ' || RPAD('-', 75, '-'));

            			FOR r IN c_rider(p.delivery_company_id, y.year) LOOP
                			DBMS_OUTPUT.PUT_LINE('    ' || RPAD(SUBSTR(r.name, 1, 26), 28) ||
                                     	LPAD(TO_CHAR(r.rating, '9.0'), 8) ||
                                     	LPAD(r.jobs, 8) ||
                                     	LPAD(r.delivered, 12) ||
                                     	LPAD(r.failed, 8));
            			END LOOP;
        		END LOOP;

        		IF v_year_partners = 0 THEN
            			DBMS_OUTPUT.PUT_LINE('  No delivery data recorded for ' || y.year);
        		END IF;
    		END LOOP;

    	DBMS_OUTPUT.PUT_LINE(CHR(10) || RPAD('=', 85, '='));
    	DBMS_OUTPUT.PUT_LINE('END OF REPORT. Evaluated ' || v_total_years || ' year(s) of history.');
    	DBMS_OUTPUT.PUT_LINE(RPAD('=', 85, '='));
END rpt_delivery_fulfilment;

CREATE OR REPLACE PROCEDURE rpt_mykasih_subsidy (
    	p_year IN NUMBER DEFAULT NULL
) AS
    	CURSOR c_year (cp_year NUMBER) IS
        	SELECT DISTINCT EXTRACT(YEAR FROM o.order_date_time) AS order_year
        	FROM Orders o
        	WHERE o.is_mykasih_purchase = 'Y'
        	AND o.order_status = 'Completed'
        	AND (cp_year IS NULL OR EXTRACT(YEAR FROM o.order_date_time) = cp_year)
        	ORDER BY order_year DESC;

    	CURSOR c_supplier IS 
        	SELECT s.supplier_id, s.company_name, s.contact_no
        	FROM Supplier s
        	WHERE EXISTS (
            		SELECT 1 
            		FROM Item i 
            		WHERE i.supplier_id = s.supplier_id 
            		AND i.is_mykasih_eligible = 'Y'
        		)
        	ORDER  BY s.company_name;

    	CURSOR c_item (cp_supplier Supplier.supplier_id%TYPE, cp_year NUMBER) IS
        	SELECT i.item_id, 
               		i.name, 
               		i.category,
               		(SELECT NVL(SUM(quantity), 0) 
                FROM Inventory 
                WHERE item_id = i.item_id) AS total_stock,
               		NVL(SUM(od.quantity), 0) AS mykasih_units_sold,
               		NVL(SUM(od.subtotal), 0) AS mykasih_revenue
        	FROM Item i
               		LEFT JOIN Order_Detail od ON i.item_id = od.item_id
               		LEFT JOIN Orders o ON od.order_no = o.order_no 
                                  AND o.is_mykasih_purchase = 'Y' 
                                  AND o.order_status = 'Completed'
                                  AND EXTRACT(YEAR FROM o.order_date_time) = cp_year
        	WHERE i.supplier_id = cp_supplier
        		AND i.is_mykasih_eligible = 'Y'
        		AND i.status = 'Active'
        	GROUP BY i.item_id, i.name, i.category
        	HAVING NVL(SUM(od.subtotal), 0) > 0  -- Only show items that had MyKasih sales in this year
        	ORDER BY mykasih_revenue DESC, i.name;

    	v_total_years NUMBER := 0;
    	v_suppliers NUMBER;
    	v_items NUMBER;
    	v_sup_revenue NUMBER;
    	v_year_revenue NUMBER;
    	v_grand_revenue NUMBER := 0;
BEGIN
    	IF p_year IS NOT NULL AND p_year NOT BETWEEN 2020 AND 2026 THEN
        	RAISE_APPLICATION_ERROR(-20341, 'The trading history runs from 2020 to 2026. Year ' || p_year || ' is outside it.');
    	END IF;

    	DBMS_OUTPUT.PUT_LINE(RPAD('=', 105, '='));
    	IF p_year IS NULL THEN
        	DBMS_OUTPUT.PUT_LINE('88 SPEEDMART   MULTI-YEAR MYKASIH WELFARE SUBSIDY REPORT (ALL YEARS)');
    	ELSE
        	DBMS_OUTPUT.PUT_LINE('88 SPEEDMART   MYKASIH WELFARE SUBSIDY REPORT FOR YEAR ' || p_year);
    	END IF;
    	DBMS_OUTPUT.PUT_LINE('Tracking sales of essential goods purchased using MyKasih government funds.');
    	DBMS_OUTPUT.PUT_LINE(RPAD('=', 105, '='));

    	FOR y IN c_year(p_year) LOOP
        	v_total_years := v_total_years + 1;
        	v_suppliers := 0;
        	v_items := 0;
        	v_year_revenue := 0;

        DBMS_OUTPUT.PUT_LINE(CHR(10) || '========================================================================================');
        DBMS_OUTPUT.PUT_LINE('>> SUBSIDY AUDIT FOR YEAR: ' || y.order_year);
        DBMS_OUTPUT.PUT_LINE('========================================================================================');

        FOR s IN c_supplier LOOP
            	v_sup_revenue := 0;

            	DECLARE
                	v_printed_header BOOLEAN := FALSE;
            	BEGIN
                	FOR i IN c_item(s.supplier_id, y.order_year) LOOP
                    		IF NOT v_printed_header THEN
                        		v_suppliers := v_suppliers + 1;
                        		DBMS_OUTPUT.PUT_LINE(CHR(10) || 'Supplier: ' || s.company_name || '   [Contact: ' || s.contact_no || ']');
                        		DBMS_OUTPUT.PUT_LINE('  ' || RPAD('Product Name', 30) || 
                                             RPAD('Category', 24) || 
                                             LPAD('Total Stock', 12) || 
                                             LPAD('Sold (MK)', 12) || 
                                             LPAD('Subsidy (RM)', 18));
                        		DBMS_OUTPUT.PUT_LINE('  ' || RPAD('-', 98, '-'));
                        		v_printed_header := TRUE;
                    		END IF;

                    		v_items := v_items + 1;
                    		v_sup_revenue := v_sup_revenue + i.mykasih_revenue;

                    		DBMS_OUTPUT.PUT_LINE('    ' || RPAD(SUBSTR(i.name, 1, 28), 30) ||
                                         RPAD(i.category, 24) ||
                                         LPAD(i.total_stock, 12) ||
                                         LPAD(i.mykasih_units_sold, 12) ||
                                         LPAD(TO_CHAR(i.mykasih_revenue, '999,990.00'), 18));
                	END LOOP;

                	IF v_printed_header THEN
                    		DBMS_OUTPUT.PUT_LINE('  ' || RPAD('-', 98, '-'));
                    		DBMS_OUTPUT.PUT_LINE(LPAD('Supplier MyKasih Total: RM ' || TO_CHAR(v_sup_revenue, '999,990.00'), 98));
                    		v_year_revenue := v_year_revenue + v_sup_revenue;
                	END IF;
            	END;
        END LOOP;

        DBMS_OUTPUT.PUT_LINE(CHR(10) || RPAD('-', 105, '-'));
        DBMS_OUTPUT.PUT_LINE('YEAR ' || y.order_year || ' SUMMARY -> ACTIVE SUPPLIERS: ' || v_suppliers || 
                             '   ITEMS SOLD: ' || v_items || 
                             '   TOTAL CLAIM: RM ' || TO_CHAR(v_year_revenue, '9,999,990.00'));
        DBMS_OUTPUT.PUT_LINE(RPAD('-', 105, '-'));

        v_grand_revenue := v_grand_revenue + v_year_revenue;
    	END LOOP;

    	DBMS_OUTPUT.PUT_LINE(CHR(10) || RPAD('=', 105, '='));
    	IF v_total_years = 0 THEN
        	DBMS_OUTPUT.PUT_LINE('No MyKasih transaction data found for the selected criteria.');
    	ELSE
        	DBMS_OUTPUT.PUT_LINE('GRAND TOTAL MYKASIH SUBSIDY CLAIM ACROSS ' || v_total_years || ' YEAR(S): RM ' || TO_CHAR(v_grand_revenue, '99,999,990.00'));
    	END IF;
    	DBMS_OUTPUT.PUT_LINE(RPAD('=', 105, '='));
END rpt_mykasih_subsidy;

CREATE OR REPLACE PROCEDURE proc_redeem_reward (
    p_member_id     IN  VARCHAR2,
    p_reward_id     IN  VARCHAR2,
    p_quantity      IN  NUMBER   DEFAULT 1,
    p_redemption_id OUT VARCHAR2
) IS
    v_member_name     Member.name%TYPE;
    v_reward_name     Reward.name%TYPE;
    v_points_required Reward.points_required%TYPE;
 
    v_cost            NUMBER;
    v_issue_date      DATE := SYSDATE;
    v_expiry_date     Reward_Redemption.expiry_date%TYPE;
    v_balance_after   Member.points_balance%TYPE;
BEGIN
    IF p_quantity IS NULL OR p_quantity < 1 THEN
        RAISE_APPLICATION_ERROR(-20020,
            'Redemption quantity must be at least 1.');
    END IF;

    BEGIN
        SELECT name INTO v_member_name
        FROM   Member WHERE member_id = p_member_id;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20021,
                'Member ' || p_member_id || ' does not exist.');
    END;
 

    BEGIN
        SELECT name, points_required INTO v_reward_name, v_points_required
        FROM   Reward WHERE reward_id = p_reward_id;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20023,
                'Reward ' || p_reward_id || ' does not exist.');
    END;
 
    v_cost := v_points_required * p_quantity;
    p_redemption_id := 'RD' || TO_CHAR(redemption_seq.NEXTVAL, 'FM00000');
 

    INSERT INTO Reward_Redemption (
        redemption_id, quantity_redeemed, point_spent,
        issue_date, expiry_date, status, member_id, reward_id
    ) VALUES (
        p_redemption_id, p_quantity, v_cost,
        v_issue_date, NULL, 'Active', p_member_id, p_reward_id
    );
 
    SELECT rr.expiry_date, m.points_balance
    INTO   v_expiry_date, v_balance_after
    FROM   Reward_Redemption rr JOIN Member m ON m.member_id = rr.member_id
    WHERE  rr.redemption_id = p_redemption_id;
 
    DBMS_OUTPUT.PUT_LINE('=== REDEMPTION SUCCESSFUL ===');
    DBMS_OUTPUT.PUT_LINE('Redemption ID    : ' || p_redemption_id);
    DBMS_OUTPUT.PUT_LINE('Member           : ' || p_member_id || ' - ' || v_member_name);
    DBMS_OUTPUT.PUT_LINE('Reward           : ' || p_reward_id || ' - ' || v_reward_name);
    DBMS_OUTPUT.PUT_LINE('Quantity         : ' || p_quantity);
    DBMS_OUTPUT.PUT_LINE('Points spent     : ' || v_cost);
    DBMS_OUTPUT.PUT_LINE('Balance left     : ' || v_balance_after);
    DBMS_OUTPUT.PUT_LINE('Issued  / Expiry : ' ||
        TO_CHAR(v_issue_date,  'DD-MON-YYYY') || ' / ' ||
        TO_CHAR(v_expiry_date, 'DD-MON-YYYY'));
 
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;              
        p_redemption_id := NULL;
        DBMS_OUTPUT.PUT_LINE('=== REDEMPTION FAILED ===');
        DBMS_OUTPUT.PUT_LINE(SQLERRM);
        RAISE;
END;

CREATE OR REPLACE PROCEDURE proc_upgrade_member_to_vip (
    p_member_id       IN  VARCHAR2,
    p_months          IN  NUMBER   DEFAULT 12,
    p_pay_now         IN  CHAR     DEFAULT 'Y',
    p_subscription_id OUT VARCHAR2
) IS
    v_member_name   Member.name%TYPE;
    v_member_status Member.status%TYPE;
    v_current_type  Member.membership_type%TYPE;
 
    v_current_end   Member_Subscription.end_date%TYPE;
    v_start_date    DATE;
    v_end_date      DATE;
    v_pay_status    Member_Subscription.payment_status%TYPE;
 
    v_new_type      Member.membership_type%TYPE;
    v_new_rate      Member.point_rate%TYPE;
    v_is_renewal    BOOLEAN := FALSE;
BEGIN
    IF p_months IS NULL OR p_months < 1 THEN
        RAISE_APPLICATION_ERROR(-20024,
            'Subscription length must be at least 1 month.');
    END IF;
 
    IF p_pay_now IS NULL OR UPPER(p_pay_now) NOT IN ('Y', 'N') THEN
        RAISE_APPLICATION_ERROR(-20025,
            'p_pay_now must be Y (paid now) or N (leave as Pending).');
    END IF;
 
    v_pay_status := CASE WHEN UPPER(p_pay_now) = 'Y' THEN 'Paid' ELSE 'Pending' END;

    BEGIN
        SELECT name, status, membership_type
        INTO   v_member_name, v_member_status, v_current_type
        FROM   Member
        WHERE  member_id = p_member_id
        FOR UPDATE;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20026,
                'Member ' || p_member_id || ' does not exist.');
    END;
 
    IF v_member_status <> 'Active' THEN
        RAISE_APPLICATION_ERROR(-20027,
            'Member ' || p_member_id || ' (' || v_member_name ||
            ') is Inactive and cannot be upgraded.');
    END IF;
	
	
    SELECT MAX(end_date)
    INTO   v_current_end
    FROM   Member_Subscription
    WHERE  member_id = p_member_id
    AND    payment_status = 'Paid'
    AND    end_date > SYSDATE;
 
    IF v_current_end IS NOT NULL THEN
        v_is_renewal := TRUE;
        v_start_date := v_current_end;
    ELSE
        v_start_date := TRUNC(SYSDATE);
    END IF;
 
    v_end_date := ADD_MONTHS(v_start_date, p_months);
    p_subscription_id := 'SUB' || TO_CHAR(subscription_seq.NEXTVAL, 'FM00000');
 
    INSERT INTO Member_Subscription (
        subscription_id, start_date, end_date,
        fee_amount, payment_status, member_id
    ) VALUES (
        p_subscription_id, v_start_date, v_end_date,
        12.00, v_pay_status, p_member_id
    );
 

    SELECT membership_type, point_rate
    INTO   v_new_type, v_new_rate
    FROM   Member
    WHERE  member_id = p_member_id;
 
    DBMS_OUTPUT.PUT_LINE('=== ' ||
        CASE WHEN v_is_renewal THEN 'VIP RENEWAL' ELSE 'VIP UPGRADE' END ||
        ' SUCCESSFUL ===');
    DBMS_OUTPUT.PUT_LINE('Subscription ID : ' || p_subscription_id);
    DBMS_OUTPUT.PUT_LINE('Member          : ' || p_member_id || ' - ' || v_member_name);
    DBMS_OUTPUT.PUT_LINE('Term            : ' ||
        TO_CHAR(v_start_date, 'DD-MON-YYYY') || ' to ' ||
        TO_CHAR(v_end_date,   'DD-MON-YYYY') || '  (' || p_months || ' month(s))');
    DBMS_OUTPUT.PUT_LINE('Fee             : RM 12.00');
    DBMS_OUTPUT.PUT_LINE('Payment status  : ' || v_pay_status);
    DBMS_OUTPUT.PUT_LINE('Membership      : ' || v_current_type || ' -> ' || v_new_type);
    DBMS_OUTPUT.PUT_LINE('Point rate      : ' || TO_CHAR(v_new_rate, 'FM90.00') || ' x');
 
    IF v_pay_status = 'Pending' THEN
        DBMS_OUTPUT.PUT_LINE('NOTE: still Pending, so the member stays ' ||
            v_new_type || '. Call sp_confirm_subscription_payment(''' ||
            p_subscription_id || ''') once the RM12 is collected.');
    END IF;
 
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        p_subscription_id := NULL;
        DBMS_OUTPUT.PUT_LINE('=== VIP UPGRADE FAILED ===');
        DBMS_OUTPUT.PUT_LINE(SQLERRM);
        RAISE;
END;

CREATE OR REPLACE PROCEDURE proc_confirm_subscription (
    p_subscription_id IN VARCHAR2
) IS
    v_member_id  Member_Subscription.member_id%TYPE;
    v_pay_status Member_Subscription.payment_status%TYPE;
    v_end_date   Member_Subscription.end_date%TYPE;
    v_new_type   Member.membership_type%TYPE;
BEGIN
    BEGIN
        SELECT member_id, payment_status, end_date
        INTO   v_member_id, v_pay_status, v_end_date
        FROM   Member_Subscription
        WHERE  subscription_id = p_subscription_id
        FOR UPDATE;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20028,
                'Subscription ' || p_subscription_id || ' does not exist.');
    END;
 
    IF v_pay_status = 'Paid' THEN
        RAISE_APPLICATION_ERROR(-20029,
            'Subscription ' || p_subscription_id || ' is already paid.');
    ELSIF v_pay_status <> 'Pending' THEN
        RAISE_APPLICATION_ERROR(-20030,
            'Subscription ' || p_subscription_id || ' is ' || v_pay_status ||
            ' and can no longer be paid.');
    END IF;
 
    UPDATE Member_Subscription
    SET    payment_status = 'Paid'
    WHERE  subscription_id = p_subscription_id;
 
    SELECT membership_type INTO v_new_type
    FROM   Member WHERE member_id = v_member_id;
 
    DBMS_OUTPUT.PUT_LINE('=== PAYMENT CONFIRMED ===');
    DBMS_OUTPUT.PUT_LINE('Subscription    : ' || p_subscription_id);
    DBMS_OUTPUT.PUT_LINE('Member          : ' || v_member_id);
    DBMS_OUTPUT.PUT_LINE('Membership now  : ' || v_new_type ||
        ' (valid to ' || TO_CHAR(v_end_date, 'DD-MON-YYYY') || ')');

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('=== PAYMENT CONFIRMATION FAILED ===');
        DBMS_OUTPUT.PUT_LINE(SQLERRM);
        RAISE;
END;

CREATE OR REPLACE PROCEDURE proc_add_new_member (
    p_name             IN  VARCHAR2,
    p_age              IN  NUMBER,
    p_gender           IN  VARCHAR2,
    p_phone_no         IN  VARCHAR2,
    p_email            IN  VARCHAR2 DEFAULT NULL,
    p_ic_number        IN  VARCHAR2 DEFAULT NULL,
    p_customer_id      IN  VARCHAR2 DEFAULT NULL,
    p_opening_points   IN  NUMBER   DEFAULT 0,
    p_member_id        OUT VARCHAR2,
    p_customer_id_used OUT VARCHAR2
) IS
    v_name    Member.name%TYPE;
    v_gender  Member.gender%TYPE;
    v_phone   Member.phone_no%TYPE;
    v_email   Member.email%TYPE;
    v_ic      Customer.ic_number%TYPE;
 
    v_dummy          NUMBER;
    v_existing_member Customer.member_id%TYPE;
    v_existing_ic     Customer.ic_number%TYPE;
    v_linked_existing BOOLEAN := FALSE;
BEGIN
    v_name   := TRIM(p_name);
    v_gender := INITCAP(TRIM(p_gender));
    v_phone  := TRIM(p_phone_no);
    v_email  := LOWER(TRIM(p_email));
    v_ic     := TRIM(p_ic_number);

    IF v_name IS NULL OR LENGTH(v_name) < 2 THEN
        RAISE_APPLICATION_ERROR(-20501,
            'Member name is required and must be at least 2 characters.');
    END IF;
 
    IF p_age IS NULL OR p_age < 12 OR p_age >= 130 THEN
        RAISE_APPLICATION_ERROR(-20502,
            'Age must be between 12 and 129. Members under 12 cannot register.');
    END IF;
 
    IF v_gender NOT IN ('Male', 'Female') THEN
        RAISE_APPLICATION_ERROR(-20503,
            'Gender must be Male or Female.');
    END IF;
 
    IF v_phone IS NULL OR LENGTH(v_phone) < 9 THEN
        RAISE_APPLICATION_ERROR(-20504,
            'A contact number is required, e.g. 012-345 6789.');
    END IF;
 
    IF v_email IS NOT NULL AND v_email NOT LIKE '%@%.%' THEN
        RAISE_APPLICATION_ERROR(-20505,
            'Email address is not in a valid format: ' || v_email);
    END IF;
 
    IF p_opening_points IS NULL OR p_opening_points < 0 THEN
        RAISE_APPLICATION_ERROR(-20506,
            'Opening points balance cannot be negative.');
    END IF;

    IF v_ic IS NOT NULL
       AND NOT REGEXP_LIKE(v_ic, '^[0-9]{6}-[0-9]{2}-[0-9]{4}$') THEN
        RAISE_APPLICATION_ERROR(-20507,
            'IC number must be in the format 020428-06-1071.');
    END IF;

    IF v_email IS NOT NULL THEN
        BEGIN
            SELECT 1 INTO v_dummy FROM Member
            WHERE  LOWER(email) = v_email AND ROWNUM = 1;
 
            RAISE_APPLICATION_ERROR(-20508,
                'Email ' || v_email || ' is already registered to another member.');
        EXCEPTION
            WHEN NO_DATA_FOUND THEN NULL;   -- good, the email is free
        END;
    END IF;
 
    IF v_ic IS NOT NULL THEN
        BEGIN
            SELECT 1 INTO v_dummy FROM Customer
            WHERE  ic_number = v_ic
            AND   (p_customer_id IS NULL OR customer_id <> p_customer_id)
            AND    ROWNUM = 1;
 
            RAISE_APPLICATION_ERROR(-20509,
                'IC number ' || v_ic || ' already belongs to another customer.');
        EXCEPTION
            WHEN NO_DATA_FOUND THEN NULL;
        END;
    END IF;

    IF p_customer_id IS NOT NULL THEN
        BEGIN
            SELECT member_id, ic_number
            INTO   v_existing_member, v_existing_ic
            FROM   Customer
            WHERE  customer_id = p_customer_id
            FOR UPDATE;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                RAISE_APPLICATION_ERROR(-20510,
                    'Customer ' || p_customer_id || ' does not exist.');
        END;
 
        IF v_existing_member IS NOT NULL THEN
            RAISE_APPLICATION_ERROR(-20511,
                'Customer ' || p_customer_id ||
                ' is already linked to member ' || v_existing_member || '.');
        END IF;
 
        v_linked_existing := TRUE;
    END IF;
 
    p_member_id := 'MB' || TO_CHAR(member_seq.NEXTVAL, 'FM00000');
 
    INSERT INTO Member (
        member_id, name, age, gender, phone_no, email,
        points_balance, membership_type, point_rate, status
    ) VALUES (
        p_member_id, v_name, p_age, v_gender, v_phone, v_email,
        NVL(p_opening_points, 0), 'Normal', 1.00, 'Active'
    );

    IF v_linked_existing THEN
        UPDATE Customer
        SET    member_id = p_member_id,
               ic_number = NVL(ic_number, v_ic)
        WHERE  customer_id = p_customer_id;
 
        p_customer_id_used := p_customer_id;
    ELSE
        p_customer_id_used := 'CUS' || TO_CHAR(customer_seq.NEXTVAL, 'FM00000');
 
        INSERT INTO Customer (customer_id, ic_number, member_id)
        VALUES (p_customer_id_used, v_ic, p_member_id);
    END IF;
 
    COMMIT;
 
    DBMS_OUTPUT.PUT_LINE('=== MEMBER REGISTERED ===');
    DBMS_OUTPUT.PUT_LINE('Member ID    : ' || p_member_id);
    DBMS_OUTPUT.PUT_LINE('Customer ID  : ' || p_customer_id_used ||
        CASE WHEN v_linked_existing
             THEN '  (existing walk-in customer, history preserved)'
             ELSE '  (newly created)' END);
    DBMS_OUTPUT.PUT_LINE('Name         : ' || v_name);
    DBMS_OUTPUT.PUT_LINE('Age / Gender : ' || p_age || ' / ' || v_gender);
    DBMS_OUTPUT.PUT_LINE('Phone        : ' || v_phone);
    DBMS_OUTPUT.PUT_LINE('Email        : ' || NVL(v_email, '(none given)'));
    DBMS_OUTPUT.PUT_LINE('IC number    : ' || NVL(v_ic, '(none given)'));
    DBMS_OUTPUT.PUT_LINE('Opening pts  : ' || NVL(p_opening_points, 0));
    DBMS_OUTPUT.PUT_LINE('Tier         : Normal (1.00x points)');
    DBMS_OUTPUT.PUT_LINE('Registered   : ' || TO_CHAR(SYSDATE, 'DD-Mon-YYYY HH24:MI'));
	
	EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN
        ROLLBACK;
        p_member_id        := NULL;
        p_customer_id_used := NULL;
        DBMS_OUTPUT.PUT_LINE('=== REGISTRATION FAILED ===');
        DBMS_OUTPUT.PUT_LINE('That email or IC number is already on file.');
        RAISE;
    WHEN OTHERS THEN
        ROLLBACK;
        p_member_id        := NULL;
        p_customer_id_used := NULL;
        DBMS_OUTPUT.PUT_LINE('=== REGISTRATION FAILED ===');
        DBMS_OUTPUT.PUT_LINE(SQLERRM);
        RAISE;
END;

-- VIEW
CREATE OR REPLACE VIEW v_order_revenue AS
SELECT o.order_no,
       o.order_date_time,
       EXTRACT(YEAR FROM o.order_date_time) AS order_year,
       TO_CHAR(o.order_date_time, 'YYYY-MM') AS order_month,
       o.order_type,
       o.service_type,
       o.branch_id,
       b.name AS branch_name,
       o.customer_id,
       c.member_id,
       CASE WHEN c.member_id IS NULL
            THEN 'Non-Member' ELSE 'Member' END AS customer_class,
       NVL(m.membership_type, 'None') AS membership_type,
       o.subtotal_amount,
       o.discount_amount,
       o.subtotal_amount - o.discount_amount AS net_sales,
       NVL(dr.delivery_fee, 0) AS delivery_fee,
       o.points_earned,
       o.redemption_id
FROM   Orders o
       JOIN Branch b  ON b.branch_id   = o.branch_id
       JOIN Customer c  ON c.customer_id = o.customer_id
       LEFT JOIN Member m  ON m.member_id  = c.member_id
       LEFT JOIN Delivery_Request dr ON dr.order_no = o.order_no
WHERE  o.order_status = 'Completed';


CREATE OR REPLACE VIEW v_order_basket AS
SELECT od.order_no,
       COUNT(DISTINCT od.item_id) AS line_count,
       SUM(od.quantity) AS units_sold,
       COUNT(DISTINCT i.category) AS category_count,
       ROUND(SUM(i.unit_weight_kg * od.quantity), 3)  AS basket_weight_kg
FROM   Order_Detail od
       JOIN Item i ON i.item_id = od.item_id
GROUP  BY od.order_no;

CREATE OR REPLACE VIEW v_member_value AS
SELECT r.member_id,
       COUNT(*)                    AS total_orders,
       SUM(r.net_sales)            AS lifetime_spend,
       SUM(r.points_earned)        AS lifetime_points_earned,
       MAX(r.order_date_time)      AS last_order_date
FROM   v_order_revenue r
WHERE  r.member_id IS NOT NULL
GROUP  BY r.member_id;

CREATE OR REPLACE VIEW v_member_home_branch AS
SELECT member_id, branch_id, branch_name, orders_at_branch
FROM  (SELECT r.member_id,
              r.branch_id,
              r.branch_name,
              COUNT(*) AS orders_at_branch,
              ROW_NUMBER() OVER (PARTITION BY r.member_id
                                 ORDER BY COUNT(*) DESC, r.branch_id) AS rn
       FROM   v_order_revenue r
       WHERE  r.member_id IS NOT NULL
       GROUP  BY r.member_id, r.branch_id, r.branch_name)
WHERE rn = 1;

SET LINESIZE 250
SET PAGESIZE 100

COLUMN CHANNEL FORMAT A10;
COLUMN YR HEADING "YEAR" FORMAT 9999;
COLUMN ORDERS FORMAT 99999;
COLUMN net_sales FORMAT $999999.99;
COLUMN pct_of_year HEADING "PREFERENCES(%)" FORMAT 99.99;
COLUMN growth_pct HEADING "GROWTH(%)" FORMAT 99.99;
COLUMN delivery_fee_charged FORMAT $99999.99;
COLUMN pct_member_orders HEADING "MEMBER(%)" FORMAT 99.99;

SELECT CASE WHEN ROW_NUMBER() OVER (PARTITION BY r.order_type
                                    ORDER BY r.order_year) = 1
            THEN r.order_type END  AS channel,
       r.order_year AS YR,
       COUNT(*) AS orders,
       ROUND(SUM(r.net_sales), 2) AS NET_SALES,
       ROUND(100 * RATIO_TO_REPORT(SUM(r.net_sales))
                   OVER (PARTITION BY r.order_year), 1) AS pct_of_year,
       ROUND(100 * (SUM(r.net_sales)
             - LAG(SUM(r.net_sales)) OVER (PARTITION BY r.order_type
                                           ORDER BY r.order_year))
             / NULLIF(LAG(SUM(r.net_sales)) OVER (PARTITION BY r.order_type
                                           ORDER BY r.order_year), 0), 1) AS growth_pct,
       ROUND(AVG(r.net_sales), 2) AS avg_basket_value,
       ROUND(AVG(bk.units_sold), 1) AS avg_units,
       ROUND(AVG(bk.basket_weight_kg), 2) AS avg_weight_kg,
       ROUND(SUM(r.delivery_fee), 2) AS delivery_fee_charged,
       ROUND(SUM(r.discount_amount), 2) AS discount_given,
       SUM(r.points_earned) AS points_issued,
       ROUND(100 * COUNT(CASE WHEN r.customer_class = 'Member'
                              THEN 1 END) / COUNT(*), 1) AS pct_member_orders
FROM   v_order_revenue r
       JOIN v_order_basket bk ON bk.order_no = r.order_no
GROUP  BY r.order_type, r.order_year
ORDER  BY r.order_type, r.order_year;

WITH sales AS (

           SUM(CASE WHEN customer_class = 'Member'     THEN net_sales END) AS member_revenue,
           SUM(CASE WHEN customer_class = 'Non-Member' THEN net_sales END) AS nonmember_revenue,

),
redeemed AS (
    SELECT EXTRACT(YEAR FROM rr.issue_date) AS order_year,
           COUNT(*) AS vouchers_issued,
           SUM(rr.point_spent) AS points_redeemed,
           SUM(r.value * rr.quantity_redeemed) AS reward_cost_rm,
           COUNT(CASE WHEN rr.status = 'Used'    THEN 1 END) AS vouchers_used,
           COUNT(CASE WHEN rr.status = 'Expired' THEN 1 END) AS vouchers_expired
    FROM   Reward_Redemption rr
           JOIN Reward r ON r.reward_id = rr.reward_id
    GROUP  BY EXTRACT(YEAR FROM rr.issue_date)
),
subs AS (
    SELECT EXTRACT(YEAR FROM start_date)   AS order_year,
           SUM(CASE WHEN payment_status = 'Paid' THEN fee_amount END) AS subscription_rm,
           COUNT(CASE WHEN payment_status = 'Paid' THEN 1 END)   AS paid_subscriptions
    FROM   Member_Subscription
    GROUP  BY EXTRACT(YEAR FROM start_date)
)
SELECT s.order_year AS year,
       s.active_members AS MEMBERS,
       ROUND(s.member_revenue, 2) AS MEM_SPENT,
       ROUND(s.nonmember_revenue, 2) AS NON_MEM_SPENT,
       ROUND(s.member_revenue    / NULLIF(s.member_orders, 0), 2)AS mem_order_avg,
       ROUND(s.nonmember_revenue / NULLIF(s.nonmember_orders, 0), 2)  AS normal_order_avg,
       ROUND(100 * ((s.member_revenue    / NULLIF(s.member_orders, 0))
                  - (s.nonmember_revenue / NULLIF(s.nonmember_orders, 0)))
             / NULLIF(s.nonmember_revenue / NULLIF(s.nonmember_orders, 0), 0), 1) AS PCT,
       s.points_issued,
       NVL(d.points_redeemed, 0) AS points_redeemed,
       ROUND(100 * NVL(d.points_redeemed, 0)
             / NULLIF(s.points_issued, 0), 1) AS redemption_rate_pct,
       NVL(d.vouchers_issued, 0) AS vouchers_issued,
       ROUND(100 * NVL(d.vouchers_expired, 0)
             / NULLIF(d.vouchers_issued, 0), 1)AS "UNUSED(%)",
       ROUND(NVL(d.reward_cost_rm, 0), 2)AS reward_cost_rm,
       ROUND(NVL(u.subscription_rm, 0), 2) AS subscription_rm,
       ROUND(NVL(u.subscription_rm, 0) - NVL(d.reward_cost_rm, 0), 2) AS earnings
FROM   sales s
       LEFT JOIN redeemed d ON d.order_year = s.order_year
       LEFT JOIN subs     u ON u.order_year = s.order_year
ORDER  BY s.order_year;

CREATE OR REPLACE PROCEDURE report_channel_revenue (
    p_date_from IN DATE DEFAULT NULL,
    p_date_to   IN DATE DEFAULT NULL,
    p_branch_id IN VARCHAR2 DEFAULT NULL
) IS
    c_width  CONSTANT NUMBER := 112;
 
    v_from   DATE;
    v_to     DATE;
    v_branch_label VARCHAR2(120);
 
    v_b_on_ord NUMBER;  v_b_on_rev NUMBER;
    v_b_wi_ord NUMBER;  v_b_wi_rev NUMBER;
    v_b_fee    NUMBER;
 
    v_g_on_ord NUMBER := 0;  v_g_on_rev NUMBER := 0;
    v_g_wi_ord NUMBER := 0;  v_g_wi_rev NUMBER := 0;
    v_g_fee    NUMBER := 0;
    v_branches NUMBER := 0;

    CURSOR c_branch (cp_from DATE, cp_to DATE, cp_branch VARCHAR2) IS
        SELECT r.branch_id,
               r.branch_name,
               SUM(r.net_sales) AS branch_revenue
        FROM   v_order_revenue r
        WHERE  r.order_date_time >= cp_from
        AND    r.order_date_time <  cp_to + 1
        AND   (cp_branch IS NULL OR r.branch_id = cp_branch)
        GROUP  BY r.branch_id, r.branch_name
        ORDER  BY SUM(r.net_sales) DESC;
 
    CURSOR c_month (cp_branch VARCHAR2, cp_from DATE, cp_to DATE) IS
        SELECT r.order_month,
               COUNT(CASE WHEN r.order_type = 'Online'  THEN 1 END) AS on_ord,
               NVL(SUM(CASE WHEN r.order_type = 'Online'  THEN r.net_sales END), 0) AS on_rev,
               COUNT(CASE WHEN r.order_type = 'Walk-In' THEN 1 END) AS wi_ord,
               NVL(SUM(CASE WHEN r.order_type = 'Walk-In' THEN r.net_sales END), 0) AS wi_rev,
               SUM(r.net_sales) AS tot_rev,
               SUM(r.delivery_fee) AS fee
        FROM   v_order_revenue r
        WHERE  r.branch_id = cp_branch
        AND    r.order_date_time >= cp_from
        AND    r.order_date_time <  cp_to + 1
        GROUP  BY r.order_month
        ORDER  BY r.order_month;

    CURSOR c_channel (cp_from DATE, cp_to DATE, cp_branch VARCHAR2) IS
        SELECT r.order_type,
               COUNT(*) AS orders,
               SUM(r.net_sales) AS revenue,
               AVG(r.net_sales) AS avg_basket,
               SUM(r.delivery_fee) AS fee,
               AVG(bk.basket_weight_kg) AS avg_weight,
               AVG(bk.units_sold) AS avg_units
        FROM   v_order_revenue r
               JOIN v_order_basket bk ON bk.order_no = r.order_no
        WHERE  r.order_date_time >= cp_from
        AND    r.order_date_time <  cp_to + 1
        AND   (cp_branch IS NULL OR r.branch_id = cp_branch)
        GROUP  BY r.order_type
        ORDER  BY r.order_type;

    FUNCTION money (p_val NUMBER, p_width NUMBER DEFAULT 13) RETURN VARCHAR2 IS
    BEGIN
        RETURN LPAD(TO_CHAR(NVL(p_val, 0), 'FM999,999,990.00'), p_width);
    END money;
 
    FUNCTION num (p_val NUMBER, p_width NUMBER DEFAULT 8) RETURN VARCHAR2 IS
    BEGIN
        RETURN LPAD(TO_CHAR(NVL(p_val, 0), 'FM999,999,990'), p_width);
    END num;
 
    FUNCTION pct (p_num NUMBER, p_den NUMBER, p_width NUMBER DEFAULT 8) RETURN VARCHAR2 IS
    BEGIN
        IF NVL(p_den, 0) = 0 THEN
            RETURN LPAD('-', p_width);
        END IF;
        RETURN LPAD(TO_CHAR(ROUND(100 * p_num / p_den, 1), 'FM990.0') || '%', p_width);
    END pct;
 
BEGIN
    IF p_date_from IS NULL OR p_date_to IS NULL THEN
        SELECT NVL(p_date_from, TRUNC(MIN(order_date_time))),
               NVL(p_date_to,   TRUNC(MAX(order_date_time)))
        INTO   v_from, v_to
        FROM   v_order_revenue;
    ELSE
        v_from := TRUNC(p_date_from);
        v_to   := TRUNC(p_date_to);
    END IF;
 
    IF v_from > v_to THEN
        RAISE_APPLICATION_ERROR(-20401,
            'Report start date must not be later than the end date.');
    END IF;
 
    IF p_branch_id IS NULL THEN
        v_branch_label := 'ALL BRANCHES';
    ELSE
        BEGIN
            SELECT p_branch_id || ' - ' || name INTO v_branch_label
            FROM   Branch WHERE branch_id = p_branch_id;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                RAISE_APPLICATION_ERROR(-20402,
                    'Branch ' || p_branch_id || ' does not exist.');
        END;
    END IF;
    DBMS_OUTPUT.PUT_LINE(RPAD('=', c_width, '='));
    DBMS_OUTPUT.PUT_LINE('SPEEDMART - CHANNEL REVENUE REPORT   (Online vs Walk-In)');
    DBMS_OUTPUT.PUT_LINE('Period  : ' || TO_CHAR(v_from, 'DD-Mon-YYYY') ||
                         '  to  '     || TO_CHAR(v_to,   'DD-Mon-YYYY'));
    DBMS_OUTPUT.PUT_LINE('Branch  : ' || v_branch_label);
    DBMS_OUTPUT.PUT_LINE('Printed : ' || TO_CHAR(SYSDATE, 'DD-Mon-YYYY HH24:MI'));
    DBMS_OUTPUT.PUT_LINE('Note    : revenue is merchandise net of discount; delivery fees are shown separately.');
    DBMS_OUTPUT.PUT_LINE(RPAD('=', c_width, '='));

    FOR b IN c_branch(v_from, v_to, p_branch_id) LOOP
 
        v_branches := v_branches + 1;
        v_b_on_ord := 0;  v_b_on_rev := 0;
        v_b_wi_ord := 0;  v_b_wi_rev := 0;
        v_b_fee    := 0;
 
        DBMS_OUTPUT.PUT_LINE(' ');
        DBMS_OUTPUT.PUT_LINE('BRANCH ' || b.branch_id || ' - ' || b.branch_name);
        DBMS_OUTPUT.PUT_LINE(RPAD('-', c_width, '-'));
        DBMS_OUTPUT.PUT_LINE(
            RPAD('Month', 9)      || LPAD('Onl Ord', 8)   || LPAD('Online Rev', 14) ||
            LPAD('Wlk Ord', 9)    || LPAD('Walk-In Rev', 15) ||
            LPAD('Total Rev', 15) || LPAD('Online %', 10) ||
            LPAD('Deliv Fee', 14) || LPAD('Avg Basket', 14));
        DBMS_OUTPUT.PUT_LINE(RPAD('-', c_width, '-'));
 
        FOR m IN c_month(b.branch_id, v_from, v_to) LOOP
 
            DBMS_OUTPUT.PUT_LINE(
                RPAD(m.order_month, 9) ||
                num(m.on_ord, 8)       || money(m.on_rev, 14) ||
                num(m.wi_ord, 9)       || money(m.wi_rev, 15) ||
                money(m.tot_rev, 15)   || pct(m.on_rev, m.tot_rev, 10) ||
                money(m.fee, 14)       ||
                money(m.tot_rev / NULLIF(m.on_ord + m.wi_ord, 0), 14));
 
            v_b_on_ord := v_b_on_ord + m.on_ord;
            v_b_on_rev := v_b_on_rev + m.on_rev;
            v_b_wi_ord := v_b_wi_ord + m.wi_ord;
            v_b_wi_rev := v_b_wi_rev + m.wi_rev;
            v_b_fee    := v_b_fee    + m.fee;
        END LOOP;

        DBMS_OUTPUT.PUT_LINE(RPAD('-', c_width, '-'));
        DBMS_OUTPUT.PUT_LINE(
            RPAD('SUBTOTAL', 9)         ||
            num(v_b_on_ord, 8)          || money(v_b_on_rev, 14) ||
            num(v_b_wi_ord, 9)          || money(v_b_wi_rev, 15) ||
            money(v_b_on_rev + v_b_wi_rev, 15) ||
            pct(v_b_on_rev, v_b_on_rev + v_b_wi_rev, 10) ||
            money(v_b_fee, 14)          ||
            money((v_b_on_rev + v_b_wi_rev)
                  / NULLIF(v_b_on_ord + v_b_wi_ord, 0), 14));
 
        v_g_on_ord := v_g_on_ord + v_b_on_ord;
        v_g_on_rev := v_g_on_rev + v_b_on_rev;
        v_g_wi_ord := v_g_wi_ord + v_b_wi_ord;
        v_g_wi_rev := v_g_wi_rev + v_b_wi_rev;
        v_g_fee    := v_g_fee    + v_b_fee;
    END LOOP;
 
    IF v_branches = 0 THEN
        DBMS_OUTPUT.PUT_LINE(' ');
        DBMS_OUTPUT.PUT_LINE('No completed orders found for the selected period and branch.');
        DBMS_OUTPUT.PUT_LINE(RPAD('=', c_width, '='));
        RETURN;
    END IF;
 
    DBMS_OUTPUT.PUT_LINE(' ');
    DBMS_OUTPUT.PUT_LINE(RPAD('=', c_width, '='));
    DBMS_OUTPUT.PUT_LINE(
        RPAD('GRAND', 9)            ||
        num(v_g_on_ord, 8)          || money(v_g_on_rev, 14) ||
        num(v_g_wi_ord, 9)          || money(v_g_wi_rev, 15) ||
        money(v_g_on_rev + v_g_wi_rev, 15) ||
        pct(v_g_on_rev, v_g_on_rev + v_g_wi_rev, 10) ||
        money(v_g_fee, 14)          ||
        money((v_g_on_rev + v_g_wi_rev)
              / NULLIF(v_g_on_ord + v_g_wi_ord, 0), 14));
    DBMS_OUTPUT.PUT_LINE(RPAD('=', c_width, '='));
 
    DBMS_OUTPUT.PUT_LINE(' ');
    DBMS_OUTPUT.PUT_LINE('CHANNEL SUMMARY ACROSS ' || v_branches || ' BRANCH(ES)');
    DBMS_OUTPUT.PUT_LINE(RPAD('-', c_width, '-'));
    DBMS_OUTPUT.PUT_LINE(
        RPAD('Channel', 10)      || LPAD('Orders', 9)     || LPAD('Net Revenue', 16) ||
        LPAD('Share', 9)         || LPAD('Avg Basket', 13) ||
        LPAD('Avg Units', 11)    || LPAD('Avg Wt(kg)', 12) || LPAD('Deliv Fee', 14));
    DBMS_OUTPUT.PUT_LINE(RPAD('-', c_width, '-'));
 
    FOR ch IN c_channel(v_from, v_to, p_branch_id) LOOP
        DBMS_OUTPUT.PUT_LINE(
            RPAD(ch.order_type, 10) ||
            num(ch.orders, 9)       || money(ch.revenue, 16) ||
            pct(ch.revenue, v_g_on_rev + v_g_wi_rev, 9) ||
            money(ch.avg_basket, 13)||
            LPAD(TO_CHAR(ROUND(ch.avg_units, 1), 'FM990.0'), 11) ||
            LPAD(TO_CHAR(ROUND(ch.avg_weight, 2), 'FM990.00'), 12) ||
            money(ch.fee, 14));
    END LOOP;
 
    DBMS_OUTPUT.PUT_LINE(RPAD('-', c_width, '-'));
    DBMS_OUTPUT.PUT_LINE('Delivery fees collected: RM ' ||
        TO_CHAR(v_g_fee, 'FM999,999,990.00') || ' (' ||
        TO_CHAR(ROUND(100 * v_g_fee / NULLIF(v_g_on_rev, 0), 1), 'FM990.0') ||
        '% of online net revenue)');
    DBMS_OUTPUT.PUT_LINE('*** END OF REPORT ***');
 
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('*** REPORT FAILED: ' || SQLERRM || ' ***');
        RAISE;
END;

CREATE OR REPLACE PROCEDURE report_vip_renewal_list (
    p_days_ahead IN NUMBER   DEFAULT 30,
    p_branch_id  IN VARCHAR2 DEFAULT NULL,
    p_min_spend  IN NUMBER   DEFAULT 0
) IS
    c_width  CONSTANT NUMBER := 104;
    c_months CONSTANT NUMBER := 12;     
 
    v_branch_label VARCHAR2(120);
    v_seq          NUMBER := 0;
 
    v_recent_spend  NUMBER;
    v_recent_orders NUMBER;
    v_shown         NUMBER;
    v_vouchers      NUMBER;
    v_priority      VARCHAR2(10);
 
    v_total_members  NUMBER := 0;
    v_total_recent   NUMBER := 0;
    v_total_points   NUMBER := 0;
    v_high           NUMBER := 0;
    v_medium         NUMBER := 0;
    v_low            NUMBER := 0;
    v_urgent         NUMBER := 0;

    CURSOR c_expiring (cp_days NUMBER, cp_branch VARCHAR2, cp_min NUMBER) IS
        SELECT m.member_id,
               m.name,
               m.phone_no,
               NVL(m.email, '(no email on file)')        AS email,
               m.points_balance,
               s.cover_end,
               TRUNC(s.cover_end) - TRUNC(SYSDATE)       AS days_left,
               NVL(v.total_orders, 0)                    AS total_orders,
               NVL(v.lifetime_spend, 0)                  AS lifetime_spend,
               v.last_order_date,
               NVL(hb.branch_id, '-')                    AS home_branch_id,
               NVL(hb.branch_name, '(no orders yet)')    AS home_branch_name
        FROM   Member m
               JOIN (SELECT member_id, MAX(end_date) AS cover_end
                     FROM   Member_Subscription
                     WHERE  payment_status = 'Paid'
                     GROUP  BY member_id) s
                 ON s.member_id = m.member_id
               LEFT JOIN v_member_value       v  ON v.member_id  = m.member_id
               LEFT JOIN v_member_home_branch hb ON hb.member_id = m.member_id
        WHERE  m.status = 'Active'
        AND    s.cover_end >= TRUNC(SYSDATE)
        AND    s.cover_end <  TRUNC(SYSDATE) + cp_days + 1
        AND   (cp_branch IS NULL OR hb.branch_id = cp_branch)
        AND    NVL(v.lifetime_spend, 0) >= cp_min
        ORDER  BY s.cover_end, NVL(v.lifetime_spend, 0) DESC;

    CURSOR c_recent_orders (cp_member VARCHAR2, cp_months NUMBER) IS
        SELECT r.order_no,
               r.order_date_time,
               r.order_type,
               r.branch_id,
               r.net_sales,
               r.points_earned
        FROM   v_order_revenue r
        WHERE  r.member_id = cp_member
        AND    r.order_date_time >= ADD_MONTHS(TRUNC(SYSDATE), -cp_months)
        ORDER  BY r.order_date_time DESC;
 

    CURSOR c_open_vouchers (cp_member VARCHAR2) IS
        SELECT rr.redemption_id,
               rw.name AS reward_name,
               rr.expiry_date,
               rr.quantity_redeemed
        FROM   Reward_Redemption rr
               JOIN Reward rw ON rw.reward_id = rr.reward_id
        WHERE  rr.member_id = cp_member
        AND    rr.status = 'Active'
        AND    (rr.expiry_date IS NULL OR rr.expiry_date > SYSDATE)
        ORDER  BY rr.expiry_date;
 
    FUNCTION money (p_val NUMBER, p_width NUMBER DEFAULT 12) RETURN VARCHAR2 IS
    BEGIN
        RETURN LPAD(TO_CHAR(NVL(p_val, 0), 'FM999,999,990.00'), p_width);
    END money;
 
BEGIN
    IF p_days_ahead IS NULL OR p_days_ahead < 1 THEN
        RAISE_APPLICATION_ERROR(-20403,
            'The look-ahead window must be at least 1 day.');
    END IF;
 
    IF p_branch_id IS NULL THEN
        v_branch_label := 'ALL BRANCHES';
    ELSE
        BEGIN
            SELECT p_branch_id || ' - ' || name INTO v_branch_label
            FROM   Branch WHERE branch_id = p_branch_id;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                RAISE_APPLICATION_ERROR(-20404,
                    'Branch ' || p_branch_id || ' does not exist.');
        END;
    END IF;

    DBMS_OUTPUT.PUT_LINE(RPAD('=', c_width, '='));
    DBMS_OUTPUT.PUT_LINE('SPEEDMART - VIP RENEWAL CALL LIST');
    DBMS_OUTPUT.PUT_LINE('Window       : subscriptions expiring by ' ||
        TO_CHAR(TRUNC(SYSDATE) + p_days_ahead, 'DD-Mon-YYYY') ||
        '  (' || p_days_ahead || ' days)');
    DBMS_OUTPUT.PUT_LINE('Branch       : ' || v_branch_label);
    DBMS_OUTPUT.PUT_LINE('Min lifetime : RM ' || TO_CHAR(NVL(p_min_spend, 0), 'FM999,990.00'));
    DBMS_OUTPUT.PUT_LINE('Printed      : ' || TO_CHAR(SYSDATE, 'DD-Mon-YYYY HH24:MI'));
    DBMS_OUTPUT.PUT_LINE(RPAD('=', c_width, '='));
 
    FOR mb IN c_expiring(p_days_ahead, p_branch_id, NVL(p_min_spend, 0)) LOOP
 
        v_seq           := v_seq + 1;
        v_recent_spend  := 0;
        v_recent_orders := 0;
        v_shown         := 0;
        v_vouchers      := 0;
 

        FOR o IN c_recent_orders(mb.member_id, c_months) LOOP
            v_recent_orders := v_recent_orders + 1;
            v_recent_spend  := v_recent_spend + o.net_sales;
        END LOOP;
		
        IF v_recent_spend >= 1000 THEN
            v_priority := 'HIGH';   v_high   := v_high + 1;
        ELSIF v_recent_spend >= 300 THEN
            v_priority := 'MEDIUM'; v_medium := v_medium + 1;
        ELSE
            v_priority := 'LOW';    v_low    := v_low + 1;
        END IF;
 
        IF mb.days_left <= 7 THEN
            v_urgent := v_urgent + 1;
        END IF;
 
        DBMS_OUTPUT.PUT_LINE(' ');
        DBMS_OUTPUT.PUT_LINE(RPAD('-', c_width, '-'));
        DBMS_OUTPUT.PUT_LINE(
            '[' || LPAD(v_seq, 3, '0') || ']  ' ||
            RPAD(mb.member_id, 10) || RPAD(mb.name, 32) ||
            'PRIORITY: ' || RPAD(v_priority, 8) ||
            CASE WHEN mb.days_left <= 7 THEN '** CALL FIRST **' END);
        DBMS_OUTPUT.PUT_LINE(
            '        Contact     : ' || mb.phone_no || '   ' || mb.email);
        DBMS_OUTPUT.PUT_LINE(
            '        Home branch : ' || mb.home_branch_id || ' - ' || mb.home_branch_name);
        DBMS_OUTPUT.PUT_LINE(
            '        VIP expires : ' || TO_CHAR(mb.cover_end, 'DD-Mon-YYYY') ||
            '   (' || mb.days_left || ' day(s) left)');
        DBMS_OUTPUT.PUT_LINE(
            '        Points held : ' || TO_CHAR(mb.points_balance, 'FM999,999,990') ||
            '        Lifetime: ' || mb.total_orders || ' orders / RM ' ||
            TO_CHAR(mb.lifetime_spend, 'FM999,999,990.00'));
        DBMS_OUTPUT.PUT_LINE(
            '        Last order  : ' ||
            NVL(TO_CHAR(mb.last_order_date, 'DD-Mon-YYYY'), 'never'));

        IF v_recent_orders = 0 THEN
            DBMS_OUTPUT.PUT_LINE(
                '        Activity    : no orders in the last ' || c_months ||
                ' months - lapsed shopper, expect a hard sell.');
        ELSE
            DBMS_OUTPUT.PUT_LINE('        Recent orders (last ' || c_months || ' months):');
            FOR o IN c_recent_orders(mb.member_id, c_months) LOOP
                EXIT WHEN v_shown >= 5;
                v_shown := v_shown + 1;
                DBMS_OUTPUT.PUT_LINE(
                    '            ' || RPAD(o.order_no, 12) ||
                    RPAD(TO_CHAR(o.order_date_time, 'DD-Mon-YYYY'), 14) ||
                    RPAD(o.order_type, 9) || RPAD(o.branch_id, 8) ||
                    money(o.net_sales, 11) ||
                    LPAD(TO_CHAR(o.points_earned, 'FM999,990') || ' pts', 12));
            END LOOP;
 
            IF v_recent_orders > v_shown THEN
                DBMS_OUTPUT.PUT_LINE(
                    '            ... and ' || (v_recent_orders - v_shown) ||
                    ' more order(s) not shown');
            END IF;
 
            DBMS_OUTPUT.PUT_LINE(
                '            ' || RPAD('SUBTOTAL', 43) ||
                money(v_recent_spend, 11) ||
                '   over ' || v_recent_orders || ' order(s)');
        END IF;

        FOR vch IN c_open_vouchers(mb.member_id) LOOP
            IF v_vouchers = 0 THEN
                DBMS_OUTPUT.PUT_LINE('        Unused vouchers:');
            END IF;
            v_vouchers := v_vouchers + 1;
            DBMS_OUTPUT.PUT_LINE(
                '            ' || RPAD(vch.redemption_id, 12) ||
                RPAD(vch.reward_name, 34) ||
                'x' || vch.quantity_redeemed ||
                '   expires ' ||
                NVL(TO_CHAR(vch.expiry_date, 'DD-Mon-YYYY'), 'no expiry'));
        END LOOP;
 
        IF v_vouchers = 0 THEN
            DBMS_OUTPUT.PUT_LINE('        Unused vouchers: none');
        END IF;

        v_total_members := v_total_members + 1;
        v_total_recent  := v_total_recent + v_recent_spend;
        v_total_points  := v_total_points + mb.points_balance;
    END LOOP;

    DBMS_OUTPUT.PUT_LINE(' ');
    DBMS_OUTPUT.PUT_LINE(RPAD('=', c_width, '='));
 
    IF v_total_members = 0 THEN
        DBMS_OUTPUT.PUT_LINE('No VIP subscriptions fall due in this window for the selected filters.');
    ELSE
        DBMS_OUTPUT.PUT_LINE('SUMMARY');
        DBMS_OUTPUT.PUT_LINE('  Members to call        : ' || v_total_members ||
            '   (HIGH ' || v_high || ' / MEDIUM ' || v_medium || ' / LOW ' || v_low || ')');
        DBMS_OUTPUT.PUT_LINE('  Expiring within 7 days : ' || v_urgent);
        DBMS_OUTPUT.PUT_LINE('  Spend at risk (' || c_months || 'm)  : RM ' ||
            TO_CHAR(v_total_recent, 'FM999,999,990.00'));
        DBMS_OUTPUT.PUT_LINE('  Avg spend per member   : RM ' ||
            TO_CHAR(ROUND(v_total_recent / v_total_members, 2), 'FM999,999,990.00'));
        DBMS_OUTPUT.PUT_LINE('  Points held by them    : ' ||
            TO_CHAR(v_total_points, 'FM999,999,990'));
        DBMS_OUTPUT.PUT_LINE('  Renewal fee income if all renew : RM ' ||
            TO_CHAR(v_total_members * 12, 'FM999,999,990.00'));
    END IF;
 
    DBMS_OUTPUT.PUT_LINE(RPAD('=', c_width, '='));
    DBMS_OUTPUT.PUT_LINE('*** END OF REPORT ***');
 
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('*** REPORT FAILED: ' || SQLERRM || ' ***');
        RAISE;
END;

-- function 1
CREATE OR REPLACE FUNCTION fn_calc_refund (
    p_order_no IN Orders.order_no%TYPE,
    p_item_id  IN Item.item_id%TYPE,
    p_quantity IN NUMBER,
    p_reason   IN Return_Request.reason%TYPE
) RETURN NUMBER
IS
    v_unit_price Order_Detail.unit_price%TYPE;
    v_line_total Order_Detail.subtotal%TYPE;
    v_refund     NUMBER(10,2);
BEGIN
    SELECT unit_price, subtotal
    INTO v_unit_price, v_line_total
    FROM Order_Detail
    WHERE order_no = p_order_no
    AND item_id  = p_item_id;

    IF p_reason = 'Changed mind after purchase' THEN
        v_refund := p_quantity * v_unit_price * 0.80;   -- 20% restocking fee
    ELSE
        v_refund := p_quantity * v_unit_price;          -- shop / supplier fault
    END IF;

    RETURN ROUND(LEAST(v_refund, v_line_total), 2);
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20045,
            'Item ' || p_item_id || ' was never bought on order ' || p_order_no || '.');
END fn_calc_refund;
/

-- function 2 

CREATE OR REPLACE FUNCTION fn_risk_band (
    p_return_rate IN NUMBER,
    p_avg_score IN NUMBER,
    p_cases IN NUMBER
) RETURN VARCHAR2
IS
BEGIN
    IF NVL(p_cases, 0) = 0 THEN
        RETURN 'NO ISSUE';
    ELSIF NVL(p_return_rate, 0) >= 2 OR NVL(p_avg_score, 5) <= 2 THEN
        RETURN 'CRITICAL';
    ELSIF NVL(p_return_rate, 0) >= 1.5  OR NVL(p_avg_score, 5) <= 3 THEN
        RETURN 'WATCH';
    ELSE
        RETURN 'HEALTHY';
    END IF;
END fn_risk_band;
/

-- function 3

CREATE OR REPLACE FUNCTION fn_sla_priority (
    p_days_waiting IN NUMBER,
    p_membership IN VARCHAR2,
    p_refund_value IN NUMBER
) RETURN VARCHAR2
IS
BEGIN
    IF p_days_waiting > 3 OR p_refund_value >= 100 THEN
        RETURN 'P1-URGENT';
    ELSIF p_days_waiting > 1 OR p_membership = 'VIP' THEN
        RETURN 'P2-HIGH';
    ELSE
        RETURN 'P3-NORMAL';
    END IF;
END fn_sla_priority;
/

-- View table 1

CREATE OR REPLACE VIEW vw_item_return_analysis AS
SELECT i.item_id,
       i.name AS item_name,
       i.category,
       s.company_name AS supplier_name,
       NVL(sold.units_sold, 0) AS units_sold,
       NVL(sold.sales_value, 0) AS sales_value,
       NVL(ret.return_cases, 0) AS return_cases,
       NVL(ret.units_returned, 0) AS units_returned,
       NVL(ret.refund_value, 0) AS refund_value,
       ret.top_reason,
       ROUND(NVL(ret.units_returned, 0) /
             NULLIF(sold.units_sold, 0) * 100, 2) AS return_rate_pct,
       ROUND(NVL(ret.refund_value, 0) /
             NULLIF(sold.sales_value, 0) * 100, 2) AS refund_rate_pct,
       rat.avg_score,
       NVL(rat.rating_count, 0) AS rating_count,
       NVL(rat.low_ratings, 0) AS low_ratings,
       fn_risk_band(ROUND(NVL(ret.units_returned, 0) /
                          NULLIF(sold.units_sold, 0) * 100, 2),
                    rat.avg_score,
                    NVL(ret.return_cases, 0)) AS risk_band
FROM   Item i
JOIN   Supplier s ON s.supplier_id = i.supplier_id
LEFT JOIN (SELECT od.item_id,
                  SUM(od.quantity) AS units_sold,
                  SUM(od.subtotal) AS sales_value
           FROM   Order_Detail od
           JOIN   Orders o ON o.order_no = od.order_no
           WHERE  o.order_status = 'Completed'
           GROUP  BY od.item_id) sold ON sold.item_id = i.item_id
LEFT JOIN (SELECT item_id,
                  SUM(cases) AS return_cases,
                  SUM(units) AS units_returned,
                  SUM(refunds) AS refund_value,
                  MAX(reason) KEEP (DENSE_RANK LAST ORDER BY cases) AS top_reason
           FROM  (SELECT item_id,
                         reason,
                         COUNT(*) AS cases,
                         SUM(return_quantity) AS units,
                         SUM(CASE WHEN status = 'Refunded'
                                  THEN refund_amount ELSE 0 END) AS refunds
                  FROM   Return_Request
                  GROUP  BY item_id, reason)
           GROUP  BY item_id) ret ON ret.item_id = i.item_id
LEFT JOIN (SELECT item_id,
                  ROUND(AVG(score), 2) AS avg_score,
                  COUNT(*) AS rating_count,
                  SUM(CASE WHEN score <= 2 THEN 1 ELSE 0 END) AS low_ratings
           FROM   Rating
           GROUP  BY item_id) rat ON rat.item_id = i.item_id;

-- Procedure 1

CREATE OR REPLACE PROCEDURE prc_process_return (
    p_order_no IN Orders.order_no%TYPE,
    p_item_id  IN Item.item_id%TYPE,
    p_reason   IN Return_Request.reason%TYPE,
    p_quantity IN Return_Request.return_quantity%TYPE,
    p_reviewer IN Employee.employee_id%TYPE,
    p_decision IN VARCHAR2 DEFAULT 'Approved'
) AS
    v_emp_name     Employee.name%TYPE;
    v_emp_position Employee.position%TYPE;
    v_emp_status   Employee.status%TYPE;
    v_item_name    Item.name%TYPE;
    v_return_id    Return_Request.return_id%TYPE;
    v_refund       NUMBER(10,2);
BEGIN
    SELECT e.name, e.position, e.status
    INTO   v_emp_name, v_emp_position, v_emp_status
    FROM   Employee e
    WHERE  e.employee_id = p_reviewer;

    SELECT i.name INTO v_item_name
    FROM   Item i
    WHERE  i.item_id = p_item_id;

    IF p_decision NOT IN ('Requested', 'Approved', 'Rejected', 'Refunded') THEN
        RAISE_APPLICATION_ERROR(-20040,
            'Decision must be Requested, Approved, Rejected or Refunded - got "' || p_decision || '".');
    ELSIF NVL(p_quantity, 0) <= 0 THEN
        RAISE_APPLICATION_ERROR(-20041, 'Return quantity must be at least one unit.');
    ELSIF v_emp_position NOT IN ('Supervisor', 'Manager') OR v_emp_status <> 'Active' THEN
        RAISE_APPLICATION_ERROR(-20042,
            v_emp_name || ' is a ' || v_emp_position || ' (' || v_emp_status ||
            ') and cannot approve a refund. Please call a Supervisor or Manager.');
    END IF;

    v_return_id := 'RTN' || LPAD(return_seq.NEXTVAL, 6, '0');
    v_refund    := fn_calc_refund(p_order_no, p_item_id, p_quantity, p_reason);

    INSERT INTO Return_Request (return_id, reason, request_date, status,
                                refund_amount, remarks, return_quantity,
                                order_no, item_id, reviewed_by)
    VALUES (v_return_id, p_reason, SYSDATE, p_decision, v_refund,
            'Lodged at counter by ' || v_emp_name ||
            CASE WHEN p_decision <> 'Requested'
                 THEN ' | ' || p_decision || ' on ' || TO_CHAR(SYSDATE, 'DD-MON-YYYY HH24:MI')
            END,
            p_quantity, p_order_no, p_item_id, p_reviewer);

    IF p_decision = 'Refunded' THEN
        UPDATE Payment
        SET    payment_status = 'Refunded'
        WHERE  order_no = p_order_no AND payment_status = 'Paid';
    END IF;

    COMMIT;

    DBMS_OUTPUT.PUT_LINE(RPAD('=', 62, '='));
    DBMS_OUTPUT.PUT_LINE(LPAD('88 SPEEDMART - RETURN SLIP', 44));
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 62, '='));
    DBMS_OUTPUT.PUT_LINE(RPAD('Return ID',   18) || ': ' || v_return_id);
    DBMS_OUTPUT.PUT_LINE(RPAD('Order No',    18) || ': ' || p_order_no);
    DBMS_OUTPUT.PUT_LINE(RPAD('Item',        18) || ': ' || v_item_name || ' (' || p_item_id || ')');
    DBMS_OUTPUT.PUT_LINE(RPAD('Quantity',    18) || ': ' || p_quantity);
    DBMS_OUTPUT.PUT_LINE(RPAD('Reason',      18) || ': ' || p_reason);
    DBMS_OUTPUT.PUT_LINE(RPAD('Refund',      18) || ': RM ' || TO_CHAR(v_refund, 'FM9,999,990.00'));
    DBMS_OUTPUT.PUT_LINE(RPAD('Decision',    18) || ': ' || p_decision);
    DBMS_OUTPUT.PUT_LINE(RPAD('Reviewed by', 18) || ': ' || v_emp_name || ' (' || v_emp_position || ')');
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 62, '='));

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        ROLLBACK;
        RAISE_APPLICATION_ERROR(-20043,
            'Employee ' || p_reviewer || ' or item ' || p_item_id || ' does not exist.');
    WHEN DUP_VAL_ON_INDEX THEN
        ROLLBACK;
        RAISE_APPLICATION_ERROR(-20044,
            'Item ' || p_item_id || ' on order ' || p_order_no || ' already has a return request on file.');
    WHEN OTHERS THEN
        ROLLBACK;
        RAISE;
END prc_process_return;
/

CREATE OR REPLACE PROCEDURE prc_rate_item (
    p_order_no IN Orders.order_no%TYPE,
    p_item_id  IN Item.item_id%TYPE,
    p_score    IN Rating.score%TYPE,
    p_comments IN Rating.comments%TYPE DEFAULT NULL
) AS
    e_orphan_line EXCEPTION;
    PRAGMA EXCEPTION_INIT(e_orphan_line, -2291);

    e_order_open  EXCEPTION;                      
    v_status    Orders.order_status%TYPE;
    v_item_name Item.name%TYPE;
    v_rating_id Rating.rating_id%TYPE;
    v_avg       NUMBER(3,2);
    v_count     NUMBER;
BEGIN
    SELECT o.order_status, i.name
    INTO   v_status, v_item_name
    FROM   Orders o, Item i
    WHERE  o.order_no = p_order_no
    AND    i.item_id  = p_item_id;

    IF p_score IS NULL OR p_score NOT BETWEEN 1 AND 5 THEN
        RAISE_APPLICATION_ERROR(-20046,
            'A rating must be a whole number from 1 to 5 - got "' || p_score || '".');
    ELSIF p_score <= 2 AND p_comments IS NULL THEN
        RAISE_APPLICATION_ERROR(-20047,
            'A rating of 1 or 2 star must come with a comment so the buyer can act on it.');
    ELSIF v_status <> 'Completed' THEN
        RAISE e_order_open;                       
    END IF;

    v_rating_id := 'RT' || LPAD(rating_seq.NEXTVAL, 5, '0');

    INSERT INTO Rating (rating_id, score, comments, rating_date, order_no, item_id)
    VALUES (v_rating_id, p_score, p_comments, SYSDATE, p_order_no, p_item_id);

    COMMIT;

    SELECT ROUND(AVG(score), 2), COUNT(*)
    INTO   v_avg, v_count
    FROM   Rating
    WHERE  item_id = p_item_id;

    DBMS_OUTPUT.PUT_LINE(RPAD('-', 62, '-'));
    DBMS_OUTPUT.PUT_LINE('THANK YOU FOR RATING ' || UPPER(v_item_name));
    DBMS_OUTPUT.PUT_LINE(RPAD('-', 62, '-'));
    DBMS_OUTPUT.PUT_LINE(RPAD('Rating ID', 16) || ': ' || v_rating_id);
    DBMS_OUTPUT.PUT_LINE(RPAD('Your score', 16) || ': ' ||
                         RPAD('*', p_score, '*') || RPAD('.', 5 - p_score, '.') ||
                         '  (' || p_score || '/5)');
    DBMS_OUTPUT.PUT_LINE(RPAD('Your comment', 16) || ': ' || NVL(p_comments, '(none given)'));
    DBMS_OUTPUT.PUT_LINE(RPAD('New average', 16) || ': ' ||
                         TO_CHAR(v_avg, 'FM90.00') || ' from ' || v_count || ' rating(s)');
    DBMS_OUTPUT.PUT_LINE(RPAD('-', 62, '-'));

EXCEPTION
    WHEN e_order_open THEN
        DBMS_OUTPUT.PUT_LINE('CANNOT RATE YET : order ' || p_order_no || ' is ' ||
            v_status || '. You may rate an item once the order is completed.');
    WHEN e_orphan_line THEN
        DBMS_OUTPUT.PUT_LINE('INVALID LINE : item ' || p_item_id ||
            ' does not appear on order ' || p_order_no || '.');
    WHEN DUP_VAL_ON_INDEX THEN
        DBMS_OUTPUT.PUT_LINE('ALREADY RATED : you have already rated item ' || p_item_id ||
            ' on order ' || p_order_no || '. One rating per purchase.');
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('NOT FOUND : order ' || p_order_no || ' or item ' ||
            p_item_id || ' does not exist.');
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('UNEXPECTED ERROR ' || SQLCODE || ' : ' || SQLERRM);
END prc_rate_item;
/

CREATE OR REPLACE PROCEDURE rpt_return_risk (
    p_category IN Item.category%TYPE DEFAULT NULL
) AS
    CURSOR c_category IS
        SELECT DISTINCT category
        FROM   vw_item_return_analysis
        WHERE  return_cases > 0
        AND   (p_category IS NULL OR UPPER(category) = UPPER(p_category))
        ORDER  BY category;

    CURSOR c_item (cp_category Item.category%TYPE) IS
        SELECT item_id, item_name, supplier_name, units_sold, units_returned,
               return_rate_pct, refund_value, avg_score, risk_band
        FROM   vw_item_return_analysis
        WHERE  return_cases > 0
        AND    category = cp_category
        ORDER  BY return_rate_pct DESC, refund_value DESC;

    c_width  CONSTANT PLS_INTEGER := 129;
    v_rule   VARCHAR2(200) := RPAD('-', 129, '-');
    v_double VARCHAR2(200) := RPAD('=', 129, '=');

    v_cat_units  NUMBER := 0;   v_cat_refund NUMBER := 0;   v_cat_lines NUMBER := 0;
    v_all_units  NUMBER := 0;   v_all_refund NUMBER := 0;   v_all_lines NUMBER := 0;
    v_critical   NUMBER := 0;   v_watch      NUMBER := 0;

    FUNCTION ctr (p_text IN VARCHAR2) RETURN VARCHAR2 IS
    BEGIN
        RETURN LPAD(p_text, FLOOR((c_width + LENGTH(p_text)) / 2));
    END ctr;
BEGIN
    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(ctr('88 SPEEDMART SDN BHD'));
    DBMS_OUTPUT.PUT_LINE(ctr('HIGH RETURN ITEMS AND REFUND EXPOSURE'));
    DBMS_OUTPUT.PUT_LINE(ctr('Scope : ' || NVL(p_category, 'All categories')));
    DBMS_OUTPUT.PUT_LINE(ctr('As at ' || TO_CHAR(SYSDATE, 'DD-MON-YYYY HH24:MI')));
    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(' ');

    FOR cat IN c_category LOOP                      
        DBMS_OUTPUT.PUT_LINE('CATEGORY : ' || UPPER(cat.category));
        DBMS_OUTPUT.PUT_LINE(
            RPAD('Item',      8) ||
            RPAD('Product',  36) ||
            RPAD('Supplier', 31) ||
            LPAD('Sold',      9) ||
            LPAD('Ret.',      7) ||
            LPAD('Rate %',    9) ||
            LPAD('Refund RM',12) ||
            LPAD('Star',      7) ||
            LPAD('Risk',     10));
        DBMS_OUTPUT.PUT_LINE(v_rule);

        v_cat_units := 0;  v_cat_refund := 0;  v_cat_lines := 0;

        FOR itm IN c_item(cat.category) LOOP         
            DBMS_OUTPUT.PUT_LINE(
                RPAD(itm.item_id, 8) ||
                RPAD(SUBSTR(itm.item_name, 1, 34), 36) ||
                RPAD(SUBSTR(itm.supplier_name, 1, 29), 31) ||
                LPAD(TO_CHAR(itm.units_sold,      'FM999,990'),    9) ||
                LPAD(TO_CHAR(itm.units_returned,  'FM9,990'),      7) ||
                LPAD(TO_CHAR(itm.return_rate_pct, 'FM990.00'),     9) ||
                LPAD(TO_CHAR(itm.refund_value,    'FM999,990.00'),12) ||
                LPAD(NVL(TO_CHAR(itm.avg_score,   'FM90.00'), 'n/a'), 7) ||
                LPAD(itm.risk_band, 10));

            v_cat_units  := v_cat_units  + itm.units_returned;
            v_cat_refund := v_cat_refund + itm.refund_value;
            v_cat_lines  := v_cat_lines  + 1;
            IF    itm.risk_band = 'CRITICAL' THEN v_critical := v_critical + 1;
            ELSIF itm.risk_band = 'WATCH'    THEN v_watch    := v_watch    + 1;
            END IF;
        END LOOP;

        DBMS_OUTPUT.PUT_LINE(v_rule);
        DBMS_OUTPUT.PUT_LINE(
            RPAD('Subtotal  (' || v_cat_lines || ' item(s))', 74) ||
            LPAD(TO_CHAR(v_cat_units,  'FM9,990'),       7) ||
            LPAD(TO_CHAR(v_cat_refund, 'FM999,990.00'), 21));
        DBMS_OUTPUT.PUT_LINE(' ');

        v_all_units  := v_all_units  + v_cat_units;
        v_all_refund := v_all_refund + v_cat_refund;
        v_all_lines  := v_all_lines  + v_cat_lines;
    END LOOP;

    IF v_all_lines = 0 THEN
        DBMS_OUTPUT.PUT_LINE(ctr('No returns recorded for ' ||
            NVL(p_category, 'any category') || '.'));
        DBMS_OUTPUT.PUT_LINE(v_double);
        RETURN;
    END IF;

    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(ctr('SUMMARY'));
    DBMS_OUTPUT.PUT_LINE(v_rule);
    DBMS_OUTPUT.PUT_LINE(RPAD('Items with returns', 30) || ': ' ||
        LPAD(v_all_lines, 12));
    DBMS_OUTPUT.PUT_LINE(RPAD('   CRITICAL band', 30) || ': ' ||
        LPAD(v_critical, 12) || '   (' ||
        TO_CHAR(ROUND(v_critical / v_all_lines * 100, 1), 'FM990.0') || '%)');
    DBMS_OUTPUT.PUT_LINE(RPAD('   WATCH band', 30) || ': ' ||
        LPAD(v_watch, 12) || '   (' ||
        TO_CHAR(ROUND(v_watch / v_all_lines * 100, 1), 'FM990.0') || '%)');
    DBMS_OUTPUT.PUT_LINE(RPAD('Total units returned', 30) || ': ' ||
        LPAD(TO_CHAR(v_all_units, 'FM999,990'), 12));
    DBMS_OUTPUT.PUT_LINE(RPAD('Total refund exposure', 30) || ': ' ||
        LPAD('RM ' || TO_CHAR(v_all_refund, 'FM999,999,990.00'), 12));
    DBMS_OUTPUT.PUT_LINE(RPAD('Average refund per item', 30) || ': ' ||
        LPAD('RM ' || TO_CHAR(ROUND(v_all_refund / v_all_lines, 2), 'FM999,990.00'), 12));
    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(LPAD('End of Report 1', c_width));

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('REPORT FAILED ' || SQLCODE || ' : ' || SQLERRM);
END rpt_return_risk;

CREATE OR REPLACE PROCEDURE rpt_pending_returns (
    p_branch_id IN Branch.branch_id%TYPE DEFAULT NULL
) AS
    CURSOR c_branch IS
        SELECT   branch_id,
                 branch_name,
                 COUNT(*)              AS cases,
                 SUM(refund_exposure)  AS exposure
        FROM     vw_pending_return_queue
        WHERE   (p_branch_id IS NULL OR branch_id = p_branch_id)
        GROUP BY branch_id, branch_name
        ORDER BY SUM(refund_exposure) DESC;

    CURSOR c_case (cp_branch Branch.branch_id%TYPE) IS
        SELECT return_id, order_no, item_name, return_quantity, reason,
               customer_name, membership_type, days_waiting,
               refund_exposure, priority
        FROM   vw_pending_return_queue
        WHERE  branch_id = cp_branch
        ORDER  BY CASE priority WHEN 'P1-URGENT' THEN 1
                                WHEN 'P2-HIGH'   THEN 2
                                ELSE 3 END,
                  days_waiting DESC, refund_exposure DESC;

    c_width  CONSTANT PLS_INTEGER := 155;
    v_rule   VARCHAR2(200) := RPAD('-', 155, '-');
    v_double VARCHAR2(200) := RPAD('=', 155, '=');

    v_branch_sum NUMBER(12,2) := 0;   v_branch_cnt NUMBER := 0;
    v_grand_sum  NUMBER(12,2) := 0;   v_grand_cnt  NUMBER := 0;
    v_p1 NUMBER := 0;  v_p2 NUMBER := 0;  v_p3 NUMBER := 0;
    v_oldest NUMBER := 0;

    FUNCTION ctr (p_text IN VARCHAR2) RETURN VARCHAR2 IS
    BEGIN
        RETURN LPAD(p_text, FLOOR((c_width + LENGTH(p_text)) / 2));
    END ctr;

    FUNCTION money_line (p_label IN VARCHAR2, p_amount IN NUMBER) RETURN VARCHAR2 IS
    BEGIN
        RETURN RPAD(p_label, 134) || LPAD(TO_CHAR(p_amount, 'FM999,990.00'), 10);
    END money_line;
BEGIN
    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(ctr('88 SPEEDMART SDN BHD'));
    DBMS_OUTPUT.PUT_LINE(ctr('VOICE OF CUSTOMER - PENDING RETURN REQUESTS'));
    DBMS_OUTPUT.PUT_LINE(ctr('Scope : ' || NVL(p_branch_id, 'All branches')));
    DBMS_OUTPUT.PUT_LINE(ctr('As at ' || TO_CHAR(SYSDATE, 'DD-MON-YYYY HH24:MI')));
    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(' ');

    FOR br IN c_branch LOOP                              -- OUTER CURSOR
        DBMS_OUTPUT.PUT_LINE('BRANCH : ' || UPPER(br.branch_name) ||
                             '  (' || br.branch_id || ')');
        DBMS_OUTPUT.PUT_LINE(
            RPAD('Return',   10) ||
            RPAD('Order',    10) ||
            RPAD('Item',     51) ||
            RPAD('Qty',       5) ||
            RPAD('Reason',   29) ||
            RPAD('Customer', 24) ||
            LPAD('Days',      5) ||
            LPAD('RM',       10) ||
            '  Priority');
        DBMS_OUTPUT.PUT_LINE(v_rule);

        v_branch_sum := 0;  v_branch_cnt := 0;

        FOR r IN c_case(br.branch_id) LOOP               -- INNER CURSOR
            DBMS_OUTPUT.PUT_LINE(
                RPAD(r.return_id, 10) ||
                RPAD(r.order_no,  10) ||
                RPAD(SUBSTR(r.item_name, 1, 49), 51) ||
                RPAD(r.return_quantity, 5) ||
                RPAD(SUBSTR(r.reason, 1, 27), 29) ||
                RPAD(SUBSTR(r.customer_name, 1, 22), 24) ||
                LPAD(r.days_waiting, 5) ||
                LPAD(TO_CHAR(r.refund_exposure, 'FM999,990.00'), 10) ||
                '  ' || r.priority);

            v_branch_sum := v_branch_sum + r.refund_exposure;
            v_branch_cnt := v_branch_cnt + 1;
            v_oldest     := GREATEST(v_oldest, r.days_waiting);

            CASE r.priority
                WHEN 'P1-URGENT' THEN v_p1 := v_p1 + 1;
                WHEN 'P2-HIGH'   THEN v_p2 := v_p2 + 1;
                ELSE                  v_p3 := v_p3 + 1;
            END CASE;
        END LOOP;

        DBMS_OUTPUT.PUT_LINE(v_rule);
        DBMS_OUTPUT.PUT_LINE(
            money_line('Branch subtotal  (' || v_branch_cnt || ' case(s))',
                       v_branch_sum));
        DBMS_OUTPUT.PUT_LINE(' ');

        v_grand_sum := v_grand_sum + v_branch_sum;
        v_grand_cnt := v_grand_cnt + v_branch_cnt;
    END LOOP;

    IF v_grand_cnt = 0 THEN
        DBMS_OUTPUT.PUT_LINE(ctr('No return request is waiting for a decision.'));
        DBMS_OUTPUT.PUT_LINE(v_double);
        RETURN;
    END IF;

    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(ctr('SUMMARY'));
    DBMS_OUTPUT.PUT_LINE(v_rule);
    DBMS_OUTPUT.PUT_LINE(RPAD('Total cases waiting', 36) || ': ' || v_grand_cnt);
    DBMS_OUTPUT.PUT_LINE(RPAD('   P1 - Urgent', 36) || ': ' || RPAD(v_p1, 6) ||
        '(' || TO_CHAR(ROUND(v_p1 / v_grand_cnt * 100, 1), 'FM990.0') || '%)');
    DBMS_OUTPUT.PUT_LINE(RPAD('   P2 - High', 36) || ': ' || RPAD(v_p2, 6) ||
        '(' || TO_CHAR(ROUND(v_p2 / v_grand_cnt * 100, 1), 'FM990.0') || '%)');
    DBMS_OUTPUT.PUT_LINE(RPAD('   P3 - Normal', 36) || ': ' || RPAD(v_p3, 6) ||
        '(' || TO_CHAR(ROUND(v_p3 / v_grand_cnt * 100, 1), 'FM990.0') || '%)');
    DBMS_OUTPUT.PUT_LINE(RPAD('Longest wait', 36) || ': ' || v_oldest || ' day(s)');
    DBMS_OUTPUT.PUT_LINE(RPAD('Total refund exposure', 36) || ': RM ' ||
        TO_CHAR(v_grand_sum, 'FM999,999,990.00'));
    DBMS_OUTPUT.PUT_LINE(RPAD('Average value per case', 36) || ': RM ' ||
        TO_CHAR(ROUND(v_grand_sum / v_grand_cnt, 2), 'FM999,990.00'));
    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(LPAD('End of Report 2', c_width));

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('REPORT FAILED ' || SQLCODE || ' : ' || SQLERRM);
END rpt_pending_returns;
/

-- View 1
CREATE OR REPLACE VIEW v_branch_monthly_revenue AS
SELECT
    EXTRACT(YEAR FROM O.order_date_time) AS sales_year,
    TRUNC(CAST(O.order_date_time AS DATE), 'MM') AS sales_month,
    B.branch_id,
    B.name AS branch_name,
    COUNT(O.order_no) AS total_orders,
    COUNT(DISTINCT O.customer_id) AS total_customers,
    ROUND(SUM(O.subtotal_amount), 2) AS gross_sales,
    ROUND(SUM(
        CASE
            WHEN R.reward_type = 'Discount' THEN O.discount_amount
            ELSE 0
        END
    ), 2) AS voucher_discount,
    ROUND(SUM(
        CASE
            WHEN P.payment_status = 'Refunded' THEN NVL(RBO.refund_amount, 0)
            ELSE 0
        END
    ), 2) AS refund_amount,
    ROUND(SUM(
        O.subtotal_amount
        - CASE
            WHEN R.reward_type = 'Discount' THEN O.discount_amount
            ELSE 0
          END
        - CASE
            WHEN P.payment_status = 'Refunded' THEN NVL(RBO.refund_amount, 0)
            ELSE 0
          END
    ), 2) AS net_sales,
    ROUND(AVG(
        O.subtotal_amount
        - CASE
            WHEN R.reward_type = 'Discount' THEN O.discount_amount
            ELSE 0
          END
        - CASE
            WHEN P.payment_status = 'Refunded' THEN NVL(RBO.refund_amount, 0)
            ELSE 0
          END
    ), 2) AS avg_net_sales_per_order
FROM Branch B
JOIN Orders O
    ON B.branch_id = O.branch_id
JOIN Payment P
    ON O.order_no = P.order_no
LEFT JOIN Reward_Redemption RR
    ON O.redemption_id = RR.redemption_id
LEFT JOIN Reward R
    ON RR.reward_id = R.reward_id
LEFT JOIN v_refund_by_order RBO
    ON O.order_no = RBO.order_no
WHERE O.order_status = 'Completed'
AND P.payment_status IN ('Paid', 'Refunded')
GROUP BY
    EXTRACT(YEAR FROM O.order_date_time),
    TRUNC(CAST(O.order_date_time AS DATE), 'MM'),
    B.branch_id,
    B.name;

-- View 2
CREATE OR REPLACE VIEW v_mykasih_branch_category AS
SELECT
    TRUNC(CAST(O.order_date_time AS DATE), 'MM') AS sales_month,
    B.branch_id,
    B.name AS branch_name,
    I.category,
    COUNT(DISTINCT O.order_no) AS mykasih_orders,
    COUNT(DISTINCT C.customer_id) AS mykasih_customers,
    COUNT(DISTINCT I.item_id) AS different_items,
    SUM(OD.quantity) AS total_units,
    ROUND(SUM(OD.subtotal), 2) AS gross_mykasih_sales,
    ROUND(
        SUM(
            CASE
                WHEN R.reward_type = 'Discount'
                    AND O.subtotal_amount > 0
                THEN (OD.subtotal / O.subtotal_amount) * O.discount_amount
                ELSE 0
            END
        ),
        2
    ) AS voucher_discount,
    ROUND(
        SUM(
            CASE
                WHEN P.payment_status = 'Refunded'
                THEN NVL(RTN.refund_amount, 0)
                ELSE 0
            END
        ),
        2
    ) AS refund_amount,
    ROUND(
        SUM(
            OD.subtotal -
            CASE
                WHEN R.reward_type = 'Discount'
                    AND O.subtotal_amount > 0
                THEN (OD.subtotal / O.subtotal_amount) * O.discount_amount
                ELSE 0
            END -
            CASE
                WHEN P.payment_status = 'Refunded'
                THEN NVL(RTN.refund_amount, 0)
                ELSE 0
            END
        ),
        2
    ) AS net_mykasih_sales
FROM Branch B
JOIN Orders O
    ON B.branch_id = O.branch_id
JOIN Customer C
    ON O.customer_id = C.customer_id
JOIN Payment P
    ON O.order_no = P.order_no
JOIN Order_Detail OD
    ON O.order_no = OD.order_no
JOIN Item I
    ON OD.item_id = I.item_id
LEFT JOIN Reward_Redemption RR
    ON O.redemption_id = RR.redemption_id
LEFT JOIN Reward R
    ON RR.reward_id = R.reward_id
LEFT JOIN Return_Request RTN
    ON O.order_no = RTN.order_no
    AND OD.item_id = RTN.item_id
    AND RTN.status = 'Refunded'
WHERE O.order_status = 'Completed'
  AND P.payment_status IN ('Paid', 'Refunded')
  AND O.is_mykasih_purchase = 'Y'
  AND P.payment_method = 'MyKasih'
  AND C.ic_number IS NOT NULL
  AND I.is_mykasih_eligible = 'Y'
GROUP BY
    TRUNC(CAST(O.order_date_time AS DATE), 'MM'),
    B.branch_id,
    B.name,
    I.category;

--additional view (Refund View — By Order)
CREATE OR REPLACE VIEW v_refund_by_order AS
SELECT
    order_no,
    ROUND(SUM(refund_amount), 2) AS refund_amount
FROM Return_Request
WHERE status = 'Refunded'
GROUP BY order_no;

WITH mykasih_analysis AS (
    SELECT
        V.*,
        DENSE_RANK() OVER (
            PARTITION BY V.sales_month, V.branch_id
            ORDER BY V.net_mykasih_sales DESC
        ) AS category_rank,
        ROUND(
            V.net_mykasih_sales * 100 /
            NULLIF(
                SUM(V.net_mykasih_sales) OVER (
                    PARTITION BY V.sales_month, V.branch_id
                ),
                0
            ),
            1
        ) AS category_share_pct
    FROM v_mykasih_branch_category V
)

    CASE
        WHEN category_rank = 1 THEN 'Top Demand'
        WHEN category_share_pct >= 25 THEN 'High Demand'
        WHEN category_share_pct >= 10 THEN 'Moderate'
        ELSE 'Low Demand'
    END AS demand_status
FROM mykasih_analysis

-- Procedure 1 (add order)

CREATE OR REPLACE PROCEDURE prc_place_order (
    p_customer_id      IN Customer.customer_id%TYPE,
    p_branch_id        IN Branch.branch_id%TYPE,
    p_employee_id      IN Employee.employee_id%TYPE,

    p_item_ids         IN SYS.ODCIVARCHAR2LIST,
    p_quantities       IN SYS.ODCINUMBERLIST,

    p_order_type       IN Orders.order_type%TYPE,
    p_service_type     IN Orders.service_type%TYPE,
    p_payment_method   IN Payment.payment_method%TYPE,

    p_mykasih          IN CHAR DEFAULT 'N',
    p_redemption_id    IN Reward_Redemption.redemption_id%TYPE
                           DEFAULT NULL,

    p_order_no         OUT Orders.order_no%TYPE
)
IS
    -- Customer
    v_customer_ic       Customer.ic_number%TYPE;
    v_customer_member   Customer.member_id%TYPE;

    -- Branch
    v_branch_name       Branch.name%TYPE;
    v_branch_status     Branch.status%TYPE;
    v_open_time         Branch.open_time%TYPE;
    v_close_time        Branch.close_time%TYPE;

    -- Employee
    v_emp_name          Employee.name%TYPE;
    v_emp_status        Employee.status%TYPE;
    v_emp_branch        Employee.branch_id%TYPE;

    -- Item / Inventory
    v_item_name         Item.name%TYPE;
    v_item_status       Item.status%TYPE;
    v_item_price        Item.unit_price%TYPE;
    v_mykasih_eligible  Item.is_mykasih_eligible%TYPE;
    v_stock             Inventory.quantity%TYPE;

    -- Order calculation
    v_item_count        NUMBER;
    v_subtotal          NUMBER(10,2) := 0;
    v_discount          NUMBER(10,2) := 0;
    v_total             NUMBER(10,2) := 0;

    -- Voucher
    v_red_member        Reward_Redemption.member_id%TYPE;
    v_red_status        Reward_Redemption.status%TYPE;
    v_expiry_date       Reward_Redemption.expiry_date%TYPE;

    v_reward_name       Reward.name%TYPE;
    v_reward_status     Reward.status%TYPE;
    v_reward_value      Reward.value%TYPE;

    -- Payment
    v_payment_id        Payment.payment_id%TYPE;

BEGIN
    -- VALIDATE ITEM COLLECTIONS 
    BEGIN
        v_item_count := p_item_ids.COUNT;

        IF v_item_count = 0 THEN
            RAISE_APPLICATION_ERROR(
                -20062,
                'The shopping basket cannot be empty.'
            );
        END IF;

        IF p_quantities.COUNT <> p_item_ids.COUNT THEN
            RAISE_APPLICATION_ERROR(
                -20063,
                'The number of item IDs must match the number of quantities.'
            );
        END IF;

    EXCEPTION
        WHEN COLLECTION_IS_NULL THEN
            RAISE_APPLICATION_ERROR(
                -20401,
                'The shopping basket cannot be empty.'
            );
    END;

    -- VALIDATE ORDER TYPE
    IF p_order_type NOT IN ('Online', 'Walk-In') THEN
        RAISE_APPLICATION_ERROR(
            -20403,
            'Order type must be Online or Walk-In.'
        );
    END IF;

    -- VALIDATE SERVICE TYPE
    IF p_service_type NOT IN ('Pickup', 'Delivery') THEN
        RAISE_APPLICATION_ERROR(
            -20404,
            'Service type must be Pickup or Delivery.'
        );
    END IF;

    -- Walk-In cannot use delivery
    IF p_order_type = 'Walk-In'
       AND p_service_type = 'Delivery' THEN

        RAISE_APPLICATION_ERROR(
            -20405,
            'Walk-In orders cannot use Delivery service.'
        );

    END IF;

    -- VALIDATE PAYMENT METHOD
    IF p_payment_method NOT IN (
        'Cash',
        'Card',
        'E-Wallet',
        'MyKasih',
        'Online Banking'
    ) THEN

        RAISE_APPLICATION_ERROR(
            -20406,
            'Invalid payment method.'
        );

    END IF;

    -- VALIDATE MYKASIH OPTION
    IF UPPER(p_mykasih) NOT IN ('Y', 'N') THEN

        RAISE_APPLICATION_ERROR(
            -20407,
            'MyKasih option must be Y or N.'
        );

    END IF;

    -- VALIDATE CUSTOMER
    BEGIN

        SELECT
            ic_number,
            member_id

        INTO
            v_customer_ic,
            v_customer_member

        FROM Customer

        WHERE customer_id = p_customer_id;

    EXCEPTION
        WHEN NO_DATA_FOUND THEN

            RAISE_APPLICATION_ERROR(
                -20060,
                'Customer ' || p_customer_id ||
                ' does not exist.'
            );
    END;

    -- VALIDATE BRANCH
    BEGIN

        SELECT
            name,
            status,
            open_time,
            close_time

        INTO
            v_branch_name,
            v_branch_status,
            v_open_time,
            v_close_time

        FROM Branch

        WHERE branch_id = p_branch_id;

    EXCEPTION
        WHEN NO_DATA_FOUND THEN

            RAISE_APPLICATION_ERROR(
                -20409,
                'Branch ' || p_branch_id ||
                ' does not exist.'
            );
    END;

    IF v_branch_status <> 'Active' THEN

        RAISE_APPLICATION_ERROR(
            -20410,
            v_branch_name || ' is currently ' ||
            v_branch_status ||
            ' and cannot accept new orders.'
        );

    END IF;

    -- BUSINESS OPERATING DAY / TIME
    IF TO_CHAR(
           SYSDATE,
           'DY',
           'NLS_DATE_LANGUAGE=ENGLISH'
       ) = 'SUN' THEN

        RAISE_APPLICATION_ERROR(
            -20411,
            'Orders cannot be processed on Sunday.'
        );

    END IF;

    IF TO_CHAR(SYSDATE, 'HH24:MI') < v_open_time
       OR
       TO_CHAR(SYSDATE, 'HH24:MI') >= v_close_time THEN

        RAISE_APPLICATION_ERROR(
            -20412,
            v_branch_name ||
            ' is currently outside its operating hours (' ||
            v_open_time || ' - ' ||
            v_close_time || ').'
        );

    END IF;

    -- VALIDATE EMPLOYEE
    BEGIN
        SELECT
            name,
            status,
            branch_id
        INTO
            v_emp_name,
            v_emp_status,
            v_emp_branch
        FROM Employee
        WHERE employee_id = p_employee_id;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(
                -20413,
                'Employee ' || p_employee_id ||
                ' does not exist.'
            );
    END;

    IF v_emp_status <> 'Active' THEN
        RAISE_APPLICATION_ERROR(
            -20414,
            'Employee ' || p_employee_id ||
            ' - ' || v_emp_name ||
            ' is inactive and cannot process an order.'
        );
    END IF;

    -- Employee can only process order at own branch
    IF v_emp_branch <> p_branch_id THEN
        RAISE_APPLICATION_ERROR(
            -20415,
            'Employee ' || p_employee_id ||
            ' is assigned to branch ' ||
            v_emp_branch ||
            ' and cannot process an order for branch ' ||
            p_branch_id || '.'
        );
    END IF;

    -- MYKASIH CUSTOMER VALIDATION
    IF UPPER(p_mykasih) = 'Y'
       AND v_customer_ic IS NULL THEN
        RAISE_APPLICATION_ERROR(
            -20416,
            'Customer ' || p_customer_id ||
            ' must present an IC for a MyKasih purchase.'
        );
    END IF;

    -- MyKasih order must uses MyKasih payment method.
    IF UPPER(p_mykasih) = 'Y'
       AND p_payment_method <> 'MyKasih' THEN
        RAISE_APPLICATION_ERROR(
            -20417,
            'A MyKasih order must use MyKasih as the payment method.'
        );
    END IF;

    IF UPPER(p_mykasih) = 'N'
       AND p_payment_method = 'MyKasih' THEN
        RAISE_APPLICATION_ERROR(
            -20418,
            'MyKasih payment cannot be used for a non-MyKasih order.'
        );
    END IF;

    -- VALIDATE EVERY ITEM + CALCULATE SUBTOTAL
    FOR i IN 1 .. p_item_ids.COUNT LOOP
        -- Quantity
        IF p_quantities(i) IS NULL
           OR p_quantities(i) <= 0 THEN
            RAISE_APPLICATION_ERROR(
                -20419,
                'Quantity for item ' ||
                p_item_ids(i) ||
                ' must be greater than zero.'
            );
        END IF;

        -- Do not allow same item twice
        IF i > 1 THEN
            FOR j IN 1 .. i - 1 LOOP
                IF p_item_ids(i) = p_item_ids(j) THEN
                    RAISE_APPLICATION_ERROR(
                        -20420,
                        'Item ' ||
                        p_item_ids(i) ||
                        ' appears more than once in the basket.'
                    );
                END IF;
            END LOOP;
        END IF;

        -- Check Item
        BEGIN
            SELECT
                name,
                status,
                unit_price,
                is_mykasih_eligible
            INTO
                v_item_name,
                v_item_status,
                v_item_price,
                v_mykasih_eligible
            FROM Item
            WHERE item_id = p_item_ids(i);
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                RAISE_APPLICATION_ERROR(
                    -20421,
                    'Item ' || p_item_ids(i) ||
                    ' does not exist.'
                );
        END;

        IF v_item_status <> 'Active' THEN
            RAISE_APPLICATION_ERROR(
                -20422,
                v_item_name ||
                ' is inactive and cannot be ordered.'
            );
        END IF;

        -- MyKasih item validation
        IF UPPER(p_mykasih) = 'Y'
           AND v_mykasih_eligible <> 'Y' THEN
            RAISE_APPLICATION_ERROR(
                -20423,
                v_item_name ||
                ' is not eligible for MyKasih purchase.'
            );
        END IF;

        -- Check Inventory
        BEGIN
            SELECT quantity
            INTO v_stock
            FROM Inventory
            WHERE branch_id = p_branch_id
              AND item_id = p_item_ids(i)
            FOR UPDATE;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                RAISE_APPLICATION_ERROR(
                    -20424,
                    v_item_name ||
                    ' is not stocked at branch ' ||
                    p_branch_id || '.'
                );
        END;

        IF v_stock < p_quantities(i) THEN
            RAISE_APPLICATION_ERROR(
                -20425,
                'Insufficient stock for ' ||
                v_item_name ||
                '. Available: ' ||
                v_stock ||
                ', requested: ' ||
                p_quantities(i) || '.'
            );
        END IF;

        -- Order subtotal
        v_subtotal :=
            v_subtotal +
            (p_quantities(i) * v_item_price);

    END LOOP;

    -- OPTIONAL VOUCHER VALIDATION
    IF p_redemption_id IS NOT NULL THEN
        BEGIN
            SELECT
                RR.member_id,
                RR.status,
                RR.expiry_date,
                R.name,
                R.status,
                R.value
            INTO
                v_red_member,
                v_red_status,
                v_expiry_date,
                v_reward_name,
                v_reward_status,
                v_reward_value
            FROM Reward_Redemption RR
            JOIN Reward R
                ON RR.reward_id = R.reward_id
            WHERE RR.redemption_id =
                  p_redemption_id;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                RAISE_APPLICATION_ERROR(
                    -20426,
                    'Redemption ' ||
                    p_redemption_id ||
                    ' does not exist.'
                );
        END;

        -- Only member owning redemption can use it
        IF v_customer_member IS NULL
           OR
           v_customer_member <> v_red_member THEN

            RAISE_APPLICATION_ERROR(
                -20427,
                'Redemption ' ||
                p_redemption_id ||
                ' does not belong to this customer.'
            );
        END IF;

        IF v_red_status <> 'Active' THEN
            RAISE_APPLICATION_ERROR(
                -20428,
                'Redemption ' ||
                p_redemption_id ||
                ' is ' ||
                v_red_status ||
                ' and cannot be used.'
            );
        END IF;

        IF v_expiry_date IS NOT NULL
           AND v_expiry_date < TRUNC(SYSDATE) THEN

            RAISE_APPLICATION_ERROR(
                -20429,
                'Redemption ' ||
                p_redemption_id ||
                ' expired on ' ||
                TO_CHAR(
                    v_expiry_date,
                    'DD-MON-YYYY'
                ) || '.'
            );
        END IF;

        IF v_reward_status <> 'Active' THEN
            RAISE_APPLICATION_ERROR(
                -20430,
                'Reward ' ||
                v_reward_name ||
                ' is no longer active.'
            );
        END IF;

        v_discount := v_reward_value;

        IF v_discount > v_subtotal THEN
            RAISE_APPLICATION_ERROR(
                -20431,
                'Voucher value RM ' ||
                TO_CHAR(v_discount, 'FM9990.00') ||
                ' exceeds the order subtotal RM ' ||
                TO_CHAR(v_subtotal, 'FM9990.00') || '.'
            );
        END IF;
    END IF;

    -- CALCULATE CURRENT TOTAL
    v_total := v_subtotal - v_discount;

    -- MYKASIH RM100 PURCHASE LIMIT
    IF UPPER(p_mykasih) = 'Y'
    AND v_total > 100 THEN

        RAISE_APPLICATION_ERROR(
            -20105,
            'MyKasih purchase amount cannot exceed RM100.00. ' ||
            'Current amount: RM ' ||
            TO_CHAR(v_total, 'FM9990.00') || '.'
        );

    END IF;

    -- For Delivery orders the delivery fee is calculated later
    -- when the Delivery_Request is created.

    -- GENERATE ORDER NUMBER
    p_order_no :=
        'ORD' ||
        TO_CHAR(
            order_seq.NEXTVAL,
            'FM000000'
        );

    -- INSERT ORDER HEADER
    INSERT INTO Orders (
        order_no,
        order_date_time,
        order_type,
        service_type,
        subtotal_amount,
        total_amount,
        points_earned,
        discount_amount,
        order_status,
        is_mykasih_purchase,
        redemption_id,
        customer_id,
        branch_id,
        handled_by
    )
    VALUES (
        p_order_no,
        SYSTIMESTAMP,
        p_order_type,
        p_service_type,
        v_subtotal,
        v_total,
        0,
        v_discount,
        'Processing',
        UPPER(p_mykasih),
        p_redemption_id,
        p_customer_id,
        p_branch_id,
        p_employee_id
    );

    -- INSERT ORDER DETAILS + REDUCE INVENTORY
    FOR i IN 1 .. p_item_ids.COUNT LOOP
        SELECT
            name,
            unit_price
        INTO
            v_item_name,
            v_item_price
        FROM Item
        WHERE item_id = p_item_ids(i);

        INSERT INTO Order_Detail (
            order_no,
            item_id,
            subtotal,
            quantity,
            unit_price
        )
        VALUES (
            p_order_no,
            p_item_ids(i),
            p_quantities(i) * v_item_price,
            p_quantities(i),
            v_item_price
        );

        -- Inventory immediately updated
        UPDATE Inventory
        SET quantity =
            quantity - p_quantities(i)
        WHERE branch_id = p_branch_id
          AND item_id = p_item_ids(i);
    END LOOP;

    -- USE VOUCHER
    IF p_redemption_id IS NOT NULL THEN
        UPDATE Reward_Redemption
        SET status = 'Used'
        WHERE redemption_id =
              p_redemption_id;
    END IF;

    -- CREATE ONE PAYMENT RECORD
    v_payment_id :=
        'PAY' ||
        TO_CHAR(
            payment_seq.NEXTVAL,
            'FM000000'
        );

    INSERT INTO Payment (
        payment_id,
        payment_method,
        payment_date_time,
        reference_no,
        payment_status,
        order_no
    )
    VALUES (
        v_payment_id,
        p_payment_method,
        SYSTIMESTAMP,
        NULL,
        'Pending',
        p_order_no
    );

    COMMIT;

    -- ORDER CONFIRMATION MESSAGE
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 72, '='));
    DBMS_OUTPUT.PUT_LINE('88 SPEEDMART - ORDER PLACEMENT CONFIRMATION');
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 72, '='));
    DBMS_OUTPUT.PUT_LINE(RPAD('Order No', 20) ||': ' || p_order_no);
    DBMS_OUTPUT.PUT_LINE(RPAD('Branch', 20) ||': ' || p_branch_id ||' - ' || v_branch_name);
    DBMS_OUTPUT.PUT_LINE(RPAD('Customer', 20) ||': ' || p_customer_id);
    DBMS_OUTPUT.PUT_LINE(RPAD('Handled By', 20) ||': ' || p_employee_id ||' - ' || v_emp_name);
    DBMS_OUTPUT.PUT_LINE(RPAD('Items', 20) ||': ' || p_item_ids.COUNT);
    DBMS_OUTPUT.PUT_LINE(RPAD('Subtotal', 20) ||': RM ' ||TO_CHAR(v_subtotal,'FM999,990.00'));
    DBMS_OUTPUT.PUT_LINE(RPAD('Voucher Discount', 20) ||': RM ' ||TO_CHAR(v_discount,'FM999,990.00'));
    DBMS_OUTPUT.PUT_LINE(RPAD('Current Total', 20) ||': RM ' ||TO_CHAR(v_total,'FM999,990.00'));

    IF p_service_type = 'Delivery' THEN
        DBMS_OUTPUT.PUT_LINE(
            'NOTE: Delivery fee will be added when the delivery request is dispatched.'
        );
    END IF;

    DBMS_OUTPUT.PUT_LINE(RPAD('Payment', 20) ||': ' ||p_payment_method ||' (Pending)');
    DBMS_OUTPUT.PUT_LINE(RPAD('Order Status', 20) ||': Processing');
    DBMS_OUTPUT.PUT_LINE(RPAD('MyKasih', 20) ||': ' || UPPER(p_mykasih));
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 72, '='));

EXCEPTION
    -- Duplicate order/redemption/order-detail
    WHEN DUP_VAL_ON_INDEX THEN
        ROLLBACK;
        p_order_no := NULL;

        RAISE_APPLICATION_ERROR(
            -20432,
            'Duplicate data detected.'
        );

    -- Unexpected error
    WHEN OTHERS THEN
        ROLLBACK;
        p_order_no := NULL;

        DBMS_OUTPUT.PUT_LINE(
            '=== ORDER PLACEMENT FAILED ==='
        );

        DBMS_OUTPUT.PUT_LINE(
            'Error Code    : ' || SQLCODE
        );

        DBMS_OUTPUT.PUT_LINE(
            'Error Message : ' || SQLERRM
        );
        RAISE;

END prc_place_order;
/


-- Procedure 2 (search item )
CREATE OR REPLACE PROCEDURE prc_search_items (
    p_branch_id       IN Branch.branch_id%TYPE,
    p_keyword         IN VARCHAR2 DEFAULT NULL,
    p_mykasih_only    IN CHAR DEFAULT 'N'
)
IS
    v_branch_name     Branch.name%TYPE;
    v_branch_status   Branch.status%TYPE;
    v_found           NUMBER := 0;
    e_no_items_found  EXCEPTION;

    CURSOR c_item IS
        SELECT
            I.item_id,
            I.name AS item_name,
            I.category,
            I.unit_price,
            INV.quantity,
            I.is_mykasih_eligible
        FROM Branch B
        JOIN Inventory INV
            ON B.branch_id =
               INV.branch_id
        JOIN Item I
            ON INV.item_id =
               I.item_id
        WHERE B.branch_id =
              p_branch_id
          AND (
                p_keyword IS NULL
                OR
                UPPER(I.name) LIKE
                    '%' ||
                    UPPER(TRIM(p_keyword)) ||
                    '%'
              )
          AND I.status = 'Active'
          AND INV.quantity > 0
          AND (
                UPPER(p_mykasih_only) = 'N'
                OR
                I.is_mykasih_eligible = 'Y'
              )
        ORDER BY I.name;

BEGIN
    -- VALIDATE Y/N
    IF UPPER(p_mykasih_only)
       NOT IN ('Y', 'N') THEN

        RAISE_APPLICATION_ERROR(
            -20441,
            'MyKasih search option must be Y or N.'
        );
    END IF;

    -- VALIDATE BRANCH
    BEGIN
        SELECT
            name,
            status
        INTO
            v_branch_name,
            v_branch_status
        FROM Branch
        WHERE branch_id =
              p_branch_id;
    EXCEPTION

        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(
                -20442,
                'Branch ' ||
                p_branch_id ||
                ' does not exist.'
            );
    END;

    IF v_branch_status <> 'Active' THEN
        RAISE_APPLICATION_ERROR(
            -20443,
            v_branch_name ||
            ' is currently ' ||
            v_branch_status ||
            ' and is unavailable for item search.'
        );
    END IF;

    -- OUTPUT HEADER
    DBMS_OUTPUT.PUT_LINE(
        RPAD('=', 115, '=')
    );
    DBMS_OUTPUT.PUT_LINE(
        '88 SPEEDMART - ADVANCED ITEM SEARCH'
    );
    DBMS_OUTPUT.PUT_LINE(
        'Branch       : ' ||
        p_branch_id ||
        ' - ' ||
        v_branch_name
    );
    DBMS_OUTPUT.PUT_LINE(
        'Search       : ' ||
        NVL(
            p_keyword,
            'All Items'
        )
    );
    DBMS_OUTPUT.PUT_LINE(
        'MyKasih Only : ' ||
        UPPER(p_mykasih_only)
    );

    DBMS_OUTPUT.PUT_LINE(
        RPAD('=', 115, '=')
    );
    DBMS_OUTPUT.PUT_LINE(
        RPAD('Item ID', 10) ||
        RPAD('Item Name', 42) ||
        RPAD('Category', 30) ||
        LPAD('Price RM', 12) ||
        LPAD('Stock', 10) ||
        LPAD('MK', 6)
    );
    DBMS_OUTPUT.PUT_LINE(
        RPAD('-', 115, '-')
    );

    -- CURSOR
    FOR item_rec IN c_item LOOP
        v_found :=
            v_found + 1;
        DBMS_OUTPUT.PUT_LINE(

            RPAD(
                item_rec.item_id,
                10
            ) ||

            RPAD(
                SUBSTR(
                    item_rec.item_name,
                    1,
                    40
                ),
                42
            ) ||

            RPAD(
                SUBSTR(
                    item_rec.category,
                    1,
                    28
                ),
                30
            ) ||

            LPAD(
                TO_CHAR(
                    item_rec.unit_price,
                    'FM9990.00'
                ),
                12
            ) ||

            LPAD(
                item_rec.quantity,
                10
            ) ||

            LPAD(
                item_rec.is_mykasih_eligible,
                6
            )
        );
    END LOOP;

    -- When NO RESULT
    IF v_found = 0 THEN
        RAISE e_no_items_found;
    END IF;

    DBMS_OUTPUT.PUT_LINE(
        RPAD('-', 115, '-')
    );

    DBMS_OUTPUT.PUT_LINE(
        'Total matching item(s): ' ||
        v_found
    );

    DBMS_OUTPUT.PUT_LINE(
        RPAD('=', 115, '=')
    );

EXCEPTION
    WHEN e_no_items_found THEN
        RAISE_APPLICATION_ERROR(
            -20444,
            'No available items matched the selected branch, keyword and MyKasih option.'
        );

    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE(
            '=== ITEM SEARCH FAILED ==='
        );

        DBMS_OUTPUT.PUT_LINE(
            'Error Code    : ' ||
            SQLCODE
        );

        DBMS_OUTPUT.PUT_LINE(
            'Error Message : ' ||
            SQLERRM
        );
        RAISE;
END prc_search_items;
/

-- REPORT 1

CREATE OR REPLACE PROCEDURE report_branch_sales(
    p_year IN NUMBER DEFAULT NULL
)
IS
    -- Available year range
    v_min_year NUMBER;
    v_max_year NUMBER;

    -- Report counters
    v_year_count         NUMBER := 0;
    v_branch_count       NUMBER;
    v_branch_month_count NUMBER;

    -- Branch yearly totals
    v_branch_orders      NUMBER;
    v_branch_gross       NUMBER;
    v_branch_discount    NUMBER;
    v_branch_refund      NUMBER;
    v_branch_net         NUMBER;

    -- Best / weakest month for each branch
    v_month_best         DATE;
    v_month_best_net     NUMBER;
    v_month_weak         DATE;
    v_month_weak_net     NUMBER;

    -- Year totals
    v_year_orders        NUMBER;
    v_year_gross         NUMBER;
    v_year_discount      NUMBER;
    v_year_refund        NUMBER;
    v_year_net           NUMBER;

    -- Grand totals for multi-year report
    v_grand_orders       NUMBER := 0;
    v_grand_gross        NUMBER := 0;
    v_grand_discount     NUMBER := 0;
    v_grand_refund       NUMBER := 0;
    v_grand_net          NUMBER := 0;

    -- Strongest / weakest branch-month for each year
    v_year_best_branch_id    Branch.branch_id%TYPE;
    v_year_best_branch_name  Branch.name%TYPE;
    v_year_best_month        DATE;
    v_year_best_net          NUMBER;

    v_year_weak_branch_id    Branch.branch_id%TYPE;
    v_year_weak_branch_name  Branch.name%TYPE;
    v_year_weak_month        DATE;
    v_year_weak_net          NUMBER;

    -- Custom exception
    e_no_report_data EXCEPTION;

    -- LEVEL 1: YEAR
    CURSOR c_year (
        cp_year NUMBER
    ) IS
        SELECT DISTINCT sales_year
        FROM v_branch_monthly_revenue
        WHERE cp_year IS NULL
           OR sales_year = cp_year
        ORDER BY sales_year DESC;

    -- LEVEL 2: BRANCH
    CURSOR c_branch (
        cp_year NUMBER
    ) IS
        SELECT DISTINCT
            branch_id,
            branch_name
        FROM v_branch_monthly_revenue
        WHERE sales_year = cp_year
        ORDER BY branch_id;

    -- LEVEL 3: MONTH
    CURSOR c_month (
        cp_year      NUMBER,
        cp_branch_id Branch.branch_id%TYPE
    ) IS
        SELECT
            sales_month,
            total_orders,
            total_customers,
            gross_sales,
            voucher_discount,
            refund_amount,
            net_sales,
            avg_net_sales_per_order
        FROM v_branch_monthly_revenue
        WHERE sales_year = cp_year
          AND branch_id = cp_branch_id
        ORDER BY sales_month;

BEGIN
    -- Find available branch revenue history
    SELECT
        MIN(sales_year),
        MAX(sales_year)
    INTO
        v_min_year,
        v_max_year
    FROM v_branch_monthly_revenue;

    -- No valid branch revenue data
    IF v_min_year IS NULL THEN
        RAISE e_no_report_data;
    END IF;

    -- Validate input year
    IF p_year IS NOT NULL THEN
        IF p_year <> TRUNC(p_year) THEN
            RAISE_APPLICATION_ERROR(
                -20461,
                'Year must be entered as a whole number, for example 2026.'
            );
        END IF;

        IF p_year < v_min_year OR p_year > v_max_year THEN
            RAISE_APPLICATION_ERROR(
                -20462,
                'Year ' || p_year ||
                ' is outside the available branch sales history (' ||
                v_min_year || ' to ' || v_max_year || ').'
            );
        END IF;
    END IF;

    -- Report title
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 120, '='));

    IF p_year IS NULL THEN
        DBMS_OUTPUT.PUT_LINE('88 SPEEDMART - MULTI-YEAR BRANCH REVENUE TREND REPORT');
    ELSE
        DBMS_OUTPUT.PUT_LINE('88 SPEEDMART - ANNUAL BRANCH REVENUE TREND REPORT FOR ' || p_year);
    END IF;

    DBMS_OUTPUT.PUT_LINE('Generated: ' || TO_CHAR(SYSDATE, 'DD-MON-YYYY HH24:MI'));
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 120, '='));

    -- LEVEL 1 LOOP: YEAR
    FOR y IN c_year(p_year) LOOP

        v_year_count := v_year_count + 1;
        v_branch_count := 0;

        v_year_orders := 0;
        v_year_gross := 0;
        v_year_discount := 0;
        v_year_refund := 0;
        v_year_net := 0;

        v_year_best_branch_id := NULL;
        v_year_best_branch_name := NULL;
        v_year_best_month := NULL;
        v_year_best_net := NULL;

        v_year_weak_branch_id := NULL;
        v_year_weak_branch_name := NULL;
        v_year_weak_month := NULL;
        v_year_weak_net := NULL;

        DBMS_OUTPUT.PUT_LINE(CHR(10));
        DBMS_OUTPUT.PUT_LINE(RPAD('=', 120, '='));
        DBMS_OUTPUT.PUT_LINE('YEAR: ' || y.sales_year);
        DBMS_OUTPUT.PUT_LINE(RPAD('=', 120, '='));

        -- LEVEL 2 LOOP: BRANCH
        FOR b IN c_branch(y.sales_year) LOOP

            v_branch_count := v_branch_count + 1;
            v_branch_month_count := 0;

            v_branch_orders := 0;
            v_branch_gross := 0;
            v_branch_discount := 0;
            v_branch_refund := 0;
            v_branch_net := 0;

            v_month_best := NULL;
            v_month_best_net := NULL;
            v_month_weak := NULL;
            v_month_weak_net := NULL;

            -- Branch heading
            DBMS_OUTPUT.PUT_LINE(CHR(10));
            DBMS_OUTPUT.PUT_LINE('BRANCH: ' || b.branch_id || ' - ' || b.branch_name);
            DBMS_OUTPUT.PUT_LINE(RPAD('-', 120, '-'));

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Month', 13) ||
                LPAD('Orders', 9) ||
                LPAD('Customers', 12) ||
                LPAD('Gross RM', 17) ||
                LPAD('Voucher RM', 16) ||
                LPAD('Refund RM', 16) ||
                LPAD('Net Sales RM', 17) ||
                LPAD('Avg Net Sale', 17)
            );

            DBMS_OUTPUT.PUT_LINE(RPAD('-', 120, '-'));

            -- LEVEL 3 LOOP: MONTH
            FOR m IN c_month(y.sales_year, b.branch_id) LOOP

                v_branch_month_count := v_branch_month_count + 1;

                -- Accumulate branch totals
                v_branch_orders := v_branch_orders + m.total_orders;
                v_branch_gross := v_branch_gross + m.gross_sales;
                v_branch_discount := v_branch_discount + m.voucher_discount;
                v_branch_refund := v_branch_refund + m.refund_amount;
                v_branch_net := v_branch_net + m.net_sales;

                -- Determine best month
                IF v_month_best_net IS NULL
                   OR m.net_sales > v_month_best_net THEN

                    v_month_best_net := m.net_sales;
                    v_month_best := m.sales_month;

                END IF;

                -- Determine weakest month
                IF v_month_weak_net IS NULL
                   OR m.net_sales < v_month_weak_net THEN

                    v_month_weak_net := m.net_sales;
                    v_month_weak := m.sales_month;
                END IF;

                -- Determine strongest branch-month of the year
                IF v_year_best_net IS NULL
                   OR m.net_sales > v_year_best_net THEN

                    v_year_best_net := m.net_sales;
                    v_year_best_month := m.sales_month;
                    v_year_best_branch_id := b.branch_id;
                    v_year_best_branch_name := b.branch_name;
                END IF;

                -- Determine weakest branch-month of the year
                IF v_year_weak_net IS NULL
                   OR m.net_sales < v_year_weak_net THEN

                    v_year_weak_net := m.net_sales;
                    v_year_weak_month := m.sales_month;
                    v_year_weak_branch_id := b.branch_id;
                    v_year_weak_branch_name := b.branch_name;
                END IF;

                -- Monthly branch detail
                DBMS_OUTPUT.PUT_LINE(
                    RPAD(TO_CHAR(m.sales_month, 'MON-YYYY'), 13) ||
                    LPAD(m.total_orders, 9) ||
                    LPAD(m.total_customers, 12) ||
                    LPAD(TO_CHAR(m.gross_sales, 'FM999,999,990.00'), 17) ||
                    LPAD(TO_CHAR(m.voucher_discount, 'FM999,999,990.00'), 16) ||
                    LPAD(TO_CHAR(m.refund_amount, 'FM999,999,990.00'), 16) ||
                    LPAD(TO_CHAR(m.net_sales, 'FM999,999,990.00'), 17) ||
                    LPAD(TO_CHAR(m.avg_net_sales_per_order, 'FM999,990.00'), 17)
                );
            END LOOP;

            -- Branch annual summary
            DBMS_OUTPUT.PUT_LINE(RPAD('-', 120, '-'));
            DBMS_OUTPUT.PUT_LINE('BRANCH YEAR SUMMARY');

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Months with sales', 28) || ': ' || v_branch_month_count
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Total orders', 28) || ': ' || v_branch_orders
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Gross sales', 28) ||
                ': RM ' || TO_CHAR(v_branch_gross, 'FM999,999,990.00')
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Voucher discounts', 28) ||
                ': RM ' || TO_CHAR(v_branch_discount, 'FM999,999,990.00')
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Refunds', 28) ||
                ': RM ' || TO_CHAR(v_branch_refund, 'FM999,999,990.00')
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Net sales', 28) ||
                ': RM ' || TO_CHAR(v_branch_net, 'FM999,999,990.00')
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Average monthly net sales', 28) ||
                ': RM ' ||
                TO_CHAR(
                    v_branch_net / NULLIF(v_branch_month_count, 0),
                    'FM999,999,990.00'
                )
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Best month', 28) ||
                ': ' || TO_CHAR(v_month_best, 'MON-YYYY') ||
                ' (RM ' || TO_CHAR(v_month_best_net, 'FM999,999,990.00') || ')'
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Weakest month', 28) ||
                ': ' || TO_CHAR(v_month_weak, 'MON-YYYY') ||
                ' (RM ' || TO_CHAR(v_month_weak_net, 'FM999,999,990.00') || ')'
            );

            DBMS_OUTPUT.PUT_LINE(RPAD('-', 120, '-'));

            -- Add branch totals into year totals
            v_year_orders := v_year_orders + v_branch_orders;
            v_year_gross := v_year_gross + v_branch_gross;
            v_year_discount := v_year_discount + v_branch_discount;
            v_year_refund := v_year_refund + v_branch_refund;
            v_year_net := v_year_net + v_branch_net;
        END LOOP;

        -- Year summary
        DBMS_OUTPUT.PUT_LINE(CHR(10));
        DBMS_OUTPUT.PUT_LINE(RPAD('=', 120, '='));
        DBMS_OUTPUT.PUT_LINE('YEAR ' || y.sales_year || ' SUMMARY');

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Branches analysed', 28) || ': ' || v_branch_count
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Total orders', 28) || ': ' || v_year_orders
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Gross sales', 28) ||
            ': RM ' || TO_CHAR(v_year_gross, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Voucher discounts', 28) ||
            ': RM ' || TO_CHAR(v_year_discount, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Refunds', 28) ||
            ': RM ' || TO_CHAR(v_year_refund, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Net sales', 28) ||
            ': RM ' || TO_CHAR(v_year_net, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Strongest branch-month', 28) ||
            ': ' || v_year_best_branch_id ||
            ' - ' || v_year_best_branch_name ||
            ' | ' || TO_CHAR(v_year_best_month, 'MON-YYYY') ||
            ' | RM ' || TO_CHAR(v_year_best_net, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Weakest branch-month', 28) ||
            ': ' || v_year_weak_branch_id ||
            ' - ' || v_year_weak_branch_name ||
            ' | ' || TO_CHAR(v_year_weak_month, 'MON-YYYY') ||
            ' | RM ' || TO_CHAR(v_year_weak_net, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(RPAD('=', 120, '='));

        -- Grand totals
        v_grand_orders := v_grand_orders + v_year_orders;
        v_grand_gross := v_grand_gross + v_year_gross;
        v_grand_discount := v_grand_discount + v_year_discount;
        v_grand_refund := v_grand_refund + v_year_refund;
        v_grand_net := v_grand_net + v_year_net;

    END LOOP;

    -- No matching year
    IF v_year_count = 0 THEN
        RAISE e_no_report_data;
    END IF;

    -- Show grand summary only for multi-year report
    IF p_year IS NULL THEN

        DBMS_OUTPUT.PUT_LINE(CHR(10));
        DBMS_OUTPUT.PUT_LINE(RPAD('=', 120, '='));
        DBMS_OUTPUT.PUT_LINE('FINAL MULTI-YEAR REPORT SUMMARY');

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Years analysed', 28) || ': ' || v_year_count
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Total orders', 28) || ': ' || v_grand_orders
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Gross sales', 28) ||
            ': RM ' || TO_CHAR(v_grand_gross, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Voucher discounts', 28) ||
            ': RM ' || TO_CHAR(v_grand_discount, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Refunds', 28) ||
            ': RM ' || TO_CHAR(v_grand_refund, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Net sales', 28) ||
            ': RM ' || TO_CHAR(v_grand_net, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(RPAD('=', 120, '='));

    END IF;

EXCEPTION
    -- Custom exception
    WHEN e_no_report_data THEN
        RAISE_APPLICATION_ERROR(
            -20463,
            'No completed and paid branch revenue data was found for the selected year.'
        );

    -- Unexpected database error
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('=== BRANCH REVENUE REPORT FAILED ===');
        DBMS_OUTPUT.PUT_LINE('Error Code    : ' || SQLCODE);
        DBMS_OUTPUT.PUT_LINE('Error Message : ' || SQLERRM);
        RAISE;

END report_branch_sales;

-- REPORT 2

CREATE OR REPLACE PROCEDURE report_annual_myKasih_category (
    p_year IN NUMBER DEFAULT NULL
)
IS
    -- Available year range
    v_min_year NUMBER;
    v_max_year NUMBER;

    -- Counters
    v_year_count      NUMBER := 0;
    v_category_count  NUMBER;
    v_branch_count    NUMBER;

    -- Category totals
    v_cat_orders      NUMBER;
    v_cat_units       NUMBER;
    v_cat_gross       NUMBER;
    v_cat_discount    NUMBER;
    v_cat_refund      NUMBER;
    v_cat_net         NUMBER;

    -- Year totals
    v_year_orders      NUMBER;
    v_year_customers   NUMBER;
    v_year_units       NUMBER;
    v_year_gross       NUMBER;
    v_year_discount    NUMBER;
    v_year_refund      NUMBER;
    v_year_net         NUMBER;

    -- Strongest / weakest category within each year
    v_best_category      Item.category%TYPE;
    v_best_category_net  NUMBER;
    v_weak_category      Item.category%TYPE;
    v_weak_category_net  NUMBER;

    -- Grand totals across all years
    v_grand_orders      NUMBER := 0;
    v_grand_units       NUMBER := 0;
    v_grand_gross       NUMBER := 0;
    v_grand_discount    NUMBER := 0;
    v_grand_refund      NUMBER := 0;
    v_grand_net         NUMBER := 0;

    -- Strongest / weakest category across all years
    v_all_best_category      Item.category%TYPE;
    v_all_best_category_net  NUMBER;
    v_all_weak_category      Item.category%TYPE;
    v_all_weak_category_net  NUMBER;

    -- Custom exception
    e_no_mykasih_data EXCEPTION;

    -- LEVEL 1: YEAR
    CURSOR c_year (
        cp_year NUMBER
    ) IS
        SELECT DISTINCT EXTRACT(YEAR FROM sales_month) AS sales_year
        FROM v_mykasih_branch_category
        WHERE cp_year IS NULL
           OR EXTRACT(YEAR FROM sales_month) = cp_year
        ORDER BY sales_year DESC;

    -- LEVEL 2: CATEGORY
    CURSOR c_category (
        cp_year NUMBER
    ) IS
        SELECT DISTINCT category
        FROM v_mykasih_branch_category
        WHERE EXTRACT(YEAR FROM sales_month) = cp_year
        ORDER BY category;

    -- LEVEL 3: BRANCH
    CURSOR c_branch (
        cp_year     NUMBER,
        cp_category Item.category%TYPE
    ) IS
        SELECT
            branch_id,
            branch_name,
            SUM(mykasih_orders) AS category_orders,
            SUM(total_units) AS total_units,
            ROUND(SUM(gross_mykasih_sales), 2) AS gross_sales,
            ROUND(SUM(voucher_discount), 2) AS voucher_discount,
            ROUND(SUM(refund_amount), 2) AS refund_amount,
            ROUND(SUM(net_mykasih_sales), 2) AS net_sales,
            COUNT(DISTINCT sales_month) AS months_active
        FROM v_mykasih_branch_category
        WHERE EXTRACT(YEAR FROM sales_month) = cp_year
          AND category = cp_category
        GROUP BY branch_id, branch_name
        ORDER BY branch_id;

    -- Category totals for one year
    CURSOR c_year_category (
        cp_year NUMBER
    ) IS
        SELECT
            category,
            SUM(total_units) AS total_units,
            ROUND(SUM(gross_mykasih_sales), 2) AS gross_sales,
            ROUND(SUM(voucher_discount), 2) AS voucher_discount,
            ROUND(SUM(refund_amount), 2) AS refund_amount,
            ROUND(SUM(net_mykasih_sales), 2) AS net_sales
        FROM v_mykasih_branch_category
        WHERE EXTRACT(YEAR FROM sales_month) = cp_year
        GROUP BY category
        ORDER BY net_sales DESC;

    -- Category totals across all years
    CURSOR c_all_category IS
        SELECT
            category,
            SUM(total_units) AS total_units,
            ROUND(SUM(gross_mykasih_sales), 2) AS gross_sales,
            ROUND(SUM(voucher_discount), 2) AS voucher_discount,
            ROUND(SUM(refund_amount), 2) AS refund_amount,
            ROUND(SUM(net_mykasih_sales), 2) AS net_sales
        FROM v_mykasih_branch_category
        GROUP BY category
        ORDER BY net_sales DESC;

BEGIN
    -- Determine available MyKasih year range
    SELECT
        MIN(EXTRACT(YEAR FROM sales_month)),
        MAX(EXTRACT(YEAR FROM sales_month))
    INTO
        v_min_year,
        v_max_year
    FROM v_mykasih_branch_category;

    -- No MyKasih data
    IF v_min_year IS NULL THEN
        RAISE e_no_mykasih_data;
    END IF;

    -- Validate year parameter
    IF p_year IS NOT NULL THEN

        IF p_year <> TRUNC(p_year) THEN
            RAISE_APPLICATION_ERROR(
                -20471,
                'Year must be entered as a whole number, for example 2026.'
            );
        END IF;

        IF p_year < v_min_year OR p_year > v_max_year THEN
            RAISE_APPLICATION_ERROR(
                -20472,
                'Year ' || p_year ||
                ' is outside the available MyKasih transaction history (' ||
                v_min_year || ' to ' || v_max_year || ').'
            );
        END IF;

    END IF;

    -- Report title
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 140, '='));

    IF p_year IS NULL THEN
        DBMS_OUTPUT.PUT_LINE('88 SPEEDMART - MULTI-YEAR MYKASIH CATEGORY CONTRIBUTION REPORT');
    ELSE
        DBMS_OUTPUT.PUT_LINE('88 SPEEDMART - ANNUAL MYKASIH CATEGORY CONTRIBUTION REPORT FOR ' || p_year);
    END IF;

    DBMS_OUTPUT.PUT_LINE('Generated: ' || TO_CHAR(SYSDATE, 'DD-MON-YYYY HH24:MI'));
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 140, '='));

    -- LEVEL 1 LOOP: YEAR
    FOR y IN c_year(p_year) LOOP

        v_year_count := v_year_count + 1;
        v_category_count := 0;

        v_year_units := 0;
        v_year_gross := 0;
        v_year_discount := 0;
        v_year_refund := 0;
        v_year_net := 0;

        v_best_category := NULL;
        v_best_category_net := NULL;
        v_weak_category := NULL;
        v_weak_category_net := NULL;

        -- Unique MyKasih orders and customers for year
        SELECT
            COUNT(DISTINCT O.order_no),
            COUNT(DISTINCT O.customer_id)
        INTO
            v_year_orders,
            v_year_customers
        FROM Orders O
        JOIN Payment P
            ON O.order_no = P.order_no
        JOIN Customer C
            ON O.customer_id = C.customer_id
        WHERE EXTRACT(YEAR FROM CAST(O.order_date_time AS DATE)) = y.sales_year
          AND O.order_status = 'Completed'
          AND P.payment_status IN ('Paid', 'Refunded')
          AND O.is_mykasih_purchase = 'Y'
          AND P.payment_method = 'MyKasih'
          AND C.ic_number IS NOT NULL;

        -- Year header
        DBMS_OUTPUT.PUT_LINE(CHR(10));
        DBMS_OUTPUT.PUT_LINE(RPAD('=', 140, '='));
        DBMS_OUTPUT.PUT_LINE('YEAR: ' || y.sales_year);
        DBMS_OUTPUT.PUT_LINE('Unique MyKasih Orders    : ' || v_year_orders);
        DBMS_OUTPUT.PUT_LINE('Unique MyKasih Customers : ' || v_year_customers);
        DBMS_OUTPUT.PUT_LINE(RPAD('=', 140, '='));

        -- LEVEL 2 LOOP: CATEGORY
        FOR c IN c_category(y.sales_year) LOOP

            v_category_count := v_category_count + 1;
            v_branch_count := 0;

            v_cat_orders := 0;
            v_cat_units := 0;
            v_cat_gross := 0;
            v_cat_discount := 0;
            v_cat_refund := 0;
            v_cat_net := 0;

            -- Category header
            DBMS_OUTPUT.PUT_LINE(CHR(10));
            DBMS_OUTPUT.PUT_LINE('CATEGORY: ' || c.category);
            DBMS_OUTPUT.PUT_LINE(RPAD('-', 140, '-'));

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Branch ID', 12) ||
                RPAD('Branch', 32) ||
                LPAD('Months', 9) ||
                LPAD('Orders', 9) ||
                LPAD('Units', 9) ||
                LPAD('Gross RM', 17) ||
                LPAD('Voucher RM', 17) ||
                LPAD('Refund RM', 17) ||
                LPAD('Net RM', 17)
            );

            DBMS_OUTPUT.PUT_LINE(RPAD('-', 140, '-'));

            -- LEVEL 3 LOOP: BRANCH
            FOR b IN c_branch(y.sales_year, c.category) LOOP

                v_branch_count := v_branch_count + 1;

                v_cat_orders := v_cat_orders + b.category_orders;
                v_cat_units := v_cat_units + b.total_units;
                v_cat_gross := v_cat_gross + b.gross_sales;
                v_cat_discount := v_cat_discount + b.voucher_discount;
                v_cat_refund := v_cat_refund + b.refund_amount;
                v_cat_net := v_cat_net + b.net_sales;

                -- Branch contribution detail
                DBMS_OUTPUT.PUT_LINE(
                    RPAD(b.branch_id, 12) ||
                    RPAD(SUBSTR(b.branch_name, 1, 30), 32) ||
                    LPAD(b.months_active, 9) ||
                    LPAD(b.category_orders, 9) ||
                    LPAD(b.total_units, 9) ||
                    LPAD(TO_CHAR(b.gross_sales, 'FM999,999,990.00'), 17) ||
                    LPAD(TO_CHAR(b.voucher_discount, 'FM999,999,990.00'), 17) ||
                    LPAD(TO_CHAR(b.refund_amount, 'FM999,999,990.00'), 17) ||
                    LPAD(TO_CHAR(b.net_sales, 'FM999,999,990.00'), 17)
                );

            END LOOP;

            -- Category year summary
            DBMS_OUTPUT.PUT_LINE(RPAD('-', 140, '-'));
            DBMS_OUTPUT.PUT_LINE('CATEGORY YEAR SUMMARY');

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Branches contributing', 30) || ': ' || v_branch_count
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Orders containing category', 30) || ': ' || v_cat_orders
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Total units', 30) || ': ' || v_cat_units
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Gross MyKasih sales', 30) ||
                ': RM ' || TO_CHAR(v_cat_gross, 'FM999,999,990.00')
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Voucher discounts', 30) ||
                ': RM ' || TO_CHAR(v_cat_discount, 'FM999,999,990.00')
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Refunds', 30) ||
                ': RM ' || TO_CHAR(v_cat_refund, 'FM999,999,990.00')
            );

            DBMS_OUTPUT.PUT_LINE(
                RPAD('Net MyKasih sales', 30) ||
                ': RM ' || TO_CHAR(v_cat_net, 'FM999,999,990.00')
            );

            DBMS_OUTPUT.PUT_LINE(RPAD('-', 140, '-'));

            -- Determine strongest category of year
            IF v_best_category_net IS NULL
               OR v_cat_net > v_best_category_net THEN

                v_best_category := c.category;
                v_best_category_net := v_cat_net;

            END IF;

            -- Determine weakest category of year
            IF v_weak_category_net IS NULL
               OR v_cat_net < v_weak_category_net THEN

                v_weak_category := c.category;
                v_weak_category_net := v_cat_net;

            END IF;

            -- Add category totals into year totals
            v_year_units := v_year_units + v_cat_units;
            v_year_gross := v_year_gross + v_cat_gross;
            v_year_discount := v_year_discount + v_cat_discount;
            v_year_refund := v_year_refund + v_cat_refund;
            v_year_net := v_year_net + v_cat_net;

        END LOOP;

        -- Year MyKasih sales by product category
        DBMS_OUTPUT.PUT_LINE(CHR(10));
        DBMS_OUTPUT.PUT_LINE('YEAR ' || y.sales_year || ' MYKASIH SALES BY PRODUCT CATEGORY');
        DBMS_OUTPUT.PUT_LINE(RPAD('-', 140, '-'));

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Category', 35) ||
            LPAD('Units', 12) ||
            LPAD('Gross RM', 20) ||
            LPAD('Voucher RM', 20) ||
            LPAD('Refund RM', 20) ||
            LPAD('Net RM', 20)
        );

        DBMS_OUTPUT.PUT_LINE(RPAD('-', 140, '-'));

        FOR yc IN c_year_category(y.sales_year) LOOP

            DBMS_OUTPUT.PUT_LINE(
                RPAD(SUBSTR(yc.category, 1, 33), 35) ||
                LPAD(yc.total_units, 12) ||
                LPAD(TO_CHAR(yc.gross_sales, 'FM999,999,990.00'), 20) ||
                LPAD(TO_CHAR(yc.voucher_discount, 'FM999,999,990.00'), 20) ||
                LPAD(TO_CHAR(yc.refund_amount, 'FM999,999,990.00'), 20) ||
                LPAD(TO_CHAR(yc.net_sales, 'FM999,999,990.00'), 20)
            );

        END LOOP;

        DBMS_OUTPUT.PUT_LINE(RPAD('-', 140, '-'));

        -- Year MyKasih summary
        DBMS_OUTPUT.PUT_LINE(CHR(10));
        DBMS_OUTPUT.PUT_LINE(RPAD('=', 140, '='));
        DBMS_OUTPUT.PUT_LINE('YEAR ' || y.sales_year || ' MYKASIH SUMMARY');

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Categories analysed', 30) || ': ' || v_category_count
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Unique MyKasih orders', 30) || ': ' || v_year_orders
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Unique MyKasih customers', 30) || ': ' || v_year_customers
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Total units', 30) || ': ' || v_year_units
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Gross MyKasih sales', 30) ||
            ': RM ' || TO_CHAR(v_year_gross, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Voucher discounts', 30) ||
            ': RM ' || TO_CHAR(v_year_discount, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Refunds', 30) ||
            ': RM ' || TO_CHAR(v_year_refund, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Net MyKasih sales', 30) ||
            ': RM ' || TO_CHAR(v_year_net, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Strongest category', 30) ||
            ': ' || v_best_category ||
            ' | RM ' || TO_CHAR(v_best_category_net, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Weakest category', 30) ||
            ': ' || v_weak_category ||
            ' | RM ' || TO_CHAR(v_weak_category_net, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(RPAD('=', 140, '='));

        -- Add year into all-year totals
        v_grand_orders := v_grand_orders + v_year_orders;
        v_grand_units := v_grand_units + v_year_units;
        v_grand_gross := v_grand_gross + v_year_gross;
        v_grand_discount := v_grand_discount + v_year_discount;
        v_grand_refund := v_grand_refund + v_year_refund;
        v_grand_net := v_grand_net + v_year_net;

    END LOOP;

    -- No matching year
    IF v_year_count = 0 THEN
        RAISE e_no_mykasih_data;
    END IF;

    -- FINAL MULTI-YEAR SECTION
    IF p_year IS NULL THEN

        v_all_best_category := NULL;
        v_all_best_category_net := NULL;

        v_all_weak_category := NULL;
        v_all_weak_category_net := NULL;

        -- Determine strongest / weakest category across all years
        FOR ac IN c_all_category LOOP

            IF v_all_best_category_net IS NULL
               OR ac.net_sales > v_all_best_category_net THEN

                v_all_best_category := ac.category;
                v_all_best_category_net := ac.net_sales;

            END IF;

            IF v_all_weak_category_net IS NULL
               OR ac.net_sales < v_all_weak_category_net THEN

                v_all_weak_category := ac.category;
                v_all_weak_category_net := ac.net_sales;

            END IF;

        END LOOP;

        -- ALL-YEAR MYKASIH SALES BY PRODUCT CATEGORY
        DBMS_OUTPUT.PUT_LINE(CHR(10));

        DBMS_OUTPUT.PUT_LINE(
            'ALL-YEAR MYKASIH SALES BY PRODUCT CATEGORY'
        );

        DBMS_OUTPUT.PUT_LINE(RPAD('-', 140, '-'));

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Category', 35) ||
            LPAD('Units', 12) ||
            LPAD('Gross RM', 20) ||
            LPAD('Voucher RM', 20) ||
            LPAD('Refund RM', 20) ||
            LPAD('Net RM', 20)
        );

        DBMS_OUTPUT.PUT_LINE(RPAD('-', 140, '-'));

        FOR ac IN c_all_category LOOP

            DBMS_OUTPUT.PUT_LINE(
                RPAD(SUBSTR(ac.category, 1, 33), 35) ||
                LPAD(ac.total_units, 12) ||
                LPAD(TO_CHAR(ac.gross_sales, 'FM999,999,990.00'), 20) ||
                LPAD(TO_CHAR(ac.voucher_discount, 'FM999,999,990.00'), 20) ||
                LPAD(TO_CHAR(ac.refund_amount, 'FM999,999,990.00'), 20) ||
                LPAD(TO_CHAR(ac.net_sales, 'FM999,999,990.00'), 20)
            );

        END LOOP;

        DBMS_OUTPUT.PUT_LINE(RPAD('-', 140, '-'));

        -- Final multi-year summary
        DBMS_OUTPUT.PUT_LINE(CHR(10));
        DBMS_OUTPUT.PUT_LINE(RPAD('=', 140, '='));
        DBMS_OUTPUT.PUT_LINE('FINAL MULTI-YEAR MYKASIH SUMMARY');

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Years analysed', 30) || ': ' || v_year_count
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Total MyKasih orders', 30) || ': ' || v_grand_orders
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Total units', 30) || ': ' || v_grand_units
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Gross MyKasih sales', 30) ||
            ': RM ' || TO_CHAR(v_grand_gross, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Voucher discounts', 30) ||
            ': RM ' || TO_CHAR(v_grand_discount, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Refunds', 30) ||
            ': RM ' || TO_CHAR(v_grand_refund, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Net MyKasih sales', 30) ||
            ': RM ' || TO_CHAR(v_grand_net, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Strongest category', 30) ||
            ': ' || v_all_best_category ||
            ' | RM ' || TO_CHAR(v_all_best_category_net, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Weakest category', 30) ||
            ': ' || v_all_weak_category ||
            ' | RM ' || TO_CHAR(v_all_weak_category_net, 'FM999,999,990.00')
        );

        DBMS_OUTPUT.PUT_LINE(RPAD('=', 140, '='));

    END IF;

EXCEPTION

    -- No valid MyKasih report data
    WHEN e_no_mykasih_data THEN
        RAISE_APPLICATION_ERROR(
            -20061,
            'No completed and paid or refunded MyKasih transaction data was found for the selected year.'
        );

    -- Unexpected database error
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('=== MYKASIH REPORT FAILED ===');
        DBMS_OUTPUT.PUT_LINE('Error Code    : ' || SQLCODE);
        DBMS_OUTPUT.PUT_LINE('Error Message : ' || SQLERRM);
        RAISE;

END report_annual_myKasih_category;
