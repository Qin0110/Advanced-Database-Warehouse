    EXECUTE IMMEDIATE 'DROP TABLE ETL_REJECT_LOG CASCADE CONSTRAINTS';
EXCEPTION WHEN OTHERS THEN
    IF SQLCODE != -942 THEN RAISE; END IF;
END;

BEGIN
    EXECUTE IMMEDIATE 'DROP SEQUENCE ETL_REJECT_SEQ';
EXCEPTION WHEN OTHERS THEN
    IF SQLCODE != -2289 THEN RAISE; END IF;
END;
/

CREATE TABLE ETL_REJECT_LOG (
    REJECT_ID      NUMBER PRIMARY KEY,
    BATCH_FROM     DATE NOT NULL,
    BATCH_TO       DATE NOT NULL,
    SOURCE_TABLE   VARCHAR2(30) NOT NULL,
    SOURCE_KEY     VARCHAR2(10),
    REASON         VARCHAR2(150) NOT NULL,
    REJECTED_AT    TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL
);

CREATE SEQUENCE ETL_REJECT_SEQ START WITH 1 INCREMENT BY 1;

CREATE OR REPLACE VIEW VW_STG_BRANCH_CLEAN AS
SELECT
    UPPER(TRIM(b.branch_id)) AS branch_id,
    REGEXP_REPLACE(TRIM(b.name), '[[:space:]]+', ' ') AS branchname,
    INITCAP(
        REGEXP_SUBSTR(
            REGEXP_REPLACE(TRIM(b.name), '[[:space:]]+', ' '),
            'speedmart[[:space:]]+(.+)$', 1, 1, 'i', 1
        )
    ) AS brancharea,
    TRIM(
        REGEXP_REPLACE(
            REGEXP_SUBSTR(TRIM(b.address), '[^,]+$'),
            '^[[:space:]]*[0-9]{5}[[:space:]]+', ''
        )
    ) AS branchstate,
    INITCAP(TRIM(b.status)) AS status,
    TRIM(b.open_time) AS branchopentime,
    TRIM(b.close_time) AS branchclosetime,
    CASE
        WHEN TRIM(b.branch_id) IS NULL THEN 'N'
        WHEN NOT REGEXP_LIKE(UPPER(TRIM(b.branch_id)), '^BR[0-9]{3}$') THEN 'N'
        WHEN TRIM(b.name) IS NULL THEN 'N'
        WHEN TRIM(b.status) IS NULL THEN 'N'
        WHEN INITCAP(TRIM(b.status)) NOT IN ('Active', 'Inactive', 'Closed') THEN 'N'
        WHEN TRIM(b.open_time) IS NULL THEN 'N'
        WHEN TRIM(b.close_time) IS NULL THEN 'N'
        WHEN NOT REGEXP_LIKE(TRIM(b.open_time),
             '^(0[0-9]|1[0-9]|2[0-3]):[0-5][0-9]$') THEN 'N'
        WHEN NOT REGEXP_LIKE(TRIM(b.close_time),
             '^(0[0-9]|1[0-9]|2[0-3]):[0-5][0-9]$') THEN 'N'
        ELSE 'Y'
    END AS valid_flag,
    CASE
        WHEN TRIM(b.branch_id) IS NULL THEN 'Branch ID is missing'
        WHEN NOT REGEXP_LIKE(UPPER(TRIM(b.branch_id)), '^BR[0-9]{3}$')
             THEN 'Invalid Branch ID format; expected BR followed by 3 digits, e.g. BR001'
        WHEN TRIM(b.name) IS NULL THEN 'Branch name is empty after trimming'
        WHEN TRIM(b.status) IS NULL THEN 'Branch status is missing'
        WHEN INITCAP(TRIM(b.status)) NOT IN ('Active', 'Inactive', 'Closed')
             THEN 'Invalid branch status; expected Active, Inactive or Closed'
        WHEN TRIM(b.open_time) IS NULL THEN 'Branch open time is missing'
        WHEN TRIM(b.close_time) IS NULL THEN 'Branch close time is missing'
        WHEN NOT REGEXP_LIKE(TRIM(b.open_time),
             '^(0[0-9]|1[0-9]|2[0-3]):[0-5][0-9]$') THEN 'Invalid branch open time; expected HH24:MI'
        WHEN NOT REGEXP_LIKE(TRIM(b.close_time),
             '^(0[0-9]|1[0-9]|2[0-3]):[0-5][0-9]$') THEN 'Invalid branch close time; expected HH24:MI'
    END AS reject_reason
FROM WAREHOUSEDATA.Branch b;

CREATE OR REPLACE VIEW VW_STG_COURIER_CLEAN AS
SELECT
    UPPER(TRIM(c.courier_id)) AS courier_id,
    REGEXP_REPLACE(
        TRIM(c.name),
        '[[:space:]]+',
        ' '
    ) AS couriername,
    UPPER(TRIM(c.delivery_company_id)) AS delivery_company_id,
    REGEXP_REPLACE(
        TRIM(dc.company_name),
        '[[:space:]]+',
        ' '
    ) AS deliverycompanyname,
    CASE
        WHEN TRIM(c.courier_id) IS NULL THEN 'N'
        WHEN NOT REGEXP_LIKE(
            UPPER(TRIM(c.courier_id)),
            '^CR[0-9]{4}$'
        ) THEN 'N'
        WHEN TRIM(c.name) IS NULL THEN 'N'
        WHEN TRIM(c.delivery_company_id) IS NULL THEN 'N'
        WHEN NOT REGEXP_LIKE(
            UPPER(TRIM(c.delivery_company_id)),
            '^DC[0-9]{3}$'
        ) THEN 'N'
        WHEN TRIM(dc.company_name) IS NULL THEN 'N'
        ELSE 'Y'
    END AS valid_flag,
    CASE
        WHEN TRIM(c.courier_id) IS NULL
            THEN 'Courier ID is missing'
        WHEN NOT REGEXP_LIKE(
            UPPER(TRIM(c.courier_id)),
            '^CR[0-9]{4}$'
        )
            THEN 'Invalid Courier ID format; expected CR followed by 4 digits, e.g. CR0002'
        WHEN TRIM(c.name) IS NULL
            THEN 'Courier name is empty after trimming'
        WHEN TRIM(c.delivery_company_id) IS NULL
            THEN 'Delivery company ID is missing'
        WHEN NOT REGEXP_LIKE(
            UPPER(TRIM(c.delivery_company_id)),
            '^DC[0-9]{3}$'
        )
            THEN 'Invalid Delivery Company ID format; expected DC followed by 3 digits, e.g. DC001'
        WHEN TRIM(dc.company_name) IS NULL
            THEN 'Delivery company name is empty after trimming'
    END AS reject_reason
FROM WAREHOUSEDATA.Courier c
JOIN WAREHOUSEDATA.Delivery_Company dc
    ON dc.delivery_company_id = c.delivery_company_id;

CREATE OR REPLACE VIEW VW_STG_PRODUCT_CLEAN AS
SELECT
    UPPER(TRIM(i.item_id)) AS item_id,
    UPPER(TRIM(i.supplier_id)) AS supplier_id,
    REGEXP_REPLACE(TRIM(s.company_name), '[[:space:]]+', ' ') AS suppliername,
    REGEXP_REPLACE(TRIM(i.name), '[[:space:]]+', ' ') AS itemname,
    i.unit_price,
    TRIM(i.unit_of_measure) AS unitofmeasure,
    i.unit_weight_kg AS unitweight,
    UPPER(TRIM(i.is_mykasih_eligible)) AS is_mykasih_eligible,
    INITCAP(TRIM(i.status)) AS productstatus,
    CASE
        WHEN TRIM(i.item_id) IS NULL THEN 'N'
        WHEN NOT REGEXP_LIKE(UPPER(TRIM(i.item_id)), '^IT[0-9]{3}$') THEN 'N'
        WHEN TRIM(i.supplier_id) IS NULL THEN 'N'
        WHEN NOT REGEXP_LIKE(UPPER(TRIM(i.supplier_id)), '^SUP[0-9]{3}$') THEN 'N'
        WHEN TRIM(i.name) IS NULL THEN 'N'
        WHEN TRIM(s.company_name) IS NULL THEN 'N'
        WHEN i.unit_price IS NULL THEN 'N'
        WHEN i.unit_price < 0 THEN 'N'
        WHEN TRIM(i.unit_of_measure) IS NULL THEN 'N'
        WHEN TRIM(i.unit_of_measure) NOT IN ('kg', 'g', 'L', 'ml', 'pcs', 'pack') THEN 'N'
        WHEN i.unit_weight_kg IS NULL THEN 'N'
        WHEN i.unit_weight_kg < 0 THEN 'N'
        WHEN TRIM(i.is_mykasih_eligible) IS NULL THEN 'N'
        WHEN UPPER(TRIM(i.is_mykasih_eligible)) NOT IN ('Y', 'N') THEN 'N'
        WHEN TRIM(i.status) IS NULL THEN 'N'
        WHEN INITCAP(TRIM(i.status)) NOT IN ('Active', 'Inactive') THEN 'N'
        ELSE 'Y'
    END AS valid_flag,
    CASE
        WHEN TRIM(i.item_id) IS NULL THEN 'Item ID is missing'
        WHEN NOT REGEXP_LIKE(UPPER(TRIM(i.item_id)), '^IT[0-9]{3}$')
             THEN 'Invalid Item ID format; expected IT followed by 3 digits, e.g. IT001'
        WHEN TRIM(i.supplier_id) IS NULL THEN 'Supplier ID is missing'
        WHEN NOT REGEXP_LIKE(UPPER(TRIM(i.supplier_id)), '^SUP[0-9]{3}$')
             THEN 'Invalid Supplier ID format; expected SUP followed by 3 digits, e.g. SUP001'
        WHEN TRIM(i.name) IS NULL THEN 'Product name is empty after trimming'
        WHEN TRIM(s.company_name) IS NULL THEN 'Supplier name is empty after trimming'
        WHEN i.unit_price IS NULL THEN 'Unit price is missing'
        WHEN i.unit_price < 0 THEN 'Unit price must be a non-negative number'
        WHEN TRIM(i.unit_of_measure) IS NULL THEN 'Unit of measure is missing'
        WHEN TRIM(i.unit_of_measure) NOT IN ('kg', 'g', 'L', 'ml', 'pcs', 'pack')
             THEN 'Invalid unit of measure'
        WHEN i.unit_weight_kg IS NULL THEN 'Unit weight is missing'
        WHEN i.unit_weight_kg < 0 THEN 'Unit weight must be a non-negative number'
        WHEN TRIM(i.is_mykasih_eligible) IS NULL THEN 'MyKasih eligibility is missing'
        WHEN UPPER(TRIM(i.is_mykasih_eligible)) NOT IN ('Y', 'N')
             THEN 'Invalid MyKasih eligibility; expected Y or N'
        WHEN TRIM(i.status) IS NULL THEN 'Product status is missing'
        WHEN INITCAP(TRIM(i.status)) NOT IN ('Active', 'Inactive')
             THEN 'Invalid product status; expected Active or Inactive'
    END AS reject_reason
FROM WAREHOUSEDATA.Item i
JOIN WAREHOUSEDATA.Supplier s
  ON s.supplier_id = i.supplier_id;

/* ============================================================
   3. SCD TYPE 2 DIMENSION LOADERS
   Pattern:
      new natural key           -> INSERT
      same natural key, no diff -> DO NOTHING
      same natural key, changed -> close old N + insert new Y
   ============================================================ */

CREATE OR REPLACE PROCEDURE PRC_LOAD_BRANCH_SCD2 (
    p_effective_date IN DATE,
    p_batch_to       IN DATE
)
IS
    v_key        BRANCH_DIM.BRANCHKEY%TYPE;
    v_name       BRANCH_DIM.BRANCHNAME%TYPE;
    v_area       BRANCH_DIM.BRANCHAREA%TYPE;
    v_state      BRANCH_DIM.BRANCHSTATE%TYPE;
    v_status     BRANCH_DIM.STATUS%TYPE;
    v_open       BRANCH_DIM.BRANCHOPENTIME%TYPE;
    v_close      BRANCH_DIM.BRANCHCLOSETIME%TYPE;
BEGIN
    FOR r IN (SELECT * FROM VW_STG_BRANCH_CLEAN) LOOP
        IF r.valid_flag = 'N' THEN
            INSERT INTO ETL_REJECT_LOG
            VALUES (ETL_REJECT_SEQ.NEXTVAL, p_effective_date, p_batch_to,
                    'BRANCH', r.branch_id, r.reject_reason, SYSTIMESTAMP);
            CONTINUE;
        END IF;

        BEGIN
            SELECT branchkey, branchname, brancharea, branchstate,
                   status, branchopentime, branchclosetime
            INTO   v_key, v_name, v_area, v_state,
                   v_status, v_open, v_close
            FROM   BRANCH_DIM
            WHERE  branchid = r.branch_id
            AND    flag = 'Y';

            IF NVL(UPPER(TRIM(v_name)), '#')  <> NVL(UPPER(TRIM(r.branchname)), '#')
            OR NVL(UPPER(TRIM(v_area)), '#')  <> NVL(UPPER(TRIM(r.brancharea)), '#')
            OR NVL(UPPER(TRIM(v_state)), '#') <> NVL(UPPER(TRIM(r.branchstate)), '#')
            OR NVL(UPPER(TRIM(v_status)), '#')<> NVL(UPPER(TRIM(r.status)), '#')
            OR NVL(TRIM(v_open), '#')         <> NVL(TRIM(r.branchopentime), '#')
            OR NVL(TRIM(v_close), '#')        <> NVL(TRIM(r.branchclosetime), '#') THEN

                UPDATE BRANCH_DIM
                SET effectiveenddate = p_effective_date - 1,
                    flag = 'N'
                WHERE branchkey = v_key;

                INSERT INTO BRANCH_DIM
                VALUES (BRANCH_DIM_SEQ.NEXTVAL, r.branch_id, r.branchname,
                        r.brancharea, r.branchstate, r.status,
                        r.branchopentime, r.branchclosetime,
                        p_effective_date, DATE '9999-12-31', 'Y');
            END IF;

        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                INSERT INTO BRANCH_DIM
                VALUES (BRANCH_DIM_SEQ.NEXTVAL, r.branch_id, r.branchname,
                        r.brancharea, r.branchstate, r.status,
                        r.branchopentime, r.branchclosetime,
                        p_effective_date, DATE '9999-12-31', 'Y');
            WHEN TOO_MANY_ROWS THEN
                INSERT INTO ETL_REJECT_LOG
                VALUES (ETL_REJECT_SEQ.NEXTVAL, p_effective_date, p_batch_to,
                        'BRANCH', r.branch_id,
                        'More than one current BRANCH_DIM row has FLAG=Y', SYSTIMESTAMP);
        END;
    END LOOP;
END;
/

CREATE OR REPLACE PROCEDURE PRC_LOAD_COURIER_SCD2 (
    p_effective_date IN DATE,
    p_batch_to       IN DATE
)
IS
    v_key       COURIER_DIM.COURIERKEY%TYPE;
    v_name      COURIER_DIM.COURIERNAME%TYPE;
    v_companyid COURIER_DIM.DELIVERYCOMPANYID%TYPE;
    v_company   COURIER_DIM.DELIVERYCOMPANYNAME%TYPE;
BEGIN
    FOR r IN (SELECT * FROM VW_STG_COURIER_CLEAN) LOOP
        IF r.valid_flag = 'N' THEN
            INSERT INTO ETL_REJECT_LOG

                    'COURIER', r.courier_id, r.reject_reason, SYSTIMESTAMP);
            CONTINUE;
        END IF;

        BEGIN
            SELECT courierkey, couriername, deliverycompanyid, deliverycompanyname
            INTO   v_key, v_name, v_companyid, v_company
            FROM   COURIER_DIM
            WHERE  courierid = r.courier_id
            AND    flag = 'Y';

            IF NVL(UPPER(TRIM(v_name)), '#') <> NVL(UPPER(TRIM(r.couriername)), '#')
            OR NVL(v_companyid, '#')         <> NVL(r.delivery_company_id, '#')
            OR NVL(UPPER(TRIM(v_company)), '#') <> NVL(UPPER(TRIM(r.deliverycompanyname)), '#') THEN

                UPDATE COURIER_DIM
                SET effectiveenddate = p_effective_date - 1,
                    flag = 'N'
                WHERE courierkey = v_key;

                INSERT INTO COURIER_DIM
                VALUES (COURIER_DIM_SEQ.NEXTVAL, r.courier_id, r.couriername,
                        r.delivery_company_id, r.deliverycompanyname,
                        p_effective_date, DATE '9999-12-31', 'Y');
            END IF;

        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                INSERT INTO COURIER_DIM
                VALUES (COURIER_DIM_SEQ.NEXTVAL, r.courier_id, r.couriername,
                        r.delivery_company_id, r.deliverycompanyname,
                        p_effective_date, DATE '9999-12-31', 'Y');
        END;
    END LOOP;
END;

CREATE OR REPLACE PROCEDURE PRC_LOAD_PRODUCT_SCD2 (
    p_effective_date IN DATE,
    p_batch_to       IN DATE
)
IS
    v_key          PRODUCT_DIM.PRODUCTKEY%TYPE;
    v_supplierid   PRODUCT_DIM.SUPPLIERID%TYPE;
    v_suppliername PRODUCT_DIM.SUPPLIERNAME%TYPE;
    v_itemname     PRODUCT_DIM.ITEMNAME%TYPE;
    v_price        PRODUCT_DIM.UNITPRICE%TYPE;
    v_uom          PRODUCT_DIM.UNITOFMEASURE%TYPE;
    v_weight       PRODUCT_DIM.UNITWEIGHT%TYPE;
    v_mykasih      PRODUCT_DIM.IS_MYKASIH_ELIGIBLE%TYPE;
    v_status       PRODUCT_DIM.PRODUCTSTATUS%TYPE;
BEGIN
    FOR r IN (SELECT * FROM VW_STG_PRODUCT_CLEAN) LOOP
        IF r.valid_flag = 'N' THEN
            INSERT INTO ETL_REJECT_LOG
            VALUES (ETL_REJECT_SEQ.NEXTVAL, p_effective_date, p_batch_to,
                    'ITEM', r.item_id, r.reject_reason, SYSTIMESTAMP);
            CONTINUE;
        END IF;

        BEGIN
            SELECT productkey, supplierid, suppliername, itemname, unitprice,
                   unitofmeasure, unitweight, is_mykasih_eligible, productstatus
            INTO   v_key, v_supplierid, v_suppliername, v_itemname, v_price,
                   v_uom, v_weight, v_mykasih, v_status
            FROM   PRODUCT_DIM
            WHERE  itemid = r.item_id
            AND    flag = 'Y';

            IF NVL(v_supplierid, '#') <> NVL(r.supplier_id, '#')
            OR NVL(UPPER(TRIM(v_suppliername)), '#') <> NVL(UPPER(TRIM(r.suppliername)), '#')
            OR NVL(UPPER(TRIM(v_itemname)), '#') <> NVL(UPPER(TRIM(r.itemname)), '#')
            OR NVL(v_price, -1) <> NVL(r.unit_price, -1)
            OR NVL(v_uom, '#') <> NVL(r.unitofmeasure, '#')
            OR NVL(v_weight, -1) <> NVL(r.unitweight, -1)
            OR NVL(v_mykasih, '#') <> NVL(r.is_mykasih_eligible, '#')
            OR NVL(v_status, '#') <> NVL(r.productstatus, '#') THEN

                UPDATE PRODUCT_DIM
                SET effectiveenddate = p_effective_date - 1,
                    flag = 'N'
                WHERE productkey = v_key;

                INSERT INTO PRODUCT_DIM
                VALUES (PRODUCT_DIM_SEQ.NEXTVAL, r.item_id, r.supplier_id,
                        r.suppliername, r.itemname, r.unit_price,
                        r.unitofmeasure, r.unitweight, r.is_mykasih_eligible,
                        r.productstatus, p_effective_date, DATE '9999-12-31', 'Y');
            END IF;

        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                INSERT INTO PRODUCT_DIM
                VALUES (PRODUCT_DIM_SEQ.NEXTVAL, r.item_id, r.supplier_id,
                        r.suppliername, r.itemname, r.unit_price,
                        r.unitofmeasure, r.unitweight, r.is_mykasih_eligible,
                        r.productstatus, p_effective_date, DATE '9999-12-31', 'Y');
        END;
    END LOOP;
END;

CREATE OR REPLACE PROCEDURE PRC_LOAD_REWARD_SCD2 (
    p_effective_date IN DATE,
    p_batch_to       IN DATE
)
IS
    v_key      REWARD_DIM.REWARD_KEY%TYPE;
    v_name     REWARD_DIM.REWARDNAME%TYPE;
    v_type     REWARD_DIM.REWARDTYPE%TYPE;
    v_points   REWARD_DIM.POINTSREQUIRED%TYPE;
    v_value    REWARD_DIM.REWARDVALUE%TYPE;
    v_status   REWARD_DIM.REWARDSTATUS%TYPE;
BEGIN
    FOR r IN (SELECT * FROM VW_STG_REWARD_CLEAN) LOOP
        IF r.valid_flag = 'N' THEN
            INSERT INTO ETL_REJECT_LOG
            VALUES (ETL_REJECT_SEQ.NEXTVAL, p_effective_date, p_batch_to,
                    'REWARD', r.reward_id, r.reject_reason, SYSTIMESTAMP);
            CONTINUE;
        END IF;

        BEGIN
            SELECT reward_key, rewardname, rewardtype, pointsrequired,
                   rewardvalue, rewardstatus
            INTO   v_key, v_name, v_type, v_points, v_value, v_status
            FROM   REWARD_DIM
            WHERE  rewardid = r.reward_id
            AND    flag = 'Y';

            IF NVL(UPPER(TRIM(v_name)), '#') <> NVL(UPPER(TRIM(r.rewardname)), '#')
            OR NVL(v_type, '#') <> NVL(r.rewardtype, '#')
            OR NVL(v_points, -1) <> NVL(r.points_required, -1)
            OR NVL(v_value, -1) <> NVL(r.rewardvalue, -1)
            OR NVL(v_status, '#') <> NVL(r.rewardstatus, '#') THEN

                UPDATE REWARD_DIM
                SET effectiveenddate = p_effective_date - 1,
                    flag = 'N'
                WHERE reward_key = v_key;

                INSERT INTO REWARD_DIM
                VALUES (REWARD_DIM_SEQ.NEXTVAL, r.reward_id, r.rewardname,
                        r.rewardtype, r.points_required, r.rewardvalue,
                        r.rewardstatus, p_effective_date, DATE '9999-12-31', 'Y');
            END IF;

        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                INSERT INTO REWARD_DIM
                VALUES (REWARD_DIM_SEQ.NEXTVAL, r.reward_id, r.rewardname,
                        r.rewardtype, r.points_required, r.rewardvalue,
                        r.rewardstatus, p_effective_date, DATE '9999-12-31', 'Y');
        END;
    END LOOP;
END;

CREATE OR REPLACE PROCEDURE PRC_LOAD_EMPLOYEE_SCD2 (
    p_effective_date IN DATE,
    p_batch_to       IN DATE
)
IS
    v_key       EMPLOYEE_DIM.EMPLOYEEKEY%TYPE;
    v_name      EMPLOYEE_DIM.EMPLOYEENAME%TYPE;
    v_position  EMPLOYEE_DIM.POSITION%TYPE;
    v_salary    EMPLOYEE_DIM.SALARY%TYPE;
    v_hiredate  EMPLOYEE_DIM.HIREDATE%TYPE;
    v_resigndate EMPLOYEE_DIM.RESIGNDATE%TYPE;
    v_new_resign EMPLOYEE_DIM.RESIGNDATE%TYPE;
BEGIN
    FOR r IN (SELECT * FROM VW_STG_EMPLOYEE_CLEAN) LOOP
        IF r.valid_flag = 'N' THEN
            INSERT INTO ETL_REJECT_LOG
            VALUES (ETL_REJECT_SEQ.NEXTVAL, p_effective_date, p_batch_to,
                    'EMPLOYEE', r.employee_id, r.reject_reason, SYSTIMESTAMP);
            CONTINUE;
        END IF;

        BEGIN
            SELECT employeekey, employeename, position, salary,
                   hiredate, resigndate
            INTO   v_key, v_name, v_position, v_salary,
                   v_hiredate, v_resigndate
            FROM   EMPLOYEE_DIM
            WHERE  employeeid = r.employee_id
            AND    flag = 'Y';

            IF r.employee_status = 'Inactive' THEN
                IF v_resigndate = DATE '9999-12-31' THEN
                    v_new_resign := p_effective_date;
                ELSE
                    v_new_resign := v_resigndate;
                END IF;
            ELSE
                v_new_resign := DATE '9999-12-31';
            END IF;

            IF NVL(UPPER(TRIM(v_name)), '#') <> NVL(UPPER(TRIM(r.employeename)), '#')
            OR NVL(v_position, '#') <> NVL(r.position, '#')
            OR NVL(v_salary, -1) <> NVL(r.salary, -1)
            OR NVL(v_hiredate, DATE '1900-01-01') <> NVL(r.hire_date, DATE '1900-01-01')
            OR NVL(v_resigndate, DATE '1900-01-01') <> NVL(v_new_resign, DATE '1900-01-01') THEN

                UPDATE EMPLOYEE_DIM
                SET effectiveenddate = p_effective_date - 1,
                    flag = 'N'
                WHERE employeekey = v_key;

                INSERT INTO EMPLOYEE_DIM
                VALUES (EMPLOYEE_DIM_SEQ.NEXTVAL, r.employee_id, r.employeename,
                        r.position, r.salary, r.hire_date, v_new_resign,
                        p_effective_date, DATE '9999-12-31', 'Y');
            END IF;

        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                v_new_resign := CASE WHEN r.employee_status = 'Inactive'
                                     THEN p_effective_date
                                     ELSE DATE '9999-12-31' END;
                INSERT INTO EMPLOYEE_DIM
                VALUES (EMPLOYEE_DIM_SEQ.NEXTVAL, r.employee_id, r.employeename,
                        r.position, r.salary, r.hire_date, v_new_resign,
                        p_effective_date, DATE '9999-12-31', 'Y');
        END;
    END LOOP;
END;

/* CUSTOMER_DIM is special because membership history comes from Member_Subscription.
   For the monthly batch, derive the member state as of p_effective_date. */
CREATE OR REPLACE PROCEDURE PRC_LOAD_CUSTOMER_SCD2 (
    p_effective_date IN DATE,
    p_batch_to       IN DATE
)
IS
    v_key        CUSTOMER_DIM.CUSTOMERKEY%TYPE;
    v_memberid   CUSTOMER_DIM.MEMBERID%TYPE;
    v_name       CUSTOMER_DIM.NAME%TYPE;
    v_gender     CUSTOMER_DIM.GENDER%TYPE;
    v_type       CUSTOMER_DIM.MEMBERSHIPTYPE%TYPE;
    v_rate       CUSTOMER_DIM.POINTRATE%TYPE;
    v_old_end    CUSTOMER_DIM.EFFECTIVEENDDATE%TYPE;
    v_target_end DATE;
BEGIN
    FOR r IN (
        SELECT
            c.customer_id,
            c.member_id,
            CASE WHEN c.member_id IS NULL THEN 'Non-Member'
                 ELSE REGEXP_REPLACE(TRIM(m.name), '[[:space:]]+', ' ') END AS customer_name,
            CASE WHEN c.member_id IS NULL THEN 'Unknown' ELSE m.gender END AS gender,
            CASE
                WHEN c.member_id IS NULL THEN 'None'
                WHEN EXISTS (
                    SELECT 1
                    FROM WAREHOUSEDATA.Member_Subscription ms
                    WHERE ms.member_id = c.member_id
                      AND ms.payment_status = 'Paid'
                      AND TRUNC(p_effective_date) BETWEEN TRUNC(ms.start_date) AND TRUNC(ms.end_date)
                ) THEN 'VIP'
                ELSE 'Normal'
            END AS membership_type,
            CASE
                WHEN c.member_id IS NULL THEN 0
                WHEN EXISTS (
                    SELECT 1
                    FROM WAREHOUSEDATA.Member_Subscription ms
                    WHERE ms.member_id = c.member_id
                      AND ms.payment_status = 'Paid'
                      AND TRUNC(p_effective_date) BETWEEN TRUNC(ms.start_date) AND TRUNC(ms.end_date)
                ) THEN 2.00
                ELSE 1.00
            END AS point_rate
        FROM WAREHOUSEDATA.Customer c
        LEFT JOIN WAREHOUSEDATA.Member m
          ON m.member_id = c.member_id
    ) LOOP

        IF r.member_id IS NOT NULL AND TRIM(r.customer_name) IS NULL THEN
            INSERT INTO ETL_REJECT_LOG
            VALUES (ETL_REJECT_SEQ.NEXTVAL, p_effective_date, p_batch_to,
                    'CUSTOMER', r.customer_id,
                    'Member name is empty after trimming', SYSTIMESTAMP);
            CONTINUE;
        END IF;

        /* End date for the membership state known at the batch date. */
        IF r.membership_type = 'VIP' THEN
            SELECT MAX(ms.end_date)
            INTO v_target_end
            FROM WAREHOUSEDATA.Member_Subscription ms
            WHERE ms.member_id = r.member_id
              AND ms.payment_status = 'Paid'
              AND TRUNC(p_effective_date) BETWEEN TRUNC(ms.start_date) AND TRUNC(ms.end_date);
        ELSE
            SELECT NVL(MIN(ms.start_date) - 1, DATE '9999-12-31')
            INTO v_target_end
            FROM WAREHOUSEDATA.Member_Subscription ms
            WHERE ms.member_id = r.member_id
              AND ms.payment_status = 'Paid'
              AND ms.start_date > p_effective_date;
        END IF;

        BEGIN
            /* Find the customer version valid at the batch effective date.
               This avoids depending on a possibly incorrect old FLAG value. */
            SELECT customerkey, memberid, name, gender, membershiptype,
                   pointrate, effectiveenddate
            INTO   v_key, v_memberid, v_name, v_gender, v_type,
                   v_rate, v_old_end
            FROM (
                SELECT cd.*
                FROM CUSTOMER_DIM cd
                WHERE cd.customerid = r.customer_id
                  AND TRUNC(p_effective_date)
                      BETWEEN TRUNC(cd.effectivestartdate) AND TRUNC(cd.effectiveenddate)
                ORDER BY cd.effectivestartdate DESC, cd.customerkey DESC
            )
            WHERE ROWNUM = 1;

            IF NVL(v_memberid, '#') <> NVL(r.member_id, '#')
            OR NVL(UPPER(TRIM(v_name)), '#') <> NVL(UPPER(TRIM(r.customer_name)), '#')
            OR NVL(v_gender, '#') <> NVL(r.gender, '#')
            OR NVL(v_type, '#') <> NVL(r.membership_type, '#')
            OR NVL(v_rate, -1) <> NVL(r.point_rate, -1) THEN

                UPDATE CUSTOMER_DIM
                SET effectiveenddate = p_effective_date - 1,
                    flag = 'N'
                WHERE customerkey = v_key;

                INSERT INTO CUSTOMER_DIM
                VALUES (CUSTOMER_DIM_SEQ.NEXTVAL, r.customer_id, r.member_id,
                        r.customer_name, r.gender, r.membership_type, r.point_rate,
                        p_effective_date, v_target_end, 'Y');
            END IF;

        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                INSERT INTO CUSTOMER_DIM
                VALUES (CUSTOMER_DIM_SEQ.NEXTVAL, r.customer_id, r.member_id,
                        r.customer_name, r.gender, r.membership_type, r.point_rate,
                        p_effective_date, v_target_end, 'Y');
        END;

        /* Keep one row marked current for this batch date. */
        UPDATE CUSTOMER_DIM
        SET flag = 'N'
        WHERE customerid = r.customer_id;

        UPDATE CUSTOMER_DIM
        SET flag = 'Y'
        WHERE customerkey = (
            SELECT customerkey
            FROM (
                SELECT cd.customerkey
                FROM CUSTOMER_DIM cd
                WHERE cd.customerid = r.customer_id
                  AND TRUNC(p_effective_date)
                      BETWEEN TRUNC(cd.effectivestartdate) AND TRUNC(cd.effectiveenddate)
                ORDER BY cd.effectivestartdate DESC, cd.customerkey DESC
            )
            WHERE ROWNUM = 1
        );
    END LOOP;
END;

CREATE OR REPLACE PROCEDURE PRC_LOAD_ORDER_PROFILE (
    p_from_date IN DATE,
    p_to_date   IN DATE
)
IS
BEGIN
    INSERT INTO ORDER_PROFILE_DIM (
        order_profile_key, order_type, service_type,
        order_status, mykasih_purchase
    )

           x.order_type, x.service_type, x.order_status, x.mykasih_purchase

          AND op.order_status = x.order_status
          AND op.mykasih_purchase = x.mykasih_purchase
    );

CREATE OR REPLACE PROCEDURE PRC_MERGE_SALES_FACT (
    p_from_date IN DATE,
    p_to_date   IN DATE
)
IS
BEGIN
    MERGE INTO SALES_FACT sf
    USING (
        SELECT
            d.DATE_KEY AS datekey,
            b.BRANCHKEY AS branchkey,
            NVL(c.CUSTOMERKEY, -1) AS customerkey,
            p.PRODUCTKEY AS productkey,
            op.ORDER_PROFILE_KEY AS orderprofilekey,
            e.EMPLOYEEKEY AS employeekey,
            o.order_no AS orderno,
            od.unit_price AS unitsellingprice,
            od.quantity AS purchasequantity,
            od.subtotal AS grosslineamount,
            CASE
                WHEN rw.reward_type = 'Discount'
                THEN ROUND(
                    o.discount_amount * od.subtotal /
                    NULLIF(o.subtotal_amount, 0),
                    2
                )
                ELSE 0
            END AS allocateddiscountamount,
            od.subtotal
            - CASE
                WHEN rw.reward_type = 'Discount'
                THEN ROUND(
                    o.discount_amount * od.subtotal /
                    NULLIF(o.subtotal_amount, 0),
                    2
                )
                ELSE 0
            END
            - CASE
                WHEN rr.status = 'Refunded'
                THEN NVL(rr.refund_amount, 0)
                ELSE 0
            END AS netlineamount,
            rt.score AS ratingscore,
            NVL(rr.return_quantity, 0) AS refundquantity,
            CASE
                WHEN rr.status = 'Refunded'
                THEN NVL(rr.refund_amount, 0)
                ELSE 0
            END  AS refundtotal
        FROM WAREHOUSEDATA.Order_Detail od
        JOIN WAREHOUSEDATA.Orders o
          ON o.order_no = od.order_no
        JOIN DATE_DIM d
          ON d.CAL_DATE = TRUNC(CAST(o.order_date_time AS DATE))
        JOIN BRANCH_DIM b
          ON b.BRANCHID = o.branch_id
         AND TRUNC(CAST(o.order_date_time AS DATE))
             BETWEEN b.EFFECTIVESTARTDATE AND b.EFFECTIVEENDDATE
        LEFT JOIN CUSTOMER_DIM c
          ON c.CUSTOMERID = o.customer_id
         AND TRUNC(CAST(o.order_date_time AS DATE))
             BETWEEN c.EFFECTIVESTARTDATE AND c.EFFECTIVEENDDATE
        JOIN PRODUCT_DIM p
          ON p.ITEMID = od.item_id
         AND TRUNC(CAST(o.order_date_time AS DATE))
             BETWEEN p.EFFECTIVESTARTDATE AND p.EFFECTIVEENDDATE
        JOIN EMPLOYEE_DIM e
          ON e.EMPLOYEEID = o.handled_by
         AND TRUNC(CAST(o.order_date_time AS DATE))
             BETWEEN e.EFFECTIVESTARTDATE AND e.EFFECTIVEENDDATE
        JOIN ORDER_PROFILE_DIM op
          ON op.order_type = o.order_type
         AND op.service_type = o.service_type
         AND op.order_status = o.order_status
         AND op.mykasih_purchase =
             CASE WHEN o.is_mykasih_purchase = 'Y'
                  THEN 'MyKasih Subsidised' ELSE 'Standard Purchase' END
        LEFT JOIN WAREHOUSEDATA.Reward_Redemption red
          ON red.redemption_id = o.redemption_id
        LEFT JOIN WAREHOUSEDATA.Reward rw
          ON rw.reward_id = red.reward_id
        LEFT JOIN WAREHOUSEDATA.Rating rt
          ON rt.order_no = od.order_no
         AND rt.item_id = od.item_id
        LEFT JOIN WAREHOUSEDATA.Return_Request rr
          ON rr.order_no = od.order_no
         AND rr.item_id = od.item_id
        WHERE (o.order_date_time >= CAST(p_from_date AS TIMESTAMP)
           AND o.order_date_time <  CAST(p_to_date AS TIMESTAMP))
           OR (rt.rating_date >= p_from_date AND rt.rating_date < p_to_date)
           OR (rr.request_date >= p_from_date AND rr.request_date < p_to_date)
           OR EXISTS (
                SELECT 1
                FROM SALES_FACT oldsf
                JOIN ORDER_PROFILE_DIM oldop
                  ON oldop.ORDER_PROFILE_KEY = oldsf.ORDERPROFILEKEY
                WHERE oldsf.ORDERNO = o.order_no
                  AND oldsf.PRODUCTKEY = p.PRODUCTKEY
                  AND (
                       oldop.order_type <> o.order_type
                    OR oldop.service_type <> o.service_type
                    OR oldop.order_status <> o.order_status
                    OR oldop.mykasih_purchase <>
                       CASE WHEN o.is_mykasih_purchase = 'Y'
                            THEN 'MyKasih Subsidised' ELSE 'Standard Purchase' END
                  )
           )
    ) s
    ON (sf.ORDERNO = s.orderno AND sf.PRODUCTKEY = s.productkey)

    WHEN MATCHED THEN UPDATE SET
        sf.DATEKEY                 = s.datekey,
        sf.BRANCHKEY               = s.branchkey,
        sf.CUSTOMERKEY             = s.customerkey,
        sf.ORDERPROFILEKEY         = s.orderprofilekey,
        sf.EMPLOYEEKEY             = s.employeekey,
        sf.UNITSELLINGPRICE        = s.unitsellingprice,
        sf.PURCHASEQUANTITY        = s.purchasequantity,
        sf.GROSSLINEAMOUNT         = s.grosslineamount,
        sf.ALLOCATEDDISCOUNTAMOUNT = s.allocateddiscountamount,
        sf.NETLINEAMOUNT           = s.netlineamount,
        sf.RATINGSCORE             = s.ratingscore,
        sf.REFUNDQUANTITY          = s.refundquantity,
        sf.REFUNDTOTAL             = s.refundtotal

    WHEN NOT MATCHED THEN INSERT (
        DATEKEY, BRANCHKEY, CUSTOMERKEY, PRODUCTKEY, ORDERPROFILEKEY,
        EMPLOYEEKEY, ORDERNO, UNITSELLINGPRICE, PURCHASEQUANTITY,
        GROSSLINEAMOUNT, ALLOCATEDDISCOUNTAMOUNT, NETLINEAMOUNT,
        RATINGSCORE, REFUNDQUANTITY, REFUNDTOTAL
    ) VALUES (
        s.datekey, s.branchkey, s.customerkey, s.productkey,
        s.orderprofilekey, s.employeekey, s.orderno,
        s.unitsellingprice, s.purchasequantity, s.grosslineamount,
        s.allocateddiscountamount, s.netlineamount,
        s.ratingscore, s.refundquantity, s.refundtotal
    );
END;

CREATE OR REPLACE PROCEDURE PRC_MERGE_DELIVERY_FACT (
    p_from_date IN DATE,
    p_to_date   IN DATE
)
IS
BEGIN
    /* Dirty-data rule: a delivery cannot happen before its order. */
    INSERT INTO ETL_REJECT_LOG
    SELECT ETL_REJECT_SEQ.NEXTVAL, p_from_date, p_to_date,
           'DELIVERY_REQUEST', dr.delivery_request_id,
           'Delivered date/time is earlier than order date/time', SYSTIMESTAMP
    FROM WAREHOUSEDATA.Delivery_Request dr
    JOIN WAREHOUSEDATA.Orders o ON o.order_no = dr.order_no
    WHERE dr.delivered_date_time IS NOT NULL
      AND dr.delivered_date_time < o.order_date_time
      AND (
            (o.order_date_time >= CAST(p_from_date AS TIMESTAMP)
             AND o.order_date_time < CAST(p_to_date AS TIMESTAMP))
         OR (dr.delivered_date_time >= CAST(p_from_date AS TIMESTAMP)
             AND dr.delivered_date_time < CAST(p_to_date AS TIMESTAMP))
      );

    MERGE INTO DELIVERY_FACT df
    USING (
        SELECT
            d.DATE_KEY AS datekey,
            NVL(dd.DATE_KEY, -1) AS delivereddatekey,
            b.BRANCHKEY AS branchkey,
            NVL(c.CUSTOMERKEY, -1) AS customerkey,
            cr.COURIERKEY AS courierkey,
            ds.DELIVERY_STATUS_KEY AS delivery_status_key,
            dr.delivery_request_id AS deliveryrequestid,
            dr.order_no AS orderno,
            NVL(dr.delivery_fee, 0) AS deliveryfee,
            o.total_amount AS ordertotalamount,
            CASE WHEN dr.delivered_date_time IS NOT NULL
                 THEN ROUND((CAST(dr.delivered_date_time AS DATE)
                           - CAST(o.order_date_time AS DATE)) * 24, 2)
            END AS deliveryleadhours,
            dr.rating AS deliveryrating,
            CASE WHEN dr.rating IS NOT NULL THEN 1 ELSE 0 END AS deliveryratingcount,
            CASE WHEN dr.status = 'Failed' THEN 1 ELSE 0 END AS faileddeliverycount
        FROM WAREHOUSEDATA.Delivery_Request dr
        JOIN WAREHOUSEDATA.Orders o ON o.order_no = dr.order_no
        JOIN DATE_DIM d
          ON d.CAL_DATE = TRUNC(CAST(o.order_date_time AS DATE))
        LEFT JOIN DATE_DIM dd
          ON dd.CAL_DATE = TRUNC(CAST(dr.delivered_date_time AS DATE))
        JOIN BRANCH_DIM b
          ON b.BRANCHID = o.branch_id
         AND TRUNC(CAST(o.order_date_time AS DATE))
             BETWEEN b.EFFECTIVESTARTDATE AND b.EFFECTIVEENDDATE
        LEFT JOIN CUSTOMER_DIM c
          ON c.CUSTOMERID = o.customer_id
         AND TRUNC(CAST(o.order_date_time AS DATE))
             BETWEEN c.EFFECTIVESTARTDATE AND c.EFFECTIVEENDDATE
        JOIN COURIER_DIM cr
          ON cr.COURIERID = dr.courier_id
         AND TRUNC(CAST(o.order_date_time AS DATE))
             BETWEEN cr.EFFECTIVESTARTDATE AND cr.EFFECTIVEENDDATE
        JOIN DELIVERY_STATUS_DIM ds
          ON ds.DELIVERY_STATUS = dr.status
        WHERE (dr.delivered_date_time IS NULL
               OR dr.delivered_date_time >= o.order_date_time)
          AND (
                (o.order_date_time >= CAST(p_from_date AS TIMESTAMP)
                 AND o.order_date_time < CAST(p_to_date AS TIMESTAMP))
             OR (dr.delivered_date_time >= CAST(p_from_date AS TIMESTAMP)
                 AND dr.delivered_date_time < CAST(p_to_date AS TIMESTAMP))
             OR EXISTS (
                    SELECT 1
                    FROM DELIVERY_FACT oldf
                    JOIN DELIVERY_STATUS_DIM oldds
                      ON oldds.DELIVERY_STATUS_KEY = oldf.DELIVERY_STATUS_KEY
                    WHERE oldf.ORDERNO = dr.order_no
                      AND (
                           oldds.DELIVERY_STATUS <> dr.status
                        OR NVL(oldf.DELIVERYRATING, -1) <> NVL(dr.rating, -1)
                        OR NVL(oldf.DELIVERYFEE, -1) <> NVL(dr.delivery_fee, -1)
                      )
                )
          )
    ) s
    ON (df.ORDERNO = s.orderno)

    WHEN MATCHED THEN UPDATE SET
        df.DATEKEY              = s.datekey,
        df.DELIVEREDDATEKEY     = s.delivereddatekey,
        df.BRANCHKEY            = s.branchkey,
        df.CUSTOMERKEY          = s.customerkey,
        df.COURIERKEY           = s.courierkey,
        df.DELIVERY_STATUS_KEY  = s.delivery_status_key,
        df.DELIVERYREQUESTID    = s.deliveryrequestid,
        df.DELIVERYFEE          = s.deliveryfee,
        df.ORDERTOTALAMOUNT     = s.ordertotalamount,
        df.DELIVERYLEADHOURS    = s.deliveryleadhours,
        df.DELIVERYRATING       = s.deliveryrating,
        df.DELIVERYRATINGCOUNT  = s.deliveryratingcount,
        df.FAILEDDELIVERYCOUNT  = s.faileddeliverycount

    WHEN NOT MATCHED THEN INSERT (
        DATEKEY, DELIVEREDDATEKEY, BRANCHKEY, CUSTOMERKEY, COURIERKEY,
        DELIVERY_STATUS_KEY, DELIVERYREQUESTID, ORDERNO,
        DELIVERYFEE, ORDERTOTALAMOUNT, DELIVERYLEADHOURS,
        DELIVERYRATING, DELIVERYRATINGCOUNT, FAILEDDELIVERYCOUNT
    ) VALUES (
        s.datekey, s.delivereddatekey, s.branchkey, s.customerkey,
        s.courierkey, s.delivery_status_key, s.deliveryrequestid,
        s.orderno, s.deliveryfee, s.ordertotalamount,
        s.deliveryleadhours, s.deliveryrating,
        s.deliveryratingcount, s.faileddeliverycount
    );
END;

CREATE OR REPLACE PROCEDURE PRC_MERGE_REWARD_FACT (
    p_from_date IN DATE,
    p_to_date   IN DATE
)
IS
BEGIN
    MERGE INTO REWARD_FACT rf
    USING (
        SELECT
            d.DATE_KEY AS datekey,
            NVL(c.CUSTOMERKEY, -1) AS customerkey,
            rd.REWARD_KEY AS rewardkey,
            rr.redemption_id AS redemptionid,
            rr.quantity_redeemed AS quantityredeemed,
            rr.point_spent AS pointspend,
            ROUND(rr.quantity_redeemed * rd.REWARDVALUE, 2) AS rewardvalue,
            CASE WHEN rr.status = 'Used' THEN 1 ELSE 0 END AS usedcount,
            CASE WHEN rr.status = 'Expired' THEN 1 ELSE 0 END AS expiredcount,
            CASE WHEN rr.status = 'Active' THEN 1 ELSE 0 END AS activecount
        FROM WAREHOUSEDATA.Reward_Redemption rr
        JOIN WAREHOUSEDATA.Customer cu
          ON cu.member_id = rr.member_id
        JOIN DATE_DIM d
          ON d.CAL_DATE = TRUNC(rr.issue_date)
        LEFT JOIN CUSTOMER_DIM c
          ON c.CUSTOMERID = cu.customer_id
         AND TRUNC(rr.issue_date)
             BETWEEN c.EFFECTIVESTARTDATE AND c.EFFECTIVEENDDATE
        JOIN REWARD_DIM rd
          ON rd.REWARDID = rr.reward_id
         AND TRUNC(rr.issue_date)
             BETWEEN rd.EFFECTIVESTARTDATE AND rd.EFFECTIVEENDDATE
        WHERE (rr.issue_date >= p_from_date AND rr.issue_date < p_to_date)
           OR EXISTS (
                SELECT 1
                FROM REWARD_FACT oldrf
                WHERE oldrf.REDEMPTIONID = rr.redemption_id
                  AND (
                       oldrf.USEDCOUNT <> CASE WHEN rr.status = 'Used' THEN 1 ELSE 0 END
                    OR oldrf.EXPIREDCOUNT <> CASE WHEN rr.status = 'Expired' THEN 1 ELSE 0 END
                    OR oldrf.ACTIVECOUNT <> CASE WHEN rr.status = 'Active' THEN 1 ELSE 0 END
                  )
           )
    ) s
    ON (rf.REDEMPTIONID = s.redemptionid)

    WHEN MATCHED THEN UPDATE SET
        rf.DATEKEY          = s.datekey,
        rf.CUSTOMERKEY      = s.customerkey,
        rf.REWARDKEY        = s.rewardkey,
        rf.QUANTITYREDEEMED = s.quantityredeemed,
        rf.POINTSPEND       = s.pointspend,
        rf.REWARDVALUE      = s.rewardvalue,
        rf.USEDCOUNT        = s.usedcount,
        rf.EXPIREDCOUNT     = s.expiredcount,
        rf.ACTIVECOUNT      = s.activecount

    WHEN NOT MATCHED THEN INSERT (
        DATEKEY, CUSTOMERKEY, REWARDKEY, REDEMPTIONID,
        QUANTITYREDEEMED, POINTSPEND, REWARDVALUE,
        USEDCOUNT, EXPIREDCOUNT, ACTIVECOUNT
    ) VALUES (
        s.datekey, s.customerkey, s.rewardkey, s.redemptionid,
        s.quantityredeemed, s.pointspend, s.rewardvalue,
        s.usedcount, s.expiredcount, s.activecount
    );
END;
/

/* ---------- 6. Master monthly procedure ---------- */
CREATE OR REPLACE PROCEDURE PRC_SUBSEQUENT_LOAD (
    p_from_date IN DATE,
    p_to_date   IN DATE
)
IS
BEGIN
    IF p_from_date IS NULL OR p_to_date IS NULL OR p_from_date >= p_to_date THEN
        RAISE_APPLICATION_ERROR(-20001, 'Invalid ETL date range.');
    END IF;

    /* Allow the same batch to be re-tested without duplicating reject rows. */
    DELETE FROM ETL_REJECT_LOG
    WHERE batch_from = p_from_date
      AND batch_to   = p_to_date;

    /* Dimensions first. */
    PRC_LOAD_BRANCH_SCD2(p_from_date, p_to_date);
    PRC_LOAD_COURIER_SCD2(p_from_date, p_to_date);
    PRC_LOAD_PRODUCT_SCD2(p_from_date, p_to_date);
    PRC_LOAD_REWARD_SCD2(p_from_date, p_to_date);
    PRC_LOAD_EMPLOYEE_SCD2(p_from_date, p_to_date);
    PRC_LOAD_CUSTOMER_SCD2(p_from_date, p_to_date);
    PRC_LOAD_ORDER_PROFILE(p_from_date, p_to_date);

    /* Facts after dimensions. */
    PRC_MERGE_SALES_FACT(p_from_date, p_to_date);
    PRC_MERGE_DELIVERY_FACT(p_from_date, p_to_date);
    PRC_MERGE_REWARD_FACT(p_from_date, p_to_date);

    COMMIT;

    DBMS_OUTPUT.PUT_LINE('Subsequent ETL completed: '
        || TO_CHAR(p_from_date, 'YYYY-MM-DD') || ' to before '
        || TO_CHAR(p_to_date, 'YYYY-MM-DD'));
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('ETL failed: ' || SQLERRM);
        RAISE;
END;
/

/* ---------- 7. Example subsequent load ---------- */
EXEC PRC_SUBSEQUENT_LOAD(DATE '2026-07-01',DATE '2026-08-01');
