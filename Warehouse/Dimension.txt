-- DIMENSION INITIAL LOAD
-- Historical transaction cutoff for initial load:
-- all transaction facts/order profiles before 01-Jul-2026,
-- therefore the initial warehouse contains data through 30-Jun-2026.

CREATE OR REPLACE PROCEDURE DATE_GENERATION
IS
    v_start_date DATE := DATE '2015-01-01';
    v_end_date DATE := DATE '2026-12-31';
BEGIN
    WHILE v_start_date <= v_end_date LOOP
        INSERT INTO DATE_DIM(
            DATE_KEY,
            CAL_DATE,
            FULL_DESC,
            DAY_WEEK,
            DAY_NUM_MONTH,
            DAY_NUM_YEAR,
            LAST_DAY_IND,
            CAL_WEEK_END_DATE,
            CAL_WEEK_YEAR,
            CAL_MONTH_NAME,
            CAL_MONTH_YEAR,
            CAL_YEAR_MONTH,
            CAL_QUARTER,
            CAL_YEAR_QUARTER,
            CAL_YEAR
        ) VALUES (
            DATE_SEQ.NEXTVAL,
            v_start_date,
            TO_CHAR(v_start_date, 'DL'),
            TO_CHAR(v_start_date, 'DAY'),
            TO_NUMBER(TO_CHAR(v_start_date, 'DD')),
            TO_NUMBER(TO_CHAR(v_start_date, 'DDD')),
            CASE WHEN v_start_date = LAST_DAY(v_start_date) THEN 'Y' ELSE 'N' END,
            TRUNC(v_start_date, 'IW') + 6,
            TO_NUMBER(TO_CHAR(v_start_date, 'IW')),
            TO_CHAR(v_start_date, 'MONTH'),
            TO_NUMBER(TO_CHAR(v_start_date, 'MM')),
            TO_CHAR(v_start_date, 'YYYYMM'),
            'Q' || TO_CHAR(v_start_date, 'Q'),
            TO_CHAR(v_start_date, 'YYYY') || '-Q' || TO_CHAR(v_start_date, 'Q'),
            TO_NUMBER(TO_CHAR(v_start_date, 'YYYY'))
        );
        v_start_date := v_start_date + 1;
    END LOOP;
END;
/

EXECUTE DATE_GENERATION;
COMMIT;

DECLARE
    v_start DATE;
BEGIN
    -- (a) Customers with no membership: single Normal-equivalent version.
    INSERT INTO CUSTOMER_DIM
    SELECT CUSTOMER_DIM_SEQ.NEXTVAL, c.customer_id, NULL,
           'Non-Member', 'Unknown', 'None', 0,
           DATE '2020-01-01', DATE '9999-12-31', 'Y'
    FROM WAREHOUSEDATA.Customer c
    WHERE c.member_id IS NULL;

    -- (b) Members: walk the paid subscription windows in date order.
    FOR m IN (
        SELECT c.customer_id, c.member_id, mb.name, mb.gender
        FROM WAREHOUSEDATA.Customer c
        JOIN WAREHOUSEDATA.Member mb
          ON mb.member_id = c.member_id
    )
    LOOP
        v_start := DATE '2020-01-01';

        FOR s IN (
            SELECT ms.start_date, ms.end_date
            FROM WAREHOUSEDATA.Member_Subscription ms
            WHERE ms.member_id = m.member_id
              AND ms.payment_status = 'Paid'
            ORDER BY ms.start_date
        )
        LOOP
            IF s.start_date > v_start THEN
                INSERT INTO CUSTOMER_DIM
                VALUES (
                    CUSTOMER_DIM_SEQ.NEXTVAL,
                    m.customer_id,
                    m.member_id,
                    m.name,
                    m.gender,
                    'Normal',
                    1.00,
                    v_start,
                    s.start_date - 1,
                    'N'
                );
            END IF;

            INSERT INTO CUSTOMER_DIM
            VALUES (
                CUSTOMER_DIM_SEQ.NEXTVAL,
                m.customer_id,
                m.member_id,
                m.name,
                m.gender,
                'VIP',
                2.00,
                s.start_date,
                s.end_date,
                'N'
            );

            v_start := s.end_date + 1;
        END LOOP;

        INSERT INTO CUSTOMER_DIM
        VALUES (
            CUSTOMER_DIM_SEQ.NEXTVAL,
            m.customer_id,
            m.member_id,
            m.name,
            m.gender,
            'Normal',
            1.00,
            v_start,
            DATE '9999-12-31',
            'N'
        );
    END LOOP;

    /* ========================================================
       SET CURRENT CUSTOMER VERSION AS AT INITIAL LOAD CUTOFF
       ======================================================== */
    UPDATE CUSTOMER_DIM
    SET FLAG = 'N'
    WHERE CUSTOMERKEY <> -1;

    UPDATE CUSTOMER_DIM
    SET FLAG = 'Y'
    WHERE CUSTOMERKEY IN (
        SELECT CUSTOMERKEY
        FROM (
            SELECT
                CUSTOMERKEY,
                ROW_NUMBER() OVER (
                    PARTITION BY CUSTOMERID
                    ORDER BY EFFECTIVESTARTDATE DESC, CUSTOMERKEY DESC
                ) AS rn
            FROM CUSTOMER_DIM
            WHERE CUSTOMERID IS NOT NULL
              AND DATE '2026-06-30'
                  BETWEEN EFFECTIVESTARTDATE AND EFFECTIVEENDDATE
        )
        WHERE rn = 1
    );

    COMMIT;
END;
/

-- FACT INITIAL LOAD
-- INITIAL LOAD CUTOFF: THROUGH 30-JUN-2026 ONLY.
-- 01-JUL-2026 TO 20-JUL-2026 IS RESERVED FOR SUBSEQUENT LOAD.
--------------------------------------------------------------------------------

INSERT INTO SALES_FACT (
    DATEKEY, BRANCHKEY, CUSTOMERKEY, PRODUCTKEY, ORDERPROFILEKEY, EMPLOYEEKEY,
    ORDERNO, UNITSELLINGPRICE, PURCHASEQUANTITY, GROSSLINEAMOUNT,
    ALLOCATEDDISCOUNTAMOUNT, NETLINEAMOUNT, RATINGSCORE,
    REFUNDQUANTITY, REFUNDTOTAL
)
SELECT
    d.DATE_KEY,
    b.BRANCHKEY,
    NVL(c.CUSTOMERKEY, -1),
    p.PRODUCTKEY,
    op.ORDER_PROFILE_KEY,
    e.EMPLOYEEKEY,
    o.order_no,
    od.unit_price,
    od.quantity,
    od.subtotal,
    CASE
        WHEN rw.reward_type = 'Discount'
        THEN ROUND(
            o.discount_amount * od.subtotal /
            NULLIF(o.subtotal_amount, 0),
            2
        )
        ELSE 0
    END,
    od.subtotal
    - CASE
        WHEN rw.reward_type = 'Discount'
        THEN ROUND(
            o.discount_amount * od.subtotal /
            NULLIF(o.subtotal_amount, 0),
            2
        )
        ELSE 0
    END
    - CASE
        WHEN rr.status = 'Refunded'
        THEN NVL(rr.refund_amount, 0)
        ELSE 0
    END,
    rt.score,
    NVL(rr.return_quantity, 0),
    CASE
        WHEN rr.status = 'Refunded'
        THEN NVL(rr.refund_amount, 0)
        ELSE 0
    END
FROM WAREHOUSEDATA.Order_Detail od
JOIN WAREHOUSEDATA.Orders o
  ON o.order_no = od.order_no
JOIN DATE_DIM d
  ON d.CAL_DATE = TRUNC(CAST(o.order_date_time AS DATE))
JOIN BRANCH_DIM b
  ON b.BRANCHID = o.branch_id
 AND CAST(o.order_date_time AS DATE)
     BETWEEN b.EFFECTIVESTARTDATE AND b.EFFECTIVEENDDATE
LEFT JOIN CUSTOMER_DIM c
  ON c.CUSTOMERID = o.customer_id
 AND CAST(o.order_date_time AS DATE)
     BETWEEN c.EFFECTIVESTARTDATE AND c.EFFECTIVEENDDATE
JOIN PRODUCT_DIM p
  ON p.ITEMID = od.item_id
 AND CAST(o.order_date_time AS DATE)
     BETWEEN p.EFFECTIVESTARTDATE AND p.EFFECTIVEENDDATE
JOIN EMPLOYEE_DIM e
  ON e.EMPLOYEEID = o.handled_by
 AND CAST(o.order_date_time AS DATE)
     BETWEEN e.EFFECTIVESTARTDATE AND e.EFFECTIVEENDDATE
JOIN ORDER_PROFILE_DIM op
  ON op.order_type = o.order_type
 AND op.service_type = o.service_type
 AND op.order_status = o.order_status
 AND op.mykasih_purchase =
     CASE
         WHEN o.is_mykasih_purchase = 'Y' THEN 'MyKasih Subsidised'
         ELSE 'Standard Purchase'
     END
LEFT JOIN WAREHOUSEDATA.Rating rt
  ON rt.order_no = od.order_no
 AND rt.item_id = od.item_id
LEFT JOIN WAREHOUSEDATA.Return_Request rr
  ON rr.order_no = od.order_no
 AND rr.item_id = od.item_id
LEFT JOIN WAREHOUSEDATA.Reward_Redemption red
  ON red.redemption_id = o.redemption_id
LEFT JOIN WAREHOUSEDATA.Reward rw
  ON rw.reward_id = red.reward_id
WHERE o.order_date_time < TIMESTAMP '2026-07-01 00:00:00';


INSERT INTO DELIVERY_FACT (
    DATEKEY, DELIVEREDDATEKEY, BRANCHKEY, CUSTOMERKEY, COURIERKEY,
    DELIVERY_STATUS_KEY, DELIVERYREQUESTID, ORDERNO,
    DELIVERYFEE, ORDERTOTALAMOUNT, DELIVERYLEADHOURS,
    DELIVERYRATING, DELIVERYRATINGCOUNT, FAILEDDELIVERYCOUNT
)
SELECT
    d.DATE_KEY,
    NVL(dd.DATE_KEY, -1),
    b.BRANCHKEY,
    NVL(c.CUSTOMERKEY, -1),
    cr.COURIERKEY,
    ds.DELIVERY_STATUS_KEY,
    dr.delivery_request_id,
    dr.order_no,
    NVL(dr.delivery_fee, 0),
    o.total_amount,
    CASE
        WHEN dr.delivered_date_time IS NOT NULL
        THEN ROUND(
            (CAST(dr.delivered_date_time AS DATE)
             - CAST(o.order_date_time AS DATE)) * 24,
            2
        )
    END,
    dr.rating,
    CASE WHEN dr.rating IS NOT NULL THEN 1 ELSE 0 END,
    CASE WHEN dr.status = 'Failed' THEN 1 ELSE 0 END
FROM WAREHOUSEDATA.Delivery_Request dr
JOIN WAREHOUSEDATA.Orders o
  ON o.order_no = dr.order_no
JOIN DATE_DIM d
  ON d.CAL_DATE = TRUNC(CAST(o.order_date_time AS DATE))
LEFT JOIN DATE_DIM dd
  ON dd.CAL_DATE = TRUNC(CAST(dr.delivered_date_time AS DATE))
JOIN BRANCH_DIM b
  ON b.BRANCHID = o.branch_id
 AND CAST(o.order_date_time AS DATE)
     BETWEEN b.EFFECTIVESTARTDATE AND b.EFFECTIVEENDDATE
LEFT JOIN CUSTOMER_DIM c
  ON c.CUSTOMERID = o.customer_id
 AND CAST(o.order_date_time AS DATE)
     BETWEEN c.EFFECTIVESTARTDATE AND c.EFFECTIVEENDDATE
JOIN COURIER_DIM cr
  ON cr.COURIERID = dr.courier_id
 AND CAST(o.order_date_time AS DATE)
     BETWEEN cr.EFFECTIVESTARTDATE AND cr.EFFECTIVEENDDATE
JOIN DELIVERY_STATUS_DIM ds
  ON ds.DELIVERY_STATUS = dr.status
WHERE o.order_date_time < TIMESTAMP '2026-07-01 00:00:00'
  AND (
        dr.delivered_date_time IS NULL
        OR dr.delivered_date_time < TIMESTAMP '2026-07-01 00:00:00'
      );


INSERT INTO REWARD_FACT (
    DATEKEY, CUSTOMERKEY, REWARDKEY, REDEMPTIONID, QUANTITYREDEEMED,
    POINTSPEND, REWARDVALUE, USEDCOUNT, EXPIREDCOUNT, ACTIVECOUNT
)
SELECT
    d.DATE_KEY,
    NVL(c.CUSTOMERKEY, -1),
    rd.REWARD_KEY,
    rr.redemption_id,
    rr.quantity_redeemed,
    rr.point_spent,
    ROUND(rr.quantity_redeemed * r.value, 2),
    CASE WHEN rr.status = 'Used' THEN 1 ELSE 0 END,
    CASE WHEN rr.status = 'Expired' THEN 1 ELSE 0 END,
    CASE WHEN rr.status = 'Active' THEN 1 ELSE 0 END
FROM WAREHOUSEDATA.Reward_Redemption rr
JOIN WAREHOUSEDATA.Reward r
  ON r.reward_id = rr.reward_id
JOIN WAREHOUSEDATA.Customer cu
  ON cu.member_id = rr.member_id
JOIN DATE_DIM d
  ON d.CAL_DATE = rr.issue_date
LEFT JOIN CUSTOMER_DIM c
  ON c.CUSTOMERID = cu.customer_id
 AND rr.issue_date
     BETWEEN c.EFFECTIVESTARTDATE AND c.EFFECTIVEENDDATE
JOIN REWARD_DIM rd
  ON rd.REWARDID = rr.reward_id
 AND rr.issue_date
     BETWEEN rd.EFFECTIVESTARTDATE AND rd.EFFECTIVEENDDATE
WHERE rr.issue_date < DATE '2026-07-01';
