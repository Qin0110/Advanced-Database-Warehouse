-- DIMENSION INITIAL LOAD
CREATE OR REPLACE PROCEDURE DATE_GENERATION
IS
	v_start_date DATE := DATE '2015-01-01';
	v_end_date DATE := DATE '2026-12-31';
BEGIN
	WHILE v_start_date <= v_end_date LOOP
		INSERT INTO DATE_DIM(
			DATE_KEY,
			CAL_DATE,			
			FULL_DESC,
			DAY_WEEK,
			DAY_NUM_MONTH,
			DAY_NUM_YEAR,
			LAST_DAY_IND,
			CAL_WEEK_END_DATE,
			CAL_WEEK_YEAR,	
			CAL_MONTH_NAME,	
			CAL_MONTH_YEAR,
			CAL_YEAR_MONTH,	
			CAL_QUARTER,
			CAL_YEAR_QUARTER,
			CAL_YEAR
		) VALUES (
			DATE_SEQ.NEXTVAL,
			v_start_date,
			TO_CHAR(v_start_date, 'DL'),
			TO_CHAR(v_start_date, 'DAY'),
			TO_NUMBER(TO_CHAR(v_start_date, 'DD')),
			TO_NUMBER(TO_CHAR(v_start_date, 'DDD')),
			CASE WHEN v_start_date = LAST_DAY(v_start_date) THEN 'Y' ELSE 'N' END,
			TRUNC(v_start_date, 'IW') + 6,
			TO_NUMBER(TO_CHAR(v_start_date, 'IW')),
			TO_CHAR(v_start_date, 'MONTH'),
			TO_NUMBER(TO_CHAR(v_start_date, 'MM')),
			TO_CHAR(v_start_date, 'YYYYMM'),
			'Q' || TO_CHAR(v_start_date, 'Q'),
			TO_CHAR(v_start_date, 'YYYY') || '-Q' || TO_CHAR(v_start_date, 'Q'),
			TO_NUMBER(TO_CHAR(v_start_date, 'YYYY'))
		);
		v_start_date := v_start_date + 1;
	END LOOP;	
END;
/

EXECUTE DATE_GENERATION;
COMMIT;

-- Customer Dimension (Fix Later)
DECLARE
    v_start DATE;
BEGIN
    -- (a) Customers with no membership: single Normal-equivalent version.
    INSERT INTO CUSTOMER_DIM
    SELECT CUSTOMER_DIM_SEQ.NEXTVAL, c.customer_id, NULL,
           'Non-Member', 'Unknown', 'None', 0,
           DATE '2020-01-01', DATE '9999-12-31', 'Y'
    FROM   WAREHOUSEDATA.Customer c
    WHERE  c.member_id IS NULL;
 
    -- (b) Members: walk the paid subscription windows in date order.
    FOR m IN (SELECT c.customer_id, c.member_id, mb.name, mb.gender
              FROM   WAREHOUSEDATA.Customer c
              JOIN   WAREHOUSEDATA.Member   mb ON mb.member_id = c.member_id)
    LOOP
        v_start := DATE '2020-01-01';
 
        FOR s IN (SELECT ms.start_date, ms.end_date
                  FROM   WAREHOUSEDATA.Member_Subscription ms
                  WHERE  ms.member_id      = m.member_id
                  AND    ms.payment_status = 'Paid'
                  ORDER  BY ms.start_date)
        LOOP
            -- Normal period before this VIP window opens
            IF s.start_date > v_start THEN
                INSERT INTO CUSTOMER_DIM
                VALUES (CUSTOMER_DIM_SEQ.NEXTVAL, m.customer_id, m.member_id,
                        m.name, m.gender, 'Normal', 1.00,
                        v_start, s.start_date - 1, 'N');
            END IF;
 
            -- The VIP window itself
            INSERT INTO CUSTOMER_DIM
            VALUES (CUSTOMER_DIM_SEQ.NEXTVAL, m.customer_id, m.member_id,
                    m.name, m.gender, 'VIP', 2.00,
                    s.start_date, s.end_date, 'N');
 
            v_start := s.end_date + 1;
        END LOOP;
 
        -- Trailing Normal period after the last subscription lapsed
        INSERT INTO CUSTOMER_DIM
        VALUES (CUSTOMER_DIM_SEQ.NEXTVAL, m.customer_id, m.member_id,
                m.name, m.gender, 'Normal', 1.00,
                v_start, DATE '9999-12-31', 'N');
    END LOOP;
 
    COMMIT;
END;

-- Fact table
CREATE TABLE SALES_FACT (
    DATEKEY                  NUMBER        NOT NULL,
    BRANCHKEY                NUMBER        NOT NULL,
    CUSTOMERKEY              NUMBER        NOT NULL,
    PRODUCTKEY               NUMBER        NOT NULL,
    ORDERPROFILEKEY          NUMBER(6)     NOT NULL,
    EMPLOYEEKEY              NUMBER        NOT NULL,
    ORDERNO                  VARCHAR2(20)  NOT NULL,  
    UNITSELLINGPRICE         NUMBER(7,2)   DEFAULT 0 NOT NULL,
    PURCHASEQUANTITY         NUMBER(7)     DEFAULT 0 NOT NULL,
    GROSSLINEAMOUNT          NUMBER(12,2)  DEFAULT 0 NOT NULL,
    ALLOCATEDDISCOUNTAMOUNT  NUMBER(12,2)  DEFAULT 0 NOT NULL,
    NETLINEAMOUNT            NUMBER(12,2)  DEFAULT 0 NOT NULL,
    RATINGSCORE              NUMBER(3,2),                     
    REFUNDQUANTITY           NUMBER(7)     DEFAULT 0 NOT NULL,
    REFUNDTOTAL              NUMBER(12,2)  DEFAULT 0 NOT NULL,
	PRIMARY KEY (DATEKEY, BRANCHKEY, CUSTOMERKEY, PRODUCTKEY, ORDERPROFILEKEY, EMPLOYEEKEY, ORDERNO),
	FOREIGN KEY (DATEKEY) REFERENCES DATE_DIM(DATE_KEY),
    FOREIGN KEY (BRANCHKEY) REFERENCES BRANCH_DIM(BRANCHKEY),
    FOREIGN KEY (CUSTOMERKEY) REFERENCES CUSTOMER_DIM(CUSTOMERKEY),
    FOREIGN KEY (PRODUCTKEY) REFERENCES PRODUCT_DIM(PRODUCTKEY),
    FOREIGN KEY (ORDERPROFILEKEY) REFERENCES ORDER_PROFILE_DIM(ORDER_PROFILE_KEY),
    FOREIGN KEY (EMPLOYEEKEY) REFERENCES EMPLOYEE_DIM(EMPLOYEEKEY),
	CONSTRAINT UQ_SALES_GRAIN UNIQUE (ORDERNO, PRODUCTKEY)
);
 
 
--------------------------------------------------------------------------------
-- FACT INITIAL LOAD
--
-- THE KEY PATTERN: SCD2 dimensions are joined on natural key AND
--   order date BETWEEN EFFECTIVESTARTDATE AND EFFECTIVEENDDATE.
--   NOT on FLAG = 'Y'. Joining on the current flag would attach a 2021 order
--   to the customer's 2026 membership tier, which destroys SCD2 entirely.
--
-- DATE_DIM uses a sequence surrogate, so the date key must be looked up by
-- joining on CAL_DATE rather than computed from the timestamp.
--------------------------------------------------------------------------------
INSERT INTO SALES_FACT (
    DATEKEY, BRANCHKEY, CUSTOMERKEY, PRODUCTKEY, ORDERPROFILEKEY, EMPLOYEEKEY,
    ORDERNO, UNITSELLINGPRICE, PURCHASEQUANTITY, GROSSLINEAMOUNT,
    ALLOCATEDDISCOUNTAMOUNT, NETLINEAMOUNT, RATINGSCORE,
    REFUNDQUANTITY, REFUNDTOTAL)
SELECT
    d.DATE_KEY,
    b.BRANCHKEY,
    NVL(c.CUSTOMERKEY, -1),
    p.PRODUCTKEY,
    op.ORDER_PROFILE_KEY,
    e.EMPLOYEEKEY,
    o.order_no,
    od.unit_price,
    od.quantity,
    od.subtotal,
    ROUND(o.discount_amount * od.subtotal / NULLIF(o.subtotal_amount, 0), 2),
    od.subtotal - ROUND(o.discount_amount * od.subtotal / NULLIF(o.subtotal_amount, 0), 2),
    rt.score,
    NVL(rr.return_quantity, 0),
    NVL(rr.refund_amount, 0)
FROM        WAREHOUSEDATA.Order_Detail od
JOIN        WAREHOUSEDATA.Orders o ON o.order_no = od.order_no
JOIN        DATE_DIM d ON d.CAL_DATE = TRUNC(CAST(o.order_date_time AS DATE))
JOIN        BRANCH_DIM b ON b.BRANCHID = o.branch_id
                        AND CAST(o.order_date_time AS DATE) BETWEEN b.EFFECTIVESTARTDATE AND b.EFFECTIVEENDDATE
LEFT JOIN   CUSTOMER_DIM c ON c.CUSTOMERID = o.customer_id
                          AND CAST(o.order_date_time AS DATE) BETWEEN c.EFFECTIVESTARTDATE AND c.EFFECTIVEENDDATE
JOIN        PRODUCT_DIM p ON p.ITEMID = od.item_id
                         AND CAST(o.order_date_time AS DATE) BETWEEN p.EFFECTIVESTARTDATE AND p.EFFECTIVEENDDATE
JOIN        EMPLOYEE_DIM e ON e.EMPLOYEEID = o.handled_by
                          AND CAST(o.order_date_time AS DATE) BETWEEN e.EFFECTIVESTARTDATE AND e.EFFECTIVEENDDATE
JOIN        ORDER_PROFILE_DIM op ON op.order_type = o.order_type
                                AND op.service_type = o.service_type
                                AND op.order_status = o.order_status
                                AND op.mykasih_purchase = CASE WHEN o.is_mykasih_purchase = 'Y' THEN 'MyKasih Subsidised' ELSE 'Standard Purchase' END
LEFT JOIN   WAREHOUSEDATA.Rating rt ON rt.order_no = od.order_no AND rt.item_id = od.item_id
LEFT JOIN   WAREHOUSEDATA.Return_Request rr ON rr.order_no = od.order_no AND rr.item_id = od.item_id;
COMMIT;
