-- Create Dimension
CREATE TABLE BRANCH_DIM(
	BRANCHKEY NUMBER PRIMARY KEY,
	BRANCHID VARCHAR2(20),
	BRANCHNAME VARCHAR2(100),
	BRANCHAREA VARCHAR2(50),
	STATUS VARCHAR2(10),
	BRANCHOPENTIME VARCHAR2(10) ,
	BRANCHCLOSETIME VARCHAR2(10),
	EFFECTIVESTARTDATE DATE NOT NULL,
	EFFECTIVEENDDATE DATE DEFAULT DATE '9999-12-31' NOT NULL,
	FLAG VARCHAR2(1) DEFAULT 'Y' NOT NULL,
	FOREIGN KEY (BRANCHID) REFERENCES WAREHOUSEDATA.BRANCH(BRANCH_ID)
);

CREATE TABLE ORDER_PROFILE_DIM(
	order_profile_key   NUMBER(6)     NOT NULL,
    order_type          VARCHAR2(20)  NOT NULL,   -- 'Online' / 'Walk-In'
    service_type        VARCHAR2(20)  NOT NULL,   -- 'Pickup' / 'Delivery'
    order_status        VARCHAR2(20)  NOT NULL,   -- 'Processing','Completed','Cancelled'
    mykasih_purchase    VARCHAR2(25)  NOT NULL    -- 'MyKasih Subsidised' / 'Standard Purchase'
);

CREATE TABLE EMPLOYEE_DIM(
	EMPLOYEEKEY NUMBER PRIMARY KEY,
	EMPLOYEEID VARCHAR2(20),
	EMPLOYEENAME VARCHAR2(100),
	POSITION VARCHAR2(50),
	SALARY NUMBER(8,2),
	HIREDATE DATE NOT NULL,
	RESIGNDATE DATE DEFAULT DATE '9999-12-31' NOT NULL,
	EFFECTIVESTARTDATE DATE NOT NULL,
	EFFECTIVEENDDATE DATE DEFAULT DATE '9999-12-31' NOT NULL,
	FLAG VARCHAR2(1) DEFAULT 'Y' NOT NULL CHECK (FLAG IN ('Y', 'N')),
	FOREIGN KEY (EMPLOYEEID) REFERENCES WAREHOUSEDATA.EMPLOYEE(EMPLOYEE_ID)
);

CREATE TABLE PRODUCT_DIM(
	PRODUCTKEY NUMBER PRIMARY KEY,
	ITEMID VARCHAR2(20),
	SUPPLIERID VARCHAR2(20),
	SUPPLIERNAME VARCHAR2(100),
	ITEMNAME VARCHAR2(100),
	UNITPRICE NUMBER(7,2),
	UNITOFMEASURE VARCHAR2(20),
	UNITWEIGHT NUMBER(7,3),
	IS_MYKASIH_ELIGIBLE CHAR(1),
	PRODUCTSTATUS VARCHAR2(10),
	EFFECTIVESTARTDATE DATE NOT NULL,
	EFFECTIVEENDDATE DATE DEFAULT DATE '9999-12-31' NOT NULL,
	FLAG VARCHAR2(1) DEFAULT 'Y' NOT NULL CHECK (FLAG IN ('Y', 'N')),
	FOREIGN KEY (ITEMID) REFERENCES WAREHOUSEDATA.ITEM(ITEM_ID),
	FOREIGN KEY (SUPPLIERID) REFERENCES WAREHOUSEDATA.SUPPLIER(SUPPLIER_ID)
);

CREATE TABLE CUSTOMER_DIM(
	CUSTOMERKEY NUMBER PRIMARY KEY,
	CUSTOMERID VARCHAR2(20),
	MEMBERID VARCHAR2(20),
	NAME VARCHAR2(100),
	GENDER VARCHAR2(10),
	MEMBERSHIPTYPE VARCHAR2(10),
	POINTRATE NUMBER(3,2),
	EFFECTIVESTARTDATE DATE NOT NULL,
	EFFECTIVEENDDATE DATE DEFAULT DATE '9999-12-31' NOT NULL,
	FLAG VARCHAR2(1) DEFAULT 'Y' NOT NULL CHECK (FLAG IN ('Y', 'N')),
	FOREIGN KEY (CUSTOMERID) REFERENCES WAREHOUSEDATA.CUSTOMER(CUSTOMER_ID)
);

CREATE TABLE COURIER_DIM(
	COURIERKEY NUMBER PRIMARY KEY,
	COURIERID VARCHAR2(20),
	COURIERNAME VARCHAR2(100),
	DELIVERYCOMPANYID VARCHAR2(20),
	DELIVERYCOMPANYNAME VARCHAR2(100),
	EFFECTIVESTARTDATE DATE NOT NULL,
	EFFECTIVEENDDATE DATE DEFAULT DATE '9999-12-31' NOT NULL,
	FLAG VARCHAR2(1) DEFAULT 'Y' NOT NULL CHECK (FLAG IN ('Y', 'N')),
	FOREIGN KEY (COURIERID) REFERENCES WAREHOUSEDATA.COURIER(COURIER_ID),
	FOREIGN KEY (DELIVERYCOMPANYID) REFERENCES WAREHOUSEDATA.DELIVERY_COMPANY(DELIVERY_COMPANY_ID)
);

CREATE TABLE DELIVERY_STATUS_DIM(
	DELIVERY_STATUS_KEY NUMBER PRIMARY KEY,
	DELIVERY_STATUS VARCHAR2(20)
);

CREATE TABLE REWARD_DIM(
	REWARD_KEY NUMBER PRIMARY KEY,
	REWARDID VARCHAR2(20),
	REWARDNAME VARCHAR2(100),
	REWARDTYPE VARCHAR2(20),
	POINTSREQUIRED NUMBER(10,0),
	REWARDVALUE NUMBER(7,2),
	REWARDSTATUS VARCHAR2(10),
	EFFECTIVESTARTDATE DATE NOT NULL,
	EFFECTIVEENDDATE DATE DEFAULT DATE '9999-12-31' NOT NULL,
	FLAG VARCHAR2(1) DEFAULT 'Y' NOT NULL CHECK (FLAG IN ('Y', 'N')),
	FOREIGN KEY (REWARDID) REFERENCES WAREHOUSEDATA.REWARD(REWARD_ID)
);
