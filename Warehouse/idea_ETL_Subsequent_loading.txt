1. create dirty-data reject table to store all reject data and the reason reject
CREATE TABLE ETL_REJECT_LOG (
    REJECT_ID      NUMBER PRIMARY KEY,                             
    BATCH_FROM     DATE NOT NULL,   						        --load from when to when
    BATCH_TO       DATE NOT NULL, 
    SOURCE_TABLE   VARCHAR2(30) NOT NULL,						-- source table that reject  	e.g DELIVERY_REQUEST
    SOURCE_KEY     VARCHAR2(50),							-- what key got error 		e.g. DR5001    
    REASON         VARCHAR2(250) NOT NULL,						-- what reason reject    	e.g.  Delivered date earlier than order date
    REJECTED_AT    TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL
);
