-- Query 1: Return Reason Analysis by Product Category
CREATE OR REPLACE FUNCTION fn_risk_band (
    p_return_rate IN NUMBER,
    p_avg_score IN NUMBER,
    p_cases IN NUMBER
) RETURN VARCHAR2
IS
BEGIN
    IF NVL(p_cases, 0) = 0 THEN
        RETURN 'NO ISSUE';
    ELSIF NVL(p_return_rate, 0) >= 2 OR NVL(p_avg_score, 5) <= 2 THEN
        RETURN 'CRITICAL';
    ELSIF NVL(p_return_rate, 0) >= 1.5  OR NVL(p_avg_score, 5) <= 3 THEN
        RETURN 'WATCH';
    ELSE
        RETURN 'HEALTHY';
    END IF;
END fn_risk_band;
/



CREATE OR REPLACE VIEW vw_item_return_analysis AS
SELECT i.item_id,
       i.name AS item_name,
       i.category,
       s.company_name AS supplier_name,
       NVL(sold.units_sold, 0) AS units_sold,
       NVL(sold.sales_value, 0) AS sales_value,
       NVL(ret.return_cases, 0) AS return_cases,
       NVL(ret.units_returned, 0) AS units_returned,
       NVL(ret.refund_value, 0) AS refund_value,
       ret.top_reason,
       ROUND(NVL(ret.units_returned, 0) /
             NULLIF(sold.units_sold, 0) * 100, 2) AS return_rate_pct,
       ROUND(NVL(ret.refund_value, 0) /
             NULLIF(sold.sales_value, 0) * 100, 2) AS refund_rate_pct,
       rat.avg_score,
       NVL(rat.rating_count, 0) AS rating_count,
       NVL(rat.low_ratings, 0) AS low_ratings,
       fn_risk_band(ROUND(NVL(ret.units_returned, 0) /
                          NULLIF(sold.units_sold, 0) * 100, 2),
                    rat.avg_score,
                    NVL(ret.return_cases, 0)) AS risk_band
FROM   Item i
JOIN   Supplier s ON s.supplier_id = i.supplier_id
LEFT JOIN (SELECT od.item_id,
                  SUM(od.quantity) AS units_sold,
                  SUM(od.subtotal) AS sales_value
           FROM   Order_Detail od
           JOIN   Orders o ON o.order_no = od.order_no
           WHERE  o.order_status = 'Completed'
           GROUP  BY od.item_id) sold ON sold.item_id = i.item_id
LEFT JOIN (SELECT item_id,
                  SUM(cases) AS return_cases,
                  SUM(units) AS units_returned,
                  SUM(refunds) AS refund_value,
                  MAX(reason) KEEP (DENSE_RANK LAST ORDER BY cases) AS top_reason
           FROM  (SELECT item_id,
                         reason,
                         COUNT(*) AS cases,
                         SUM(return_quantity) AS units,
                         SUM(CASE WHEN status = 'Refunded'
                                  THEN refund_amount ELSE 0 END) AS refunds
                  FROM   Return_Request
                  GROUP  BY item_id, reason)
           GROUP  BY item_id) ret ON ret.item_id = i.item_id
LEFT JOIN (SELECT item_id,
                  ROUND(AVG(score), 2) AS avg_score,
                  COUNT(*) AS rating_count,
                  SUM(CASE WHEN score <= 2 THEN 1 ELSE 0 END) AS low_ratings
           FROM   Rating
           GROUP  BY item_id) rat ON rat.item_id = i.item_id;


SET SERVEROUTPUT ON SIZE UNLIMITED FORMAT WRAPPED
SET TAB OFF
SET LINESIZE 135
SET PAGESIZE 90

COLUMN category FORMAT A26 HEADING 'Product Category'
COLUMN reason FORMAT A28 HEADING 'Reason Given by the Customer'
COLUMN fault_type FORMAT A17 HEADING 'Fault Type'
COLUMN branches_hit FORMAT 999 HEADING 'Branches|Affected'
COLUMN cases FORMAT 9,999 HEADING 'Cases'
COLUMN units FORMAT 9,999 HEADING 'Units'
COLUMN refund_value FORMAT 99,990.00 HEADING 'Refunded|RM'
COLUMN avg_refund FORMAT 990.00 HEADING 'Avg per|Case RM'
COLUMN pct_cases FORMAT 990.9 HEADING '% of|Returns'
COLUMN pending FORMAT 999 HEADING 'Still|Pending'

TTITLE CENTER '88 SPEEDMART SDN BHD - RETURN REASON BY PRODUCT CATEGORY' SKIP 1 -
       CENTER '========================================================' SKIP 2

BREAK ON category SKIP 1 ON REPORT
COMPUTE SUM LABEL 'Category total' OF cases units refund_value ON category
COMPUTE SUM LABEL 'ALL CATEGORIES' OF cases units refund_value ON REPORT

SELECT v.category,
       rr.reason,
       CASE WHEN rr.reason = 'Changed mind after purchase'
            THEN 'Customer choice'
            ELSE 'Shop or supplier'
       END AS fault_type,
       COUNT(DISTINCT o.branch_id) AS branches_hit,
       COUNT(*) AS cases,
       SUM(rr.return_quantity) AS units,
       SUM(CASE WHEN rr.status = 'Refunded' THEN rr.refund_amount ELSE 0 END) AS refund_value,
       ROUND(SUM(CASE WHEN rr.status = 'Refunded' THEN rr.refund_amount ELSE 0 END)
             / COUNT(*), 2) AS avg_refund,
       ROUND(COUNT(*) * 100 / (SELECT COUNT(*) FROM Return_Request), 1) AS pct_cases,
       SUM(CASE WHEN rr.status = 'Requested' THEN 1 ELSE 0 END) AS pending
FROM   vw_item_return_analysis v
JOIN   Return_Request rr ON rr.item_id  = v.item_id
JOIN   Orders o          ON o.order_no  = rr.order_no
JOIN   Branch b          ON b.branch_id = o.branch_id
GROUP  BY v.category, rr.reason
ORDER  BY v.category, refund_value DESC;




-- Query 2 : Low Rating Service Recovery List

CREATE OR REPLACE VIEW vw_low_rating_recovery AS
SELECT r.rating_id,
       r.rating_date,
       TRUNC(SYSDATE) - TRUNC(r.rating_date) AS days_since,
       r.score,
       NVL(r.comments, '(no comment)') AS comments,
       B.branch_id, b.name AS branch_name,
       o.order_no,
       I.item_id, i.name AS item_name, i.category,
       NVL(m.name, 'Walk-in customer') AS customer_name,
       COALESCE(m.phone_no, dr.recipient_phone_no, 'No Number') AS contact,
       NVL(m.membership_type, 'Non-member') AS membership_type,
       CASE WHEN rr.return_id IS NULL THEN 'No return'
            ELSE rr.status
       END AS return_status,
       CASE WHEN m.phone_no IS NULL AND dr.recipient_phone_no IS NULL
                 THEN 'P5 - No Contact'
            WHEN r.score = 1 AND NVL(m.membership_type, 'Non-member') = 'VIP'
                 THEN 'P1 - Manager Call'
            WHEN TRUNC(SYSDATE) - TRUNC(r.rating_date) > 14 AND rr.return_id IS NULL
                 THEN 'P2 - Phone Call'
            WHEN r.score = 1
                 THEN 'P3 - Phone Call'
            ELSE 'P4 - Email'
       END AS recovery_action
FROM   Rating r
JOIN   Item     i  ON i.item_id     = r.item_id
JOIN   Orders   o  ON o.order_no    = r.order_no
JOIN   Branch   b  ON b.branch_id   = o.branch_id
JOIN   Customer c  ON c.customer_id = o.customer_id
LEFT   JOIN Member m ON m.member_id = c.member_id
LEFT   JOIN Delivery_Request dr ON dr.order_no = r.order_no
LEFT   JOIN Return_Request rr ON rr.order_no = r.order_no AND rr.item_id = r.item_id
WHERE  r.score <= 2
AND    r.rating_date >= ADD_MONTHS(SYSDATE, -3);


SET LINESIZE 225
SET PAGESIZE 60

COLUMN branch_name     FORMAT A30 HEADING 'Branch'
COLUMN rating_id       FORMAT A9  HEADING 'Rating|ID'
COLUMN rated_on        FORMAT A10 HEADING 'Rated|On'
COLUMN days_since      FORMAT 999 HEADING 'Days|Ago'
COLUMN item_name       FORMAT A22 HEADING 'Item'
COLUMN score           FORMAT 9   HEADING 'Star'
COLUMN comments        FORMAT A50 HEADING 'Customer Comments'
COLUMN customer_name   FORMAT A25 HEADING 'Customer'
COLUMN contact         FORMAT A16 HEADING 'Contact Number'
COLUMN membership_type FORMAT A10 HEADING 'Member|Type'
COLUMN return_status   FORMAT A13 HEADING 'Return|Lodged?'
COLUMN recovery_action FORMAT A18 HEADING 'Recovery|Action'

TTITLE CENTER '88 SPEEDMART SDN BHD - LOW RATING SERVICE RECOVERY LIST' SKIP 1 -
       CENTER '=======================================================' SKIP 2

SELECT branch_name,
       rating_id,
       TO_CHAR(rating_date, 'DD-MON-YY') AS rated_on,
       days_since,
       SUBSTR(item_name, 1, 22) AS item_name,
       score,
       SUBSTR(comments, 1, 50) AS comments,
       customer_name,
       contact,
       membership_type,
       return_status,
       recovery_action
FROM   vw_low_rating_recovery
ORDER  BY branch_name,
          CASE WHEN score = 1 THEN 1 ELSE 2 END,
          days_since DESC;
