-- function 1
CREATE OR REPLACE FUNCTION fn_calc_refund (
    p_order_no IN Orders.order_no%TYPE,
    p_item_id  IN Item.item_id%TYPE,
    p_quantity IN NUMBER,
    p_reason   IN Return_Request.reason%TYPE
) RETURN NUMBER
IS
    v_unit_price Order_Detail.unit_price%TYPE;
    v_line_total Order_Detail.subtotal%TYPE;
    v_refund     NUMBER(10,2);
BEGIN
    SELECT unit_price, subtotal
    INTO v_unit_price, v_line_total
    FROM Order_Detail
    WHERE order_no = p_order_no
    AND item_id  = p_item_id;

    IF p_reason = 'Changed mind after purchase' THEN
        v_refund := p_quantity * v_unit_price * 0.80;   -- 20% restocking fee
    ELSE
        v_refund := p_quantity * v_unit_price;          -- shop / supplier fault
    END IF;

    RETURN ROUND(LEAST(v_refund, v_line_total), 2);
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20045,
            'Item ' || p_item_id || ' was never bought on order ' || p_order_no || '.');
END fn_calc_refund;
/

-- function 2 

CREATE OR REPLACE FUNCTION fn_risk_band (
    p_return_rate IN NUMBER,
    p_avg_score IN NUMBER,
    p_cases IN NUMBER
) RETURN VARCHAR2
IS
BEGIN
    IF NVL(p_cases, 0) = 0 THEN
        RETURN 'NO ISSUE';
    ELSIF NVL(p_return_rate, 0) >= 2 OR NVL(p_avg_score, 5) <= 2 THEN
        RETURN 'CRITICAL';
    ELSIF NVL(p_return_rate, 0) >= 1.5  OR NVL(p_avg_score, 5) <= 3 THEN
        RETURN 'WATCH';
    ELSE
        RETURN 'HEALTHY';
    END IF;
END fn_risk_band;
/

-- function 3

CREATE OR REPLACE FUNCTION fn_sla_priority (
    p_days_waiting IN NUMBER,
    p_membership IN VARCHAR2,
    p_refund_value IN NUMBER
) RETURN VARCHAR2
IS
BEGIN
    IF p_refund_value >= 100 THEN
        RETURN 'P1-URGENT';
    ELSIF p_days_waiting > 3 OR p_membership = 'VIP' THEN
        RETURN 'P2-HIGH';
    ELSE
        RETURN 'P3-NORMAL';
    END IF;
END fn_sla_priority;
/


-- View table 1

CREATE OR REPLACE VIEW vw_item_return_analysis AS
SELECT i.item_id,
       i.name AS item_name,
       i.category,
       s.company_name AS supplier_name,
       NVL(sold.units_sold, 0) AS units_sold,
       NVL(sold.sales_value, 0) AS sales_value,
       NVL(ret.return_cases, 0) AS return_cases,
       NVL(ret.units_returned, 0) AS units_returned,
       NVL(ret.refund_value, 0) AS refund_value,
       ret.top_reason,
       ROUND(NVL(ret.units_returned, 0) /
             NULLIF(sold.units_sold, 0) * 100, 2) AS return_rate_pct,
       ROUND(NVL(ret.refund_value, 0) /
             NULLIF(sold.sales_value, 0) * 100, 2) AS refund_rate_pct,
       rat.avg_score,
       NVL(rat.rating_count, 0) AS rating_count,
       NVL(rat.low_ratings, 0) AS low_ratings,
       fn_risk_band(ROUND(NVL(ret.units_returned, 0) /
                          NULLIF(sold.units_sold, 0) * 100, 2),
                    rat.avg_score,
                    NVL(ret.return_cases, 0)) AS risk_band
FROM   Item i
JOIN   Supplier s ON s.supplier_id = i.supplier_id
LEFT JOIN (SELECT od.item_id,
                  SUM(od.quantity) AS units_sold,
                  SUM(od.subtotal) AS sales_value
           FROM   Order_Detail od
           JOIN   Orders o ON o.order_no = od.order_no
           WHERE  o.order_status = 'Completed'
           GROUP  BY od.item_id) sold ON sold.item_id = i.item_id
LEFT JOIN (SELECT item_id,
                  SUM(cases) AS return_cases,
                  SUM(units) AS units_returned,
                  SUM(refunds) AS refund_value,
                  MAX(reason) KEEP (DENSE_RANK LAST ORDER BY cases) AS top_reason
           FROM  (SELECT item_id,
                         reason,
                         COUNT(*) AS cases,
                         SUM(return_quantity) AS units,
                         SUM(CASE WHEN status = 'Refunded'
                                  THEN refund_amount ELSE 0 END) AS refunds
                  FROM   Return_Request
                  GROUP  BY item_id, reason)
           GROUP  BY item_id) ret ON ret.item_id = i.item_id
LEFT JOIN (SELECT item_id,
                  ROUND(AVG(score), 2) AS avg_score,
                  COUNT(*) AS rating_count,
                  SUM(CASE WHEN score <= 2 THEN 1 ELSE 0 END) AS low_ratings
           FROM   Rating
           GROUP  BY item_id) rat ON rat.item_id = i.item_id;

CREATE OR REPLACE VIEW vw_low_rating_recovery AS
SELECT r.rating_id,
       r.rating_date,
       TRUNC(SYSDATE) - TRUNC(r.rating_date) AS days_since,
       r.score,
       NVL(r.comments, '(no comment)') AS comments,
       B.branch_id, b.name AS branch_name,
       o.order_no,
       I.item_id, i.name AS item_name, i.category,
       NVL(m.name, 'Walk-in customer') AS customer_name,
       COALESCE(m.phone_no, dr.recipient_phone_no, 'No Number') AS contact,
       NVL(m.membership_type, 'Non-member') AS membership_type,
       CASE WHEN rr.return_id IS NULL THEN 'No return'
            ELSE rr.status
       END AS return_status,
       CASE WHEN m.phone_no IS NULL AND dr.recipient_phone_no IS NULL
                 THEN 'P5 - No Contact'
            WHEN r.score = 1 AND NVL(m.membership_type, 'Non-member') = 'VIP'
                 THEN 'P1 - Manager Call'
            WHEN TRUNC(SYSDATE) - TRUNC(r.rating_date) > 14 AND rr.return_id IS NULL
                 THEN 'P2 - Phone Call'
            WHEN r.score = 1
                 THEN 'P3 - Phone Call'
            ELSE 'P4 - Email'
       END AS recovery_action
FROM   Rating r
JOIN   Item     i  ON i.item_id     = r.item_id
JOIN   Orders   o  ON o.order_no    = r.order_no
JOIN   Branch   b  ON b.branch_id   = o.branch_id
JOIN   Customer c  ON c.customer_id = o.customer_id
LEFT   JOIN Member m ON m.member_id = c.member_id
LEFT   JOIN Delivery_Request dr ON dr.order_no = r.order_no
LEFT   JOIN Return_Request rr ON rr.order_no = r.order_no AND rr.item_id = r.item_id
WHERE  r.score <= 2
AND    r.rating_date >= ADD_MONTHS(SYSDATE, -3);

CREATE OR REPLACE VIEW vw_pending_return_queue AS
SELECT rr.return_id,
       rr.request_date,
       TRUNC(SYSDATE) - TRUNC(rr.request_date) AS days_waiting,
       b.branch_id,
       b.name AS branch_name,
       o.order_no,
       TRUNC(o.order_date_time) AS order_date,
       o.order_type,
       i.item_id,
       i.name AS item_name,
       i.category,
       rr.return_quantity,
       rr.reason,
       od.unit_price,
       ROUND(rr.return_quantity * od.unit_price, 2) AS refund_exposure,
       NVL(m.name, 'Walk-in customer') AS customer_name,
       NVL(m.membership_type, 'Non-member') AS membership_type,
       fn_sla_priority(TRUNC(SYSDATE) - TRUNC(rr.request_date),
                       NVL(m.membership_type, 'Non-member'),
                       ROUND(rr.return_quantity * od.unit_price, 2)) AS priority
FROM   Return_Request rr
JOIN   Order_Detail od ON od.order_no = rr.order_no AND od.item_id = rr.item_id
JOIN   Orders  o ON o.order_no  = rr.order_no
JOIN   Item    i ON i.item_id   = rr.item_id
JOIN   Branch  b ON b.branch_id = o.branch_id
JOIN   Customer c ON c.customer_id = o.customer_id
LEFT   JOIN Member m ON m.member_id = c.member_id
WHERE  rr.status = 'Requested';

CREATE OR REPLACE PROCEDURE prc_process_return (
    p_order_no IN Orders.order_no%TYPE,
    p_item_id  IN Item.item_id%TYPE,
    p_reason   IN Return_Request.reason%TYPE,
    p_quantity IN Return_Request.return_quantity%TYPE,
    p_reviewer IN Employee.employee_id%TYPE,
    p_decision IN VARCHAR2 DEFAULT 'Approved'
) AS
    v_emp_name     Employee.name%TYPE;
    v_emp_position Employee.position%TYPE;
    v_emp_status   Employee.status%TYPE;
    v_item_name    Item.name%TYPE;
    v_return_id    Return_Request.return_id%TYPE;
    v_refund       NUMBER(10,2);
BEGIN
    SELECT e.name, e.position, e.status
    INTO   v_emp_name, v_emp_position, v_emp_status
    FROM   Employee e
    WHERE  e.employee_id = p_reviewer;

    SELECT i.name INTO v_item_name
    FROM   Item i
    WHERE  i.item_id = p_item_id;

    IF p_decision NOT IN ('Requested', 'Approved', 'Rejected', 'Refunded') THEN
        RAISE_APPLICATION_ERROR(-20040,
            'Decision must be Requested, Approved, Rejected or Refunded - got "' || p_decision || '".');
    ELSIF NVL(p_quantity, 0) <= 0 THEN
        RAISE_APPLICATION_ERROR(-20041, 'Return quantity must be at least one unit.');
    ELSIF v_emp_position NOT IN ('Supervisor', 'Manager') OR v_emp_status <> 'Active' THEN
        RAISE_APPLICATION_ERROR(-20042,
            v_emp_name || ' is a ' || v_emp_position || ' (' || v_emp_status ||
            ') and cannot approve a refund. Please call a Supervisor or Manager.');
    END IF;

    v_return_id := 'RTN' || LPAD(return_seq.NEXTVAL, 6, '0');
    v_refund    := fn_calc_refund(p_order_no, p_item_id, p_quantity, p_reason);

    INSERT INTO Return_Request (return_id, reason, request_date, status,
                                refund_amount, remarks, return_quantity,
                                order_no, item_id, reviewed_by)
    VALUES (v_return_id, p_reason, SYSDATE, p_decision, v_refund,
            'Lodged at counter by ' || v_emp_name ||
            CASE WHEN p_decision <> 'Requested'
                 THEN ' | ' || p_decision || ' on ' || TO_CHAR(SYSDATE, 'DD-MON-YYYY HH24:MI')
            END,
            p_quantity, p_order_no, p_item_id, p_reviewer);

    IF p_decision = 'Refunded' THEN
        UPDATE Payment
        SET    payment_status = 'Refunded'
        WHERE  order_no = p_order_no AND payment_status = 'Paid';
    END IF;

    COMMIT;

    DBMS_OUTPUT.PUT_LINE(RPAD('=', 62, '='));
    DBMS_OUTPUT.PUT_LINE(LPAD('88 SPEEDMART - RETURN SLIP', 44));
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 62, '='));
    DBMS_OUTPUT.PUT_LINE(RPAD('Return ID',   18) || ': ' || v_return_id);
    DBMS_OUTPUT.PUT_LINE(RPAD('Order No',    18) || ': ' || p_order_no);
    DBMS_OUTPUT.PUT_LINE(RPAD('Item',        18) || ': ' || v_item_name || ' (' || p_item_id || ')');
    DBMS_OUTPUT.PUT_LINE(RPAD('Quantity',    18) || ': ' || p_quantity);
    DBMS_OUTPUT.PUT_LINE(RPAD('Reason',      18) || ': ' || p_reason);
    DBMS_OUTPUT.PUT_LINE(RPAD('Refund',      18) || ': RM ' || TO_CHAR(v_refund, 'FM9,999,990.00'));
    DBMS_OUTPUT.PUT_LINE(RPAD('Decision',    18) || ': ' || p_decision);
    DBMS_OUTPUT.PUT_LINE(RPAD('Reviewed by', 18) || ': ' || v_emp_name || ' (' || v_emp_position || ')');
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 62, '='));

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        ROLLBACK;
        RAISE_APPLICATION_ERROR(-20043,
            'Employee ' || p_reviewer || ' or item ' || p_item_id || ' does not exist.');
    WHEN DUP_VAL_ON_INDEX THEN
        ROLLBACK;
        RAISE_APPLICATION_ERROR(-20044,
            'Item ' || p_item_id || ' on order ' || p_order_no || ' already has a return request on file.');
    WHEN OTHERS THEN
        ROLLBACK;
        RAISE;
END prc_process_return;
/

CREATE OR REPLACE PROCEDURE prc_rate_item (
    p_order_no IN Orders.order_no%TYPE,
    p_item_id  IN Item.item_id%TYPE,
    p_score    IN Rating.score%TYPE,
    p_comments IN Rating.comments%TYPE DEFAULT NULL
) AS
    e_orphan_line EXCEPTION;
    PRAGMA EXCEPTION_INIT(e_orphan_line, -2291);

    e_order_open  EXCEPTION;                      
    v_status    Orders.order_status%TYPE;
    v_item_name Item.name%TYPE;
    v_rating_id Rating.rating_id%TYPE;
    v_avg       NUMBER(3,2);
    v_count     NUMBER;
BEGIN
    SELECT o.order_status, i.name
    INTO   v_status, v_item_name
    FROM   Orders o, Item i
    WHERE  o.order_no = p_order_no
    AND    i.item_id  = p_item_id;

    IF p_score IS NULL OR p_score NOT BETWEEN 1 AND 5 THEN
        RAISE_APPLICATION_ERROR(-20046,
            'A rating must be a whole number from 1 to 5 - got "' || p_score || '".');
    ELSIF p_score <= 2 AND p_comments IS NULL THEN
        RAISE_APPLICATION_ERROR(-20047,
            'A rating of 1 or 2 star must come with a comment so the buyer can act on it.');
    ELSIF v_status <> 'Completed' THEN
        RAISE e_order_open;                       
    END IF;

    v_rating_id := 'RT' || LPAD(rating_seq.NEXTVAL, 5, '0');

    INSERT INTO Rating (rating_id, score, comments, rating_date, order_no, item_id)
    VALUES (v_rating_id, p_score, p_comments, SYSDATE, p_order_no, p_item_id);

    COMMIT;

    SELECT ROUND(AVG(score), 2), COUNT(*)
    INTO   v_avg, v_count
    FROM   Rating
    WHERE  item_id = p_item_id;

    DBMS_OUTPUT.PUT_LINE(RPAD('-', 62, '-'));
    DBMS_OUTPUT.PUT_LINE('THANK YOU FOR RATING ' || UPPER(v_item_name));
    DBMS_OUTPUT.PUT_LINE(RPAD('-', 62, '-'));
    DBMS_OUTPUT.PUT_LINE(RPAD('Rating ID', 16) || ': ' || v_rating_id);
    DBMS_OUTPUT.PUT_LINE(RPAD('Your score', 16) || ': ' ||
                         RPAD('*', p_score, '*') || RPAD('.', 5 - p_score, '.') ||
                         '  (' || p_score || '/5)');
    DBMS_OUTPUT.PUT_LINE(RPAD('Your comment', 16) || ': ' || NVL(p_comments, '(none given)'));
    DBMS_OUTPUT.PUT_LINE(RPAD('New average', 16) || ': ' ||
                         TO_CHAR(v_avg, 'FM90.00') || ' from ' || v_count || ' rating(s)');
    DBMS_OUTPUT.PUT_LINE(RPAD('-', 62, '-'));

EXCEPTION
    WHEN e_order_open THEN
        DBMS_OUTPUT.PUT_LINE('CANNOT RATE YET : order ' || p_order_no || ' is ' ||
            v_status || '. You may rate an item once the order is completed.');
    WHEN e_orphan_line THEN
        DBMS_OUTPUT.PUT_LINE('INVALID LINE : item ' || p_item_id ||
            ' does not appear on order ' || p_order_no || '.');
    WHEN DUP_VAL_ON_INDEX THEN
        DBMS_OUTPUT.PUT_LINE('ALREADY RATED : you have already rated item ' || p_item_id ||
            ' on order ' || p_order_no || '. One rating per purchase.');
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('NOT FOUND : order ' || p_order_no || ' or item ' ||
            p_item_id || ' does not exist.');
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('UNEXPECTED ERROR ' || SQLCODE || ' : ' || SQLERRM);
END prc_rate_item;
/

CREATE OR REPLACE PROCEDURE rpt_return_risk (
    p_category IN Item.category%TYPE DEFAULT NULL
) AS
    CURSOR c_category IS
        SELECT DISTINCT category
        FROM   vw_item_return_analysis
        WHERE  return_cases > 0
        AND   (p_category IS NULL OR UPPER(category) = UPPER(p_category))
        ORDER  BY category;

    CURSOR c_item (cp_category Item.category%TYPE) IS
        SELECT item_id, item_name, supplier_name, units_sold, units_returned,
               return_rate_pct, refund_value, avg_score, risk_band
        FROM   vw_item_return_analysis
        WHERE  return_cases > 0
        AND    category = cp_category
        ORDER  BY return_rate_pct DESC, refund_value DESC;

    c_width  CONSTANT PLS_INTEGER := 129;
    v_rule   VARCHAR2(200) := RPAD('-', 129, '-');
    v_double VARCHAR2(200) := RPAD('=', 129, '=');

    v_cat_units  NUMBER := 0;   v_cat_refund NUMBER := 0;   v_cat_lines NUMBER := 0;
    v_all_units  NUMBER := 0;   v_all_refund NUMBER := 0;   v_all_lines NUMBER := 0;
    v_critical   NUMBER := 0;   v_watch      NUMBER := 0;

    FUNCTION ctr (p_text IN VARCHAR2) RETURN VARCHAR2 IS
    BEGIN
        RETURN LPAD(p_text, FLOOR((c_width + LENGTH(p_text)) / 2));
    END ctr;
BEGIN
    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(ctr('88 SPEEDMART SDN BHD'));
    DBMS_OUTPUT.PUT_LINE(ctr('HIGH RETURN ITEMS AND REFUND EXPOSURE'));
    DBMS_OUTPUT.PUT_LINE(ctr('Scope : ' || NVL(p_category, 'All categories')));
    DBMS_OUTPUT.PUT_LINE(ctr('As at ' || TO_CHAR(SYSDATE, 'DD-MON-YYYY HH24:MI')));
    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(' ');

    FOR cat IN c_category LOOP                      
        DBMS_OUTPUT.PUT_LINE('CATEGORY : ' || UPPER(cat.category));
        DBMS_OUTPUT.PUT_LINE(
            RPAD('Item',      8) ||
            RPAD('Product',  36) ||
            RPAD('Supplier', 31) ||
            LPAD('Sold',      9) ||
            LPAD('Ret.',      7) ||
            LPAD('Rate %',    9) ||
            LPAD('Refund RM',12) ||
            LPAD('Star',      7) ||
            LPAD('Risk',     10));
        DBMS_OUTPUT.PUT_LINE(v_rule);

        v_cat_units := 0;  v_cat_refund := 0;  v_cat_lines := 0;

        FOR itm IN c_item(cat.category) LOOP         
            DBMS_OUTPUT.PUT_LINE(
                RPAD(itm.item_id, 8) ||
                RPAD(SUBSTR(itm.item_name, 1, 34), 36) ||
                RPAD(SUBSTR(itm.supplier_name, 1, 29), 31) ||
                LPAD(TO_CHAR(itm.units_sold,      'FM999,990'),    9) ||
                LPAD(TO_CHAR(itm.units_returned,  'FM9,990'),      7) ||
                LPAD(TO_CHAR(itm.return_rate_pct, 'FM990.00'),     9) ||
                LPAD(TO_CHAR(itm.refund_value,    'FM999,990.00'),12) ||
                LPAD(NVL(TO_CHAR(itm.avg_score,   'FM90.00'), 'n/a'), 7) ||
                LPAD(itm.risk_band, 10));

            v_cat_units  := v_cat_units  + itm.units_returned;
            v_cat_refund := v_cat_refund + itm.refund_value;
            v_cat_lines  := v_cat_lines  + 1;
            IF    itm.risk_band = 'CRITICAL' THEN v_critical := v_critical + 1;
            ELSIF itm.risk_band = 'WATCH'    THEN v_watch    := v_watch    + 1;
            END IF;
        END LOOP;

        DBMS_OUTPUT.PUT_LINE(v_rule);
        DBMS_OUTPUT.PUT_LINE(
            RPAD('Subtotal  (' || v_cat_lines || ' item(s))', 84) ||
            LPAD(TO_CHAR(v_cat_units,  'FM9,990'),       7) ||
            LPAD(TO_CHAR(v_cat_refund, 'FM999,990.00'), 21));
        DBMS_OUTPUT.PUT_LINE(' ');

        v_all_units  := v_all_units  + v_cat_units;
        v_all_refund := v_all_refund + v_cat_refund;
        v_all_lines  := v_all_lines  + v_cat_lines;
    END LOOP;

    IF v_all_lines = 0 THEN
        DBMS_OUTPUT.PUT_LINE(ctr('No returns recorded for ' ||
            NVL(p_category, 'any category') || '.'));
        DBMS_OUTPUT.PUT_LINE(v_double);
        RETURN;
    END IF;

    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(ctr('SUMMARY'));
    DBMS_OUTPUT.PUT_LINE(v_rule);
    DBMS_OUTPUT.PUT_LINE(RPAD('Items with returns', 30) || ': ' ||
        LPAD(v_all_lines, 12));
    DBMS_OUTPUT.PUT_LINE(RPAD('   CRITICAL band', 30) || ': ' ||
        LPAD(v_critical, 12) || '   (' ||
        TO_CHAR(ROUND(v_critical / v_all_lines * 100, 1), 'FM990.0') || '%)');
    DBMS_OUTPUT.PUT_LINE(RPAD('   WATCH band', 30) || ': ' ||
        LPAD(v_watch, 12) || '   (' ||
        TO_CHAR(ROUND(v_watch / v_all_lines * 100, 1), 'FM990.0') || '%)');
    DBMS_OUTPUT.PUT_LINE(RPAD('Total units returned', 30) || ': ' ||
        LPAD(TO_CHAR(v_all_units, 'FM999,990'), 12));
    DBMS_OUTPUT.PUT_LINE(RPAD('Total refund exposure', 30) || ': ' ||
        LPAD('RM ' || TO_CHAR(v_all_refund, 'FM999,999,990.00'), 12));
    DBMS_OUTPUT.PUT_LINE(RPAD('Average refund per item', 30) || ': ' ||
        LPAD('RM ' || TO_CHAR(ROUND(v_all_refund / v_all_lines, 2), 'FM999,990.00'), 12));
    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(LPAD('End of Report 1', c_width));

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('REPORT FAILED ' || SQLCODE || ' : ' || SQLERRM);
END rpt_return_risk;

-- report 2
CREATE OR REPLACE PROCEDURE rpt_pending_returns (
    p_branch_id IN Branch.branch_id%TYPE DEFAULT NULL
) AS
    CURSOR c_branch IS
        SELECT   branch_id,
                 branch_name,
                 COUNT(*)              AS cases,
                 SUM(refund_exposure)  AS exposure
        FROM     vw_pending_return_queue
        WHERE   (p_branch_id IS NULL OR branch_id = p_branch_id)
        GROUP BY branch_id, branch_name
        ORDER BY SUM(refund_exposure) DESC;

    CURSOR c_case (cp_branch Branch.branch_id%TYPE) IS
        SELECT return_id, order_no, item_name, return_quantity, reason,
               customer_name, membership_type, days_waiting,
               refund_exposure, priority
        FROM   vw_pending_return_queue
        WHERE  branch_id = cp_branch
        ORDER  BY CASE priority WHEN 'P1-URGENT' THEN 1
                                WHEN 'P2-HIGH'   THEN 2
                                ELSE 3 END,
                  days_waiting DESC, refund_exposure DESC;

    c_width  CONSTANT PLS_INTEGER := 155;
    v_rule   VARCHAR2(200) := RPAD('-', 155, '-');
    v_double VARCHAR2(200) := RPAD('=', 155, '=');

    v_branch_sum NUMBER(12,2) := 0;   v_branch_cnt NUMBER := 0;
    v_grand_sum  NUMBER(12,2) := 0;   v_grand_cnt  NUMBER := 0;
    v_p1 NUMBER := 0;  v_p2 NUMBER := 0;  v_p3 NUMBER := 0;
    v_oldest NUMBER := 0;

    FUNCTION ctr (p_text IN VARCHAR2) RETURN VARCHAR2 IS
    BEGIN
        RETURN LPAD(p_text, FLOOR((c_width + LENGTH(p_text)) / 2));
    END ctr;

    FUNCTION money_line (p_label IN VARCHAR2, p_amount IN NUMBER) RETURN VARCHAR2 IS
    BEGIN
        RETURN RPAD(p_label, 134) || LPAD(TO_CHAR(p_amount, 'FM999,990.00'), 10);
    END money_line;
BEGIN
    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(ctr('88 SPEEDMART SDN BHD'));
    DBMS_OUTPUT.PUT_LINE(ctr('VOICE OF CUSTOMER - PENDING RETURN REQUESTS'));
    DBMS_OUTPUT.PUT_LINE(ctr('Scope : ' || NVL(p_branch_id, 'All branches')));
    DBMS_OUTPUT.PUT_LINE(ctr('As at ' || TO_CHAR(SYSDATE, 'DD-MON-YYYY HH24:MI')));
    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(' ');

    FOR br IN c_branch LOOP                              -- OUTER CURSOR
        DBMS_OUTPUT.PUT_LINE('BRANCH : ' || UPPER(br.branch_name) ||
                             '  (' || br.branch_id || ')');
        DBMS_OUTPUT.PUT_LINE(
            RPAD('Return',   10) ||
            RPAD('Order',    10) ||
            RPAD('Item',     51) ||
            RPAD('Qty',       5) ||
            RPAD('Reason',   29) ||
            RPAD('Customer', 24) ||
            LPAD('Days',      5) ||
            LPAD('RM',       10) ||
            '  Priority');
        DBMS_OUTPUT.PUT_LINE(v_rule);

        v_branch_sum := 0;  v_branch_cnt := 0;

        FOR r IN c_case(br.branch_id) LOOP               -- INNER CURSOR
            DBMS_OUTPUT.PUT_LINE(
                RPAD(r.return_id, 10) ||
                RPAD(r.order_no,  10) ||
                RPAD(SUBSTR(r.item_name, 1, 49), 51) ||
                RPAD(r.return_quantity, 5) ||
                RPAD(SUBSTR(r.reason, 1, 27), 29) ||
                RPAD(SUBSTR(r.customer_name, 1, 22), 24) ||
                LPAD(r.days_waiting, 5) ||
                LPAD(TO_CHAR(r.refund_exposure, 'FM999,990.00'), 10) ||
                '  ' || r.priority);

            v_branch_sum := v_branch_sum + r.refund_exposure;
            v_branch_cnt := v_branch_cnt + 1;
            v_oldest     := GREATEST(v_oldest, r.days_waiting);

            CASE r.priority
                WHEN 'P1-URGENT' THEN v_p1 := v_p1 + 1;
                WHEN 'P2-HIGH'   THEN v_p2 := v_p2 + 1;
                ELSE                  v_p3 := v_p3 + 1;
            END CASE;
        END LOOP;

        DBMS_OUTPUT.PUT_LINE(v_rule);
        DBMS_OUTPUT.PUT_LINE(
            money_line('Branch subtotal  (' || v_branch_cnt || ' case(s))',
                       v_branch_sum));
        DBMS_OUTPUT.PUT_LINE(' ');

        v_grand_sum := v_grand_sum + v_branch_sum;
        v_grand_cnt := v_grand_cnt + v_branch_cnt;
    END LOOP;

    IF v_grand_cnt = 0 THEN
        DBMS_OUTPUT.PUT_LINE(ctr('No return request is waiting for a decision.'));
        DBMS_OUTPUT.PUT_LINE(v_double);
        RETURN;
    END IF;

    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(ctr('SUMMARY'));
    DBMS_OUTPUT.PUT_LINE(v_rule);
    DBMS_OUTPUT.PUT_LINE(RPAD('Total cases waiting', 36) || ': ' || v_grand_cnt);
    DBMS_OUTPUT.PUT_LINE(RPAD('   P1 - Urgent', 36) || ': ' || RPAD(v_p1, 6) ||
        '(' || TO_CHAR(ROUND(v_p1 / v_grand_cnt * 100, 1), 'FM990.0') || '%)');
    DBMS_OUTPUT.PUT_LINE(RPAD('   P2 - High', 36) || ': ' || RPAD(v_p2, 6) ||
        '(' || TO_CHAR(ROUND(v_p2 / v_grand_cnt * 100, 1), 'FM990.0') || '%)');
    DBMS_OUTPUT.PUT_LINE(RPAD('   P3 - Normal', 36) || ': ' || RPAD(v_p3, 6) ||
        '(' || TO_CHAR(ROUND(v_p3 / v_grand_cnt * 100, 1), 'FM990.0') || '%)');
    DBMS_OUTPUT.PUT_LINE(RPAD('Longest wait', 36) || ': ' || v_oldest || ' day(s)');
    DBMS_OUTPUT.PUT_LINE(RPAD('Total refund exposure', 36) || ': RM ' ||
        TO_CHAR(v_grand_sum, 'FM999,999,990.00'));
    DBMS_OUTPUT.PUT_LINE(RPAD('Average value per case', 36) || ': RM ' ||
        TO_CHAR(ROUND(v_grand_sum / v_grand_cnt, 2), 'FM999,990.00'));
    DBMS_OUTPUT.PUT_LINE(v_double);
    DBMS_OUTPUT.PUT_LINE(LPAD('End of Report 2', c_width));

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('REPORT FAILED ' || SQLCODE || ' : ' || SQLERRM);
END rpt_pending_returns;
/
